33/// Write `content` to `path` only if the file does not exist or its
34/// current contents differ. Avoids unnecessary filesystem writes and
35/// inode updates on warm restarts.
36pub(crate) fn write_if_changed(path: &std::path::Path, content: &str) -> anyhow::Result<()> {
37    if path.exists()
38        && let Ok(existing) = std::fs::read_to_string(path)
39        && existing == content
40    {
41        return Ok(());
42    }
43    std::fs::write(path, content)?;
44    Ok(())
45}

101    /// Start kubernix by consuming the provided configuration
102    pub fn start(mut config: Config) -> Result<()> {
103        Self::prepare_env(&mut config)?;
104
105        // Bootstrap if we're not inside a nix shell
106        if Nix::is_active() {
107            if config.is_rootless() && std::env::var(ROOTLESS_ENV).as_deref() != Ok("1") {
108                return Self::reexec_rootless();
109            }
110            Self::bootstrap_cluster(config)
111        } else {
112            Nix::bootstrap(config)
113        }
114    }

116    /// Spawn a new shell into the provided configuration environment.
117    pub fn new_shell(mut config: Config) -> Result<()> {
118        Self::prepare_env(&mut config)?;
119
120        if config.is_rootless() && std::env::var(ROOTLESS_ENV).as_deref() != Ok("1") {
121            bail!(
122                "The cluster was started in rootless mode. \
123                 Run 'kubernix shell' from within the rootlesskit session."
124            )
125        }
126
127        info!(
128            "Spawning new kubernix shell in: '{}'",
129            config.root().display()
130        );
131
132        let env_file = Self::env_file(&config);
133        if !env_file.exists() {
134            bail!(
135                "Necessary environment file '{}' does not exist",
136                env_file.display()
137            )
138        }
139
140        let shell_cmd = format!(
141            ". '{}' && exec '{}'",
142            env_file.display(),
143            config.shell_ok()?
144        );
145        Nix::run(&config, &["bash", "-c", &shell_cmd])?;
146
147        info!("Bye, leaving the Kubernix environment");
148        Ok(())
149    }

151    /// Prepare the environment based on the provided config
152    fn prepare_env(config: &mut Config) -> Result<()> {
153        let env_rootless = std::env::var(ROOTLESS_ENV).as_deref() == Ok("1");
154        let rootless = if !getuid().is_root() {
155            true
156        } else if env_rootless {
157            // KUBERNIX_ROOTLESS=1 is set by reexec_rootless() inside rootlesskit.
158            // Only honor it when actually inside a user namespace to prevent
159            // a broken cluster from `sudo KUBERNIX_ROOTLESS=1 kubernix`.
160            System::in_user_namespace()
161        } else {
162            false
163        };
164
165        // Prepare the configuration
166        if config.root().exists() {
167            config.try_load_file()?;
168        } else {
169            config.to_file()?;
170        }
171        config.canonicalize_root()?;
172
173        // Set rootless after config loading since try_load_file
174        // replaces the whole struct (resetting #[serde(skip)] fields).
175        config.set_rootless(rootless);
176
177        // Setup the logger
178        set_boxed_logger(Logger::new(config.log_level(), config.log_format()))
179            .context("Unable to set logger")?;
180
181        if config.is_rootless() && !Nix::is_active() {
182            info!("Running in rootless mode");
183        }
184
185        Ok(())
186    }

194    /// Re-exec kubernix inside rootlesskit for rootless operation.
195    /// All child processes inherit the user namespace.
196    fn reexec_rootless() -> Result<()> {
197        info!("Re-executing inside rootlesskit");
198
199        let exe = std::env::current_exe().context("Unable to determine current executable")?;
200        let args: Vec<String> = std::env::args().skip(1).collect();
201
202        let kubernix_args = args
203            .iter()
204            .map(|a| Self::shell_escape(a))
205            .collect::<Vec<_>>()
206            .join(" ");
207        let kubernix_exe = Self::shell_escape(&exe.display().to_string());
208
209        // Inside rootlesskit's cgroup namespace, evacuate processes from the
210        // namespace root into a child and enable controllers, then exec.
211        let rootlesskit_cmd = format!(
212            concat!(
213                "mount -t cgroup2 none /sys/fs/cgroup 2>/dev/null; ",
214                "mkdir -p /sys/fs/cgroup/kubernix; ",
215                "for p in $(cat /sys/fs/cgroup/cgroup.procs); do ",
216                "echo $p >/sys/fs/cgroup/kubernix/cgroup.procs 2>/dev/null; done; ",
217                "for c in memory cpu pids io; do ",
218                "echo \"+$c\" >/sys/fs/cgroup/cgroup.subtree_control 2>/dev/null; done; ",
219                "grep -q memory /sys/fs/cgroup/cgroup.subtree_control 2>/dev/null ",
220                "|| echo 'WARNING: cgroup memory controller not delegated, pods may fail' >&2; ",
221                "exec {} {}",
222            ),
223            kubernix_exe, kubernix_args,
224        );
225
226        // Evacuate all processes from the scope's root cgroup into a child
227        // so we can enable controllers in subtree_control (cgroup v2's
228        // "no internal processes" rule forbids it otherwise). This must
229        // happen outside rootlesskit's cgroup namespace.
230        let outer_cmd = format!(
231            concat!(
232                "CGRP=/sys/fs/cgroup$(cat /proc/self/cgroup | cut -d: -f3); ",
233                "mkdir -p \"$CGRP/init\"; ",
234                "for p in $(cat \"$CGRP/cgroup.procs\"); do ",
235                "echo $p >\"$CGRP/init/cgroup.procs\" 2>/dev/null; done; ",
236                "for c in memory cpu pids io; do ",
237                "echo \"+$c\" >\"$CGRP/cgroup.subtree_control\" 2>/dev/null; done; ",
238                "grep -q memory \"$CGRP/cgroup.subtree_control\" 2>/dev/null ",
239                "|| echo 'WARNING: cgroup memory controller not delegated, pods may fail' >&2; ",
240                "exec rootlesskit --net=host --cgroupns ",
241                "--copy-up=/etc --copy-up=/run --copy-up=/var/cache ",
242                "--copy-up=/var/lib --copy-up=/var/log --copy-up=/var/run ",
243                "bash -c {}",
244            ),
245            Self::shell_escape(&rootlesskit_cmd),
246        );
247
248        let mut cmd = Command::new("systemd-run");
249        cmd.args([
250            "--user",
251            "--scope",
252            "-p",
253            "Delegate=yes",
254            "--",
255            "bash",
256            "-c",
257            &outer_cmd,
258        ]);
259        cmd.env(ROOTLESS_ENV, "1");
260
261        let status = cmd.status().context(
262            "Failed to start rootless session. \
263             Rootless mode requires systemd-run (with an active user session) \
264             and rootlesskit to be installed",
265        )?;
266
267        // No Kubernix struct or other Drop-bearing resources exist yet,
268        // only the logger from prepare_env.
269        std::process::exit(status.code().unwrap_or(1));
270    }

284        if config.is_rootless() { base } else { base + 1 }

287    /// Bootstrap the whole cluster, which assumes to be inside a nix shell
288    fn bootstrap_cluster(config: Config) -> Result<()> {
289        // Setup the progress bar
290        const BASE_STEPS: u64 = 15;
291        let steps = if config.multi_node() && !config.is_rootless() {
292            u64::from(config.nodes()) * 2 + BASE_STEPS
293        } else {
294            BASE_STEPS
295        } + Self::processes(&config);
296        let p = Progress::new(steps, config.log_level());
297        info!("Bootstrapping cluster");
298
299        // Ensure that the system is prepared
300        let system = System::setup(&config).context("Unable to setup system")?;
301        Container::build(&config)?;
302
303        // Setup the network
304        let network = Network::new(&config)?;
305
306        // Setup the public key infrastructure
307        let pki = Pki::new(&config, &network)?;
308
309        // Setup the configs
310        let kubeconfig = KubeConfig::new(&config, &pki)?;
311        let kubectl = Kubectl::new(kubeconfig.admin());
312        let encryptionconfig = EncryptionConfig::new(&config)?;
313
314        let ctx = ClusterContext {
315            config: &config,
316            network: &network,
317            pki: &pki,
318            kubeconfig: &kubeconfig,
319            encryptionconfig: &encryptionconfig,
320            kubectl: &kubectl,
321        };
322
323        let (processes, all_ok) = Self::register_components(&config).run(&ctx);
324
325        let spawn_shell = !config.no_shell();
326        let addon_shutdown = Arc::new(AtomicBool::new(false));
327        let mut kubernix = Kubernix {
328            addon_shutdown: Arc::clone(&addon_shutdown),
329            addon_thread: None,
330            config,
331            network,
332            kubectl,
333            processes,
334            system,
335        };
336
337        if all_ok {
338            if let Err(e) = kubernix.write_env_file() {
339                p.reset();
340                error!("Unable to write environment file: {}", e);
341                return Err(e);
342            }
343
344            kubernix.spawn_addons(addon_shutdown);
345            info!("Everything is up and running");
346            p.reset();
347
348            if spawn_shell {
349                kubernix.spawn_shell()?;
350            } else {
351                kubernix.wait()?;
352            }
353        } else {
354            p.reset();
355            bail!("Unable to start all processes")
356        }
357
358        Ok(())
359    }

361    fn register_components(config: &Config) -> ComponentRegistry {
362        let mut registry = ComponentRegistry::new();
363        registry.register(Box::new(etcd::EtcdComponent));
364        registry.register(Box::new(apiserver::ApiServerComponent));
365        registry.register(Box::new(controllermanager::ControllerManagerComponent));
366        registry.register(Box::new(scheduler::SchedulerComponent));
367        for node in 0..config.nodes() {
368            match config.cri_runtime() {
369                config::CriRuntime::Crio => {
370                    registry.register(Box::new(crio::CrioComponent::new(node)));
371                }
372                config::CriRuntime::Containerd => {
373                    registry.register(Box::new(containerd::ContainerdComponent::new(node)));
374                }
375            }
376            registry.register(Box::new(kubelet::KubeletComponent::new(node)));
377        }
378        if !config.is_rootless() {
379            registry.register(Box::new(proxy::ProxyComponent));
380        }
381        registry
382    }

384    fn spawn_addons(&mut self, addon_shutdown: Arc<AtomicBool>) {
385        let addon_config = self.config.clone();
386        let addon_network = self.network.clone();
387        let addon_kubeconfig = self.kubectl.kubeconfig().to_path_buf();
388        self.addon_thread = Some(thread::spawn(move || {
389            let kubectl = Kubectl::new(&addon_kubeconfig);
390            if addon_shutdown.load(Ordering::Relaxed) {
391                return;
392            }
393            if addon_config.addons().iter().any(|a| a == "coredns")
394                && let Err(e) = CoreDns::apply(&addon_config, &addon_network, &kubectl)
395                && !addon_shutdown.load(Ordering::Relaxed)
396            {
397                error!("Failed to deploy CoreDNS addon: {}", e);
398            }
399        }));
400    }

402    /// Wait until a termination signal occurs
403    fn wait(&self) -> Result<()> {
404        // Setup the signal handlers
405        let term = Arc::new(AtomicBool::new(false));
406        flag::register(SIGTERM, Arc::clone(&term))?;
407        flag::register(SIGINT, Arc::clone(&term))?;
408        flag::register(SIGHUP, Arc::clone(&term))?;
409        info!("Waiting for interrupt…");
410
411        // Write the pid file
412        let pid_file = self.config.root().join("kubernix.pid");
413        debug!("Writing pid file to: {}", pid_file.display());
414        fs::write(pid_file, id().to_string())?;
415
416        // Wait for the signals
417        while !term.load(Ordering::Relaxed) {
418            sleep(Duration::from_millis(100));
419        }
420        Ok(())
421    }

423    /// Spawn a new interactive default system shell
424    fn spawn_shell(&self) -> Result<()> {
425        info!("Spawning interactive shell");
426        info!("Please be aware that the cluster stops if you exit the shell");
427
428        let mut cmd = Command::new(self.config.shell_ok()?);
429        cmd.current_dir(self.config.root());
430        Self::apply_env_file(&Self::env_file(&self.config), &mut cmd)?;
431        cmd.status()?;
432        Ok(())
433    }

435    /// Parse the env file and apply its variables to a Command.
436    /// Handles `export KEY=VALUE` and `KEY=VALUE` formats, stripping
437    /// surrounding single or double quotes from values.
438    fn apply_env_file(env_file: &Path, cmd: &mut Command) -> Result<()> {
439        let content = fs::read_to_string(env_file)
440            .with_context(|| format!("Unable to read env file '{}'", env_file.display()))?;
441        for line in content.lines() {
442            let line = line.strip_prefix("export ").unwrap_or(line);
443            if let Some((key, value)) = line.split_once('=') {
444                let value = value
445                    .strip_prefix('"')
446                    .and_then(|v| v.strip_suffix('"'))
447                    .or_else(|| value.strip_prefix('\'').and_then(|v| v.strip_suffix('\'')))
448                    .unwrap_or(value);
449                cmd.env(key, value);
450            }
451        }
452        Ok(())
453    }

476    /// Remove all stale mounts
477    fn umount(&self) {
478        if self.config.is_rootless() {
479            debug!("Skipping mount cleanup in rootless mode");
480            return;
481        }
482        debug!("Removing active mounts");
483        let now = Instant::now();
484        while now.elapsed().as_secs() < 15 {
485            match Self::read_mount_points(self.config.root()) {
486                Err(e) => {
487                    debug!("Unable to retrieve mounts: {}", e);
488                    sleep(Duration::from_secs(1));
489                }
490                Ok(mount_points) => {
491                    if mount_points.is_empty() {
492                        break;
493                    }
494                    for dest in &mount_points {
495                        debug!("Removing mount: {}", dest.display());
496                        if let Err(e) = umount2(dest, MntFlags::MNT_FORCE) {
497                            debug!("Unable to umount '{}': {}", dest.display(), e);
498                        }
499                    }
500                    sleep(Duration::from_millis(500));
501                }
502            };
503        }
504    }

506    /// Remove CRI storage directories while still inside the user namespace.
507    /// Image layers contain files owned by unmapped uids that become
508    /// unremovable after rootlesskit exits. Unmounts any lingering
509    /// container overlays first so remove_dir_all can succeed.
510    fn cleanup_rootless_storage(&self) {
511        if !self.config.is_rootless() {
512            return;
513        }
514        debug!("Removing rootless CRI storage");
515        let root = self.config.root();
516        for entry in ["crio", "containerd"] {
517            let dir = root.join(entry);
518            if !dir.exists() {
519                continue;
520            }
521            if let Ok(mounts) = Self::read_mount_points(&dir) {
522                for mp in &mounts {
523                    if let Err(e) = umount2(mp.as_path(), MntFlags::MNT_DETACH) {
524                        debug!("Unable to umount '{}': {}", mp.display(), e);
525                    }
526                }
527            }
528            if let Err(e) = fs::remove_dir_all(&dir) {
529                debug!("Unable to remove '{}': {}", dir.display(), e);
530            }
531        }
532    }

534    /// Read mount points from /proc/mounts filtered by the given root path,
535    /// sorted deepest-first for safe unmounting.
536    fn read_mount_points(root: &Path) -> Result<Vec<PathBuf>> {
537        let file = fs::File::open("/proc/mounts").context("Unable to open /proc/mounts")?;
538        let reader = BufReader::new(file);
539        let mut points: Vec<PathBuf> = reader
540            .lines()
541            .map_while(Result::ok)
542            .filter_map(|line| line.split_whitespace().nth(1).map(PathBuf::from))
543            .filter(|p| p.starts_with(root) && p != root)
544            .collect();
545        points.sort_by_key(|p| std::cmp::Reverse(p.components().count()));
546        Ok(points)
547    }

550impl Drop for Kubernix {
551    fn drop(&mut self) {
552        let p = Progress::new(Self::processes(&self.config), self.config.log_level());
553
554        info!("Cleaning up");
555
556        // Signal the addon thread to stop and give it a short window
557        // to finish. If it does not complete in time, proceed with
558        // shutdown rather than blocking for up to 120s.
559        self.addon_shutdown.store(true, Ordering::Relaxed);
560        if let Some(handle) = self.addon_thread.take() {
561            debug!("Waiting for addon deployment to finish");
562            let deadline = Instant::now() + Duration::from_secs(5);
563            while !handle.is_finished() && Instant::now() < deadline {
564                sleep(Duration::from_millis(100));
565            }
566            if handle.is_finished() {
567                let _ = handle.join();
568            } else {
569                debug!("Addon thread did not finish in time, proceeding with shutdown");
570            }
571        }
572
573        self.stop();
574        self.umount();
575        self.cleanup_rootless_storage();
576        self.system.cleanup();
577        info!("Cleanup done");
578
579        p.reset();
580        debug!("All done");
581    }
582}
