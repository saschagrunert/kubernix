1//! Podman container runtime integration.
2//!
3//! Provides podman-specific CLI arguments for image builds and container
4//! execution, including the OCI runtime and storage driver configuration.
5//! Podman finds conmon in its own helper binaries directory.
6
7use crate::{Config, process, system::System};
8use anyhow::Result;
9use log::{LevelFilter, debug};
10use std::{
11    collections::HashMap,
12    env::var_os,
13    ffi::OsStr,
14    fs,
15    path::{Path, PathBuf},
16    process::Stdio,
17    sync::{Mutex, PoisonError},
18};
19
20/// Podman-specific argument builder for container operations.
21pub struct Podman;
22
23impl Podman {
24    /// The executable name
25    pub const EXECUTABLE: &'static str = "podman";
26
27    /// Returns true if podman is configured as container runtime
28    pub fn is_configured(config: &Config) -> bool {
29        Self::is_configured_in(config, var_os("PATH").as_deref())
30    }
31
32    /// Returns true if podman is configured as container runtime, looking
33    /// it up in the provided `PATH`. Besides its name, the executable gets
34    /// checked after resolving symbolic links, and its version output, which
35    /// covers shims like the `docker` command of podman-docker.
36    pub fn is_configured_in(config: &Config, path: Option<&OsStr>) -> bool {
37        let runtime = config.container_runtime();
38        if Self::is_podman_name(Path::new(runtime)) {
39            return true;
40        }
41        let executable = if runtime.contains('/') {
42            PathBuf::from(runtime)
43        } else {
44            match System::find_executable_in(runtime, path) {
45                Ok(executable) => executable,
46                Err(_) => return false,
47            }
48        };
49        let Ok(resolved) = fs::canonicalize(&executable) else {
50            return false;
51        };
52        if Self::is_podman_name(&resolved) {
53            return true;
54        }
55
56        // Running the version command is cheap, but happens for every
57        // container runtime command otherwise
58        static VERSIONS: Mutex<Option<HashMap<PathBuf, bool>>> = Mutex::new(None);
59        let mut versions = VERSIONS.lock().unwrap_or_else(PoisonError::into_inner);
60        *versions
61            .get_or_insert_with(HashMap::new)
62            .entry(resolved)
63            .or_insert_with(|| Self::reports_podman(&executable))
64    }
65
66    /// Whether the file name of the executable is the one of podman.
67    fn is_podman_name(executable: &Path) -> bool {
68        executable.file_name() == Some(OsStr::new(Self::EXECUTABLE))
69    }
70
71    /// Whether the version output of the executable is the one of podman.
72    fn reports_podman(executable: &Path) -> bool {
73        match process::command(executable)
74            .arg("--version")
75            .stderr(Stdio::null())
76            .output()
77        {
78            Ok(output) => output.status.success() && output.stdout.starts_with(b"podman version"),
79            Err(e) => {
80                debug!("Unable to run '{} --version': {}", executable.display(), e);
81                false
82            }
83        }
84    }
85
86    /// Retrieve the podman args for a command which verifies image
87    /// signatures, like `build` or `import`.
88    pub fn signature_args(
89        config: &Config,
90        command: &str,
91        policy_json: &Path,
92    ) -> Result<Vec<String>> {
93        let mut args = Self::default_args(config)?;
94        args.extend(vec![
95            command.into(),
96            format!("--signature-policy={}", policy_json.display()),
97        ]);
98
99        Ok(args)
100    }
101
102    /// Podman args which should apply to every command
103    pub fn default_args(config: &Config) -> Result<Vec<String>> {
104        Self::default_args_in(config, var_os("PATH").as_deref())
105    }
106
107    /// Podman args which should apply to every command, looking up crun
108    /// in the provided `PATH`.
109    pub fn default_args_in(config: &Config, path: Option<&OsStr>) -> Result<Vec<String>> {
110        Ok(Self::args(
111            config.log_level(),
112            &System::find_executable_in("crun", path)?,
113            System::in_container()?,
114        ))
115    }
116
117    /// Podman args for the log level, the crun executable and whether
118    /// kubernix runs inside a container.
119    fn args(log_level: LevelFilter, crun: &Path, in_container: bool) -> Vec<String> {
120        let log_level = if log_level >= LevelFilter::Debug {
121            "debug".into()
122        } else {
123            log_level.to_string()
124        };
125        let mut args = vec![
126            format!("--log-level={}", log_level),
127            format!("--runtime={}", crun.display()),
128            "--cgroup-manager=cgroupfs".into(),
129            "--events-backend=none".into(),
130        ];
131        if in_container {
132            args.push("--storage-driver=vfs".into());
133        }
134        args
135    }
136}

138#[cfg(test)]
139mod tests {
140    use super::*;
141    use crate::config::tests::parse;

144    #[test]
145    fn is_configured_success() -> Result<()> {
146        let c = parse(&["--container-runtime=podman"])?;
147        assert!(Podman::is_configured_in(&c, None));
148        let c = parse(&["--container-runtime=/usr/bin/podman"])?;
149        assert!(Podman::is_configured_in(&c, None));
150        let c = parse(&["--container-runtime=docker"])?;
151        assert!(!Podman::is_configured_in(&c, None));
152        Ok(())
153    }
154
155    #[test]
156    fn is_configured_resolves_executable() -> Result<()> {
157        use std::os::unix::fs::{PermissionsExt, symlink};
158        let dir = tempfile::tempdir()?;
159        let script = |name: &str, version: &str| -> Result<()> {
160            let file = dir.path().join(name);
161            fs::write(&file, format!("#!/bin/sh\necho '{version}'\n"))?;
162            fs::set_permissions(&file, fs::Permissions::from_mode(0o755))?;
163            // Another test thread may fork while the script is open for
164            // writing, which makes running it fail until that child execs.
165            for _ in 0..50 {
166                match std::process::Command::new(&file).output() {
167                    Err(e) if e.raw_os_error() == Some(nix::libc::ETXTBSY) => {
168                        std::thread::sleep(std::time::Duration::from_millis(20))
169                    }
170                    result => {
171                        result?;
172                        break;
173                    }
174                }
175            }
176            Ok(())
177        };
178        script("podman", "not podman")?;
179        script("shim", "podman version 5.0.0")?;
180        script("docker", "Docker version 27.0.0")?;
181        symlink(dir.path().join("podman"), dir.path().join("link"))?;
182        let path = dir.path().as_os_str();
183        let configured = |runtime: &str| -> Result<bool> {
184            let c = parse(&[&format!("--container-runtime={runtime}")])?;
185            Ok(Podman::is_configured_in(&c, Some(path)))
186        };
187
188        // A symbolic link to podman, by name or by path
189        assert!(configured("link")?);
190        assert!(configured(&dir.path().join("link").display().to_string())?);
191
192        // A shim which runs podman, also when cached
193        for _ in 0..2 {
194            assert!(configured("shim")?);
195            assert!(!configured("docker")?);
196        }
197        assert!(!configured("missing")?);
198        Ok(())
199    }

201    #[test]
202    fn args_success() {
203        let crun = PathBuf::from("/nix/bin/crun");
204        assert_eq!(
205            Podman::args(LevelFilter::Info, &crun, false),
206            [
207                "--log-level=INFO",
208                "--runtime=/nix/bin/crun",
209                "--cgroup-manager=cgroupfs",
210                "--events-backend=none",
211            ]
212        );
213        let args = Podman::args(LevelFilter::Trace, &crun, true);
214        assert_eq!(args[0], "--log-level=debug");
215        assert_eq!(
216            args.last().map(String::as_str),
217            Some("--storage-driver=vfs")
218        );
219        assert!(!args.iter().any(|a| a.starts_with("--conmon")));
220    }
