1//! Kubeconfig file generation for cluster components.
2//!
3//! Creates kubeconfig files for each identity (admin, kubelet, proxy,
4//! controller-manager, scheduler) as JSON, which is valid YAML. The files
5//! embed the client keys, so only their owner can read them. When running
6//! via sudo, the admin kubeconfig belongs to the invoking user. The files
7//! get rewritten if the certificates change.
8

27/// The name of the cluster in the kubeconfigs.
28const CLUSTER: &str = "kubernetes";
29/// The name of the context in the kubeconfigs.
30const CONTEXT: &str = "kubernix";

32/// A kubeconfig with a single cluster, user and context, as written by
33/// `kubectl config` with embedded certificates.
34#[derive(Debug, PartialEq, Serialize)]
35#[cfg_attr(test, derive(serde::Deserialize))]
36#[serde(rename_all = "kebab-case")]
37struct Document {
38    #[serde(rename = "apiVersion")]
39    api_version: String,
40    kind: String,
41    clusters: Vec<NamedCluster>,
42    users: Vec<NamedUser>,
43    contexts: Vec<NamedContext>,
44    current_context: String,
45}

91impl Document {
92    /// Build the kubeconfig of the identity, which embeds the base64
93    /// encoded certificates.
94    fn new(identity: &Identity, ca: &Path) -> Result<Document> {
95        let read = |path: &Path| {
96            fs::read(path)
97                .map(|data| STANDARD.encode(data))
98                .with_context(|| format!("Unable to read '{}'", path.display()))
99        };
100        Ok(Document {
101            api_version: "v1".into(),
102            kind: "Config".into(),
103            clusters: vec![NamedCluster {
104                name: CLUSTER.into(),
105                cluster: Cluster {
106                    certificate_authority_data: read(ca)?,
107                    server: format!("https://{}:{API_SERVER_PORT}", Ipv4Addr::LOCALHOST),
108                },
109            }],
110            users: vec![NamedUser {
111                name: identity.user().into(),
112                user: User {
113                    client_certificate_data: read(identity.cert())?,
114                    client_key_data: read(identity.key())?,
115                },
116            }],
117            contexts: vec![NamedContext {
118                name: CONTEXT.into(),
119                context: KubeContext {
120                    cluster: CLUSTER.into(),
121                    user: identity.user().into(),
122                },
123            }],
124            current_context: CONTEXT.into(),
125        })
126    }
127}

129/// Paths of the generated kubeconfig files for all cluster identities.
130#[must_use]
131pub struct KubeConfig {
132    kubelets: Vec<PathBuf>,
133    proxy: PathBuf,
134    controller_manager: PathBuf,
135    scheduler: PathBuf,
136    admin: PathBuf,
137}
138
139impl KubeConfig {
140    /// The directory of the kubeconfigs in the root.
141    pub const DIR: &'static str = "kubeconfig";
142
143    /// Kubeconfig file paths for each kubelet node.
144    pub fn kubelets(&self) -> &[PathBuf] {
145        &self.kubelets
146    }
147
148    /// Kubeconfig file path for kube-proxy.
149    pub fn proxy(&self) -> &Path {
150        &self.proxy
151    }
152
153    /// Kubeconfig file path for the controller manager.
154    pub fn controller_manager(&self) -> &Path {
155        &self.controller_manager
156    }
157
158    /// Kubeconfig file path for the scheduler.
159    pub fn scheduler(&self) -> &Path {
160        &self.scheduler
161    }
162
163    /// Kubeconfig file path for the cluster admin (used by kubectl).
164    pub fn admin(&self) -> &Path {
165        &self.admin
166    }
167
168    /// Write the kubeconfig files for all cluster components, if they do
169    /// not exist or differ, for example because the certificates changed.
170    pub fn new(config: &Config, pki: &Pki) -> Result<KubeConfig> {
171        let dir = config.root().join(Self::DIR);
172        create_dir_all(&dir).context("Unable to create kubeconfig directory")?;
173
174        let ca = pki.ca().cert();
175        let admin_owner = Self::sudo_owner(Uid::effective().is_root() && !config.is_rootless());
176
177        // The kubeconfig files are independent of each other
178        let mut identities = vec![
179            (pki.proxy(), None),
180            (pki.controller_manager(), None),
181            (pki.scheduler(), None),
182            (pki.admin(), admin_owner),
183        ];
184        identities.extend(pki.kubelets().iter().map(|i| (i, None)));
185        let written = identities
186            .into_par_iter()
187            .map(|(identity, owner)| Self::setup_kubeconfig(&dir, identity, ca, owner))
188            .collect::<Result<Vec<bool>>>()?;
189        if written.contains(&true) {
190            info!("Created kubeconfigs");
191        } else {
192            info!("Reusing existing kubeconfigs");
193        }
194
195        Ok(KubeConfig {
196            kubelets: pki
197                .kubelets()
198                .iter()
199                .map(|i| Self::target_config(&dir, i))
200                .collect(),
201            proxy: Self::target_config(&dir, pki.proxy()),
202            controller_manager: Self::target_config(&dir, pki.controller_manager()),
203            scheduler: Self::target_config(&dir, pki.scheduler()),
204            admin: Self::target_config(&dir, pki.admin()),
205        })
206    }
207
208    /// Write the kubeconfig of the identity, owned by `owner` or the
209    /// current user, if it does not exist or differs. Returns true if it
210    /// has been written.
211    fn setup_kubeconfig(
212        dir: &Path,
213        identity: &Identity,
214        ca: &Path,
215        owner: Option<Owner>,
216    ) -> Result<bool> {
217        let kubeconfig = Self::target_config(dir, identity);
218        let document = Document::new(identity, ca)?;
219        let written =
220            write_private_if_changed(&kubeconfig, &serde_json::to_vec_pretty(&document)?, owner)?;
221        if written {
222            debug!("Kubeconfig created for {}", identity.name());
223        }
224        Ok(written)
225    }
226
227    fn target_config(dir: &Path, identity: &Identity) -> PathBuf {
228        dir.join(format!("{}.kubeconfig", identity.name()))
229    }
230
231    /// The user and group which invoked kubernix via sudo, if running as
232    /// real root.
233    fn sudo_owner(root: bool) -> Option<Owner> {
234        if !root {
235            return None;
236        }
237        let uid = env::var("SUDO_UID").ok()?.parse().ok()?;
238        let gid = env::var("SUDO_GID").ok()?.parse().ok()?;
239        Some((Uid::from_raw(uid), Gid::from_raw(gid)))
240    }
241}

277    #[test]
278    fn kubeconfigs_embed_identities() -> Result<()> {
279        let c = test_config()?;
280        let n = test_network()?;
281        let p = Pki::new(&c, &n)?;
282        let kc = KubeConfig::new(&c, &p)?;
283        let ca = fs::read(p.ca().cert())?;
284
285        for (file, identity) in [
286            (kc.admin(), p.admin()),
287            (kc.controller_manager(), p.controller_manager()),
288            (kc.proxy(), p.proxy()),
289            (kc.scheduler(), p.scheduler()),
290            (&kc.kubelets()[0], &p.kubelets()[0]),
291        ] {
292            let document: Document = serde_json::from_slice(&fs::read(file)?)?;
293            assert_eq!(document, Document::new(identity, p.ca().cert())?);
294            assert_eq!(document.api_version, "v1");
295            assert_eq!(document.kind, "Config");
296            assert_eq!(document.current_context, "kubernix");
297
298            let [cluster] = &document.clusters[..] else {
299                panic!("expected a single cluster");
300            };
301            assert_eq!(cluster.name, "kubernetes");
302            assert_eq!(cluster.cluster.server, "https://127.0.0.1:6443");
303            assert_eq!(decode(&cluster.cluster.certificate_authority_data)?, ca);
304
305            let [user] = &document.users[..] else {
306                panic!("expected a single user");
307            };
308            assert_eq!(user.name, identity.user());
309            assert_eq!(
310                decode(&user.user.client_certificate_data)?,
311                fs::read(identity.cert())?
312            );
313            assert_eq!(
314                decode(&user.user.client_key_data)?,
315                fs::read(identity.key())?
316            );
317
318            let [context] = &document.contexts[..] else {
319                panic!("expected a single context");
320            };
321            assert_eq!(context.name, "kubernix");
322            assert_eq!(context.context.cluster, "kubernetes");
323            assert_eq!(context.context.user, identity.user());
324        }
325        Ok(())
326    }

404    #[test]
405    fn new_rewrites_changed() -> Result<()> {
406        let c = test_config()?;
407        let n = test_network()?;
408        let p = Pki::new(&c, &n)?;
409        let kc = KubeConfig::new(&c, &p)?;
410        let admin = fs::read(kc.admin())?;
411        let inode = |path: &Path| -> Result<u64> { Ok(fs::metadata(path)?.ino()) };
412        let proxy = inode(kc.proxy())?;
413
414        // A regenerated certificate and a missing file
415        fs::remove_file(p.admin().key())?;
416        fs::remove_file(kc.scheduler())?;
417        let p = Pki::new(&c, &n)?;
418        let kc = KubeConfig::new(&c, &p)?;
419        assert_ne!(fs::read(kc.admin())?, admin);
420        assert_eq!(
421            serde_json::from_slice::<Document>(&fs::read(kc.admin())?)?,
422            Document::new(p.admin(), p.ca().cert())?
423        );
424        assert!(kc.scheduler().exists());
425        // Unchanged files are kept
426        assert_eq!(inode(kc.proxy())?, proxy);
427        Ok(())
428    }
