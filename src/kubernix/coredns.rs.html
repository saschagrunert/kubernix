3use crate::{
4    API_SERVER_PORT, config::Config, kubectl::Kubectl, network::Network, write_if_changed,

10// CoreDNS runs in the host network, so its DNS server listens on an
11// unprivileged port, which the Service maps to port 53, and its health and
12// readiness ports replace the defaults 8080 and 8181. So they do not
13// collide with other services on the host, like a DNS server listening on
14// port 53 of all addresses.
15const DNS_PORT: u16 = 10053;
16const HEALTH_PORT: u16 = 10054;
17const READY_PORT: u16 = 10055;
18const METRICS_PORT: u16 = 9153;
19
20/// Deploys the CoreDNS addon to the cluster.
21pub struct CoreDns;
22
23impl CoreDns {
24    /// Render the CoreDNS manifest, apply it, and wait for the pod to be ready
25    /// or until `shutdown` is set.
26    pub fn apply(
27        config: &Config,
28        network: &Network,
29        kubectl: &Kubectl,
30        shutdown: &AtomicBool,
31    ) -> Result<()> {
32        debug!("Deploying CoreDNS and waiting to be ready");
33
34        let dir = config.root().join("coredns");
35        create_dir_all(&dir).context("Unable to create CoreDNS directory")?;
36
37        let yml = Self::render(network.dns()?, config.is_rootless());
38        let file = dir.join("coredns.yml");
39        write_if_changed(&file, &yml)?;
40
41        kubectl.apply(&file).context("Unable to deploy CoreDNS")?;
42        kubectl.wait_ready("coredns", shutdown)?;
43        debug!("CoreDNS deployed");
44        Ok(())
45    }
46
47    fn render(dns: Ipv4Addr, skip_resources: bool) -> String {
48        // CoreDNS uses hostNetwork, so it can always reach the API server
49        // at 127.0.0.1. Setting these env vars explicitly avoids depending
50        // on kube-proxy ClusterIP routing being ready when CoreDNS starts.
51        // All servers bind only the pod IP, which is the node IP with
52        // hostNetwork and the endpoint of the Service. It is the gateway
53        // address of the bridge of the node in root mode (see
54        // Network::node_ip), so that the resolver and the metrics do not
55        // listen on the addresses of the host, which other machines use.
56        // The address is not advertised, but a machine in the same layer 2
57        // network which routes the CIDR via the host can still reach it,
58        // since the host accepts packets for all of its addresses on every
59        // interface. In rootless mode, it is the address of the network
60        // namespace of rootlesskit.
61        let env = format!(
62            concat!(
63                "        env:\n",
64                "        - name: KUBERNETES_SERVICE_HOST\n",
65                "          value: \"127.0.0.1\"\n",
66                "        - name: KUBERNETES_SERVICE_PORT\n",
67                "          value: \"{}\"\n",
68                "        - name: POD_IP\n",
69                "          valueFrom:\n",
70                "            fieldRef:\n",
71                "              fieldPath: status.podIP\n",
72            ),
73            API_SERVER_PORT,
74        );
75        let resources = if skip_resources {
76            ""
77        } else {
78            concat!(
79                "        resources:\n",
80                "          limits:\n",
81                "            memory: 170Mi\n",
82                "          requests:\n",
83                "            cpu: 100m\n",
84                "            memory: 70Mi\n",
85            )
86        };
87        format!(
88            include_str!("assets/coredns.yml"),
89            dns = dns,
90            env = env,
91            dns_port = DNS_PORT,
92            health_port = HEALTH_PORT,
93            ready_port = READY_PORT,
94            metrics_port = METRICS_PORT,
95            resources = resources,
96        )
97    }
98}

106    const DNS: Ipv4Addr = Ipv4Addr::new(10, 10, 1, 2);

108    /// Render and parse the manifest into its documents.
109    fn documents(rootless: bool) -> Vec<Value> {
110        Deserializer::from_str(&CoreDns::render(DNS, rootless))
111            .map(|d| Value::deserialize(d).unwrap())
112            .collect()
113    }
114
115    fn kind<'a>(docs: &'a [Value], kind: &str) -> &'a Value {
116        docs.iter()
117            .find(|d| d["kind"].as_str() == Some(kind))
118            .unwrap_or_else(|| panic!("no {} in the manifest", kind))
119    }
120
121    fn container(docs: &[Value]) -> &Value {
122        let containers = &kind(docs, "Deployment")["spec"]["template"]["spec"]["containers"];
123        assert_eq!(containers.as_sequence().unwrap().len(), 1);
124        &containers[0]
125    }
126
127    fn env<'a>(docs: &'a [Value], name: &str) -> &'a Value {
128        container(docs)["env"]
129            .as_sequence()
130            .unwrap()
131            .iter()
132            .find(|e| e["name"].as_str() == Some(name))
133            .unwrap_or_else(|| panic!("no env {}", name))
134    }
135
136    fn corefile(docs: &[Value]) -> &str {
137        kind(docs, "ConfigMap")["data"]["Corefile"]
138            .as_str()
139            .unwrap()
140    }

211    /// The container ports by name.
212    fn ports(docs: &[Value]) -> Vec<(String, u64)> {
213        container(docs)["ports"]
214            .as_sequence()
215            .unwrap()
216            .iter()
217            .map(|p| {
218                (

224    }

226    #[test]
227    fn render_uses_uncommon_ports() {
228        let docs = documents(false);
229        let port = |name: &str| {
230            ports(&docs)
231                .into_iter()
232                .find(|(n, _)| n == name)
233                .map(|(_, p)| p)
234        };
235        assert_eq!(port("dns"), Some(u64::from(DNS_PORT)));
236        assert_eq!(port("dns-tcp"), Some(u64::from(DNS_PORT)));
237        assert_eq!(port("metrics"), Some(u64::from(METRICS_PORT)));
238        assert_eq!(port("liveness-probe"), Some(u64::from(HEALTH_PORT)));
239        assert_eq!(port("readiness-probe"), Some(u64::from(READY_PORT)));
240        assert!(ports(&docs).iter().all(|(_, p)| *p >= 1024));
241
242        let yml = CoreDns::render(DNS, false);
243        assert!(!yml.contains("8080"));
244        assert!(!yml.contains("8181"));
245        // The image grants the binary NET_BIND_SERVICE as file capability,
246        // which has to stay in the bounding set for exec to succeed
247        let capabilities = &container(&docs)["securityContext"]["capabilities"];
248        assert_eq!(capabilities["add"][0].as_str(), Some("NET_BIND_SERVICE"));
249        assert_eq!(capabilities["drop"][0].as_str(), Some("ALL"));
250    }
