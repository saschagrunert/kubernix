1//! Kubernetes API server component.
2//!
3//! Starts `kube-apiserver` with TLS, etcd connectivity, RBAC authorization,
4//! the NodeRestriction admission plugin and encryption-at-rest, then
5//! applies the kubelet API RBAC rules. Anonymous requests are rejected.
6
7use crate::{
8    component::{ClusterContext, Component, Phase},
9    config::Config,
10    encryptionconfig::EncryptionConfig,
11    kubectl::Kubectl,
12    network::Network,
13    pki::Pki,
14    process::{Process, ProcessState, stoppable},
15    write_if_changed,
16};
17use anyhow::{Context, Result};
18use log::debug;
19use std::{fs::create_dir_all, path::Path};

21/// Component wrapper for registry-based startup.
22pub struct ApiServerComponent;
23
24impl Component for ApiServerComponent {
25    fn name(&self) -> &str {
26        "API Server"
27    }
28
29    fn phase(&self) -> Phase {
30        Phase::ControlPlane
31    }
32
33    fn start(&self, ctx: &ClusterContext<'_>) -> ProcessState {
34        ApiServer::start(
35            ctx.config,
36            ctx.network,
37            ctx.pki,
38            ctx.encryptionconfig,
39            ctx.kubectl,
40        )
41    }
42}

44/// Manages the `kube-apiserver` process lifecycle.
45pub struct ApiServer {
46    process: Process,
47}
48
49impl ApiServer {
50    /// Start the API server with the given cluster configuration.
51    pub fn start(
52        config: &Config,
53        network: &Network,
54        pki: &Pki,
55        encryptionconfig: &EncryptionConfig,
56        kubectl: &Kubectl,
57    ) -> ProcessState {
58        let dir = config.root().join("apiserver");
59        create_dir_all(&dir).context("Unable to create API server directory")?;
60
61        let audit_policy = dir.join("audit-policy.yml");
62        write_if_changed(&audit_policy, include_str!("assets/audit-policy.yml"))?;
63
64        // In root mode, the API server advertises the node IP of the first
65        // node instead of the address of the default route of the host, see
66        // Network::advertise_address.
67        let advertise = network
68            .advertise_address()?
69            .map(|ip| format!("--advertise-address={}", ip));
70        let args: &[&str] = &[
71            // Like kubeadm and kind, workloads like node agents need it.
72            "--allow-privileged=true",
73            // Nothing needs unauthenticated access, which would expose
74            // for example the version to the network in root mode.
75            "--anonymous-auth=false",
76            "--audit-log-maxage=30",
77            "--audit-log-maxbackup=3",
78            "--audit-log-maxsize=100",
79            &format!("--audit-log-path={}", dir.join("audit.log").display()),
80            &format!("--audit-policy-file={}", audit_policy.display()),
81            "--authorization-mode=Node,RBAC",
82            "--bind-address=0.0.0.0",
83            &format!("--client-ca-file={}", pki.ca().cert().display()),
84            "--enable-admission-plugins=NodeRestriction",
85            &format!("--etcd-cafile={}", pki.etcd_ca().cert().display()),
86            &format!(
87                "--etcd-certfile={}",
88                pki.apiserver_etcd_client().cert().display()
89            ),
90            &format!(
91                "--etcd-keyfile={}",
92                pki.apiserver_etcd_client().key().display()
93            ),
94            &format!("--etcd-servers=https://{}", network.etcd_client()),
95            "--event-ttl=1h",
96            &format!(
97                "--encryption-provider-config={}",
98                encryptionconfig.path().display()
99            ),
100            &format!(
101                "--kubelet-certificate-authority={}",
102                pki.kubelet_ca().cert().display()
103            ),
104            &format!(
105                "--kubelet-client-certificate={}",
106                pki.apiserver_kubelet_client().cert().display()
107            ),
108            &format!(
109                "--kubelet-client-key={}",
110                pki.apiserver_kubelet_client().key().display()
111            ),
112            // The node names do not resolve if the hosts file cannot be
113            // written (see System::setup), while the node IPs are in the
114            // kubelet certificates and the kubelets listen on all
115            // addresses, with a port per node.
116            "--kubelet-preferred-address-types=InternalIP,Hostname",
117            // Request header authentication for aggregated API servers
118            &format!(
119                "--proxy-client-cert-file={}",
120                pki.front_proxy_client().cert().display()
121            ),
122            &format!(
123                "--proxy-client-key-file={}",
124                pki.front_proxy_client().key().display()
125            ),
126            &format!(
127                "--requestheader-allowed-names={}",
128                pki.front_proxy_client().user()
129            ),
130            &format!(
131                "--requestheader-client-ca-file={}",
132                pki.front_proxy_ca().cert().display()
133            ),
134            "--requestheader-extra-headers-prefix=X-Remote-Extra-",
135            "--requestheader-group-headers=X-Remote-Group",
136            "--requestheader-username-headers=X-Remote-User",
137            "--service-account-issuer=https://kubernetes.default.svc.cluster.local",
138            &format!(
139                "--service-account-key-file={}",
140                pki.service_account().public_key().display()
141            ),
142            &format!(
143                "--service-account-signing-key-file={}",
144                pki.service_account().key().display()
145            ),
146            &format!("--service-cluster-ip-range={}", network.service_cidr()),
147            &format!("--tls-cert-file={}", pki.apiserver().cert().display()),
148            &format!("--tls-private-key-file={}", pki.apiserver().key().display()),
149            "--profiling=false",
150            "--v=2",
151        ];
152        let args = advertise
153            .as_deref()
154            .into_iter()
155            .chain(args.iter().copied())
156            .collect::<Vec<_>>();
157        let mut process = Process::start(&dir, "API Server", "kube-apiserver", &args)?;
158
159        process.wait_ready("Serving securely")?;
160        // The log line only indicates that the server listens, controllers
161        // started right after it may fail on post start hooks which are not
162        // done yet (for example rbac/bootstrap-roles).
163        kubectl.wait_api_ready()?;
164        Self::setup_rbac(&dir, kubectl)?;
165        Ok(Box::new(Self { process }))
166    }
167
168    fn setup_rbac(dir: &Path, kubectl: &Kubectl) -> Result<()> {
169        debug!("Creating API Server RBAC rule for kubelet");
170        let file = dir.join("rbac.yml");
171        write_if_changed(&file, include_str!("assets/apiserver.yml"))?;
172
173        kubectl
174            .apply(&file)
175            .context("Unable to deploy API server RBAC rules")?;
176
177        debug!("API Server RBAC rule created");
178        Ok(())
179    }
180}

184#[cfg(test)]
185mod tests {
186    use serde::Deserialize;
187    use serde_norway::{Deserializer, Value};
188
189    #[test]
190    fn rbac_asset_contains_expected_rules() {
191        let docs: Vec<Value> = Deserializer::from_str(include_str!("assets/apiserver.yml"))
192            .map(|d| Value::deserialize(d).unwrap())
193            .collect();
194        assert_eq!(docs.len(), 2);
195
196        let role = &docs[0];
197        assert_eq!(role["kind"].as_str(), Some("ClusterRole"));
198        let name = role["metadata"]["name"].as_str().unwrap();
199        assert_eq!(name, "system:kube-apiserver-to-kubelet");
200        let rule = &role["rules"][0];
201        let strings = |v: &Value| -> Vec<String> {
202            v.as_sequence()
203                .unwrap()
204                .iter()
205                .map(|s| s.as_str().unwrap().to_owned())
206                .collect()
207        };
208        assert_eq!(strings(&rule["apiGroups"]), [""]);
209        assert_eq!(strings(&rule["verbs"]), ["*"]);
210        let resources = strings(&rule["resources"]);
211        for resource in ["nodes/proxy", "nodes/stats", "nodes/log", "nodes/metrics"] {
212            assert!(resources.iter().any(|r| r == resource), "{}", resource);
213        }
214
215        // The user of the kubelet client certificate of the API server
216        let binding = &docs[1];
217        assert_eq!(binding["kind"].as_str(), Some("ClusterRoleBinding"));
218        assert_eq!(binding["roleRef"]["kind"].as_str(), Some("ClusterRole"));
219        assert_eq!(binding["roleRef"]["name"].as_str(), Some(name));
220        assert_eq!(binding["subjects"][0]["kind"].as_str(), Some("User"));
221        assert_eq!(
222            binding["subjects"][0]["name"].as_str(),
223            Some("kube-apiserver-kubelet-client")
224        );
225    }
226
227    #[test]
228    fn audit_policy_asset_logs_metadata() {
229        let policy = include_str!("assets/audit-policy.yml");
230        assert!(policy.contains("apiVersion: audit.k8s.io/v1"));
231        assert!(policy.contains("kind: Policy"));
232        // The catch-all rule has to be the last one
233        assert!(policy.trim_end().ends_with("- level: Metadata"));
234    }
235}
