1//! Multi-node container management.
2//!
3//! Handles building the base container image and running cluster
4//! components inside containers for multi-node setups. Each container
5//! shares the host network namespace and mounts the runtime root directory.
6
7use crate::{Config, nix::Nix, podman::Podman, process::Process, system::System};
8use anyhow::{Context, Result, bail};
9use log::{LevelFilter, debug, info, trace};
10use std::{
11    fmt::Display,
12    fs,
13    path::{Path, PathBuf},
14    process::{Command, Stdio},
15};
16
17const DEFAULT_IMAGE: &str = "kubernix:base";
18const DEFAULT_ROOT: &str = "kubernix";
19
20/// Provides container image building and process execution for multi-node clusters.
21pub struct Container;
22
23impl Container {
24    /// Build the base image used for the nodes
25    pub fn build(config: &Config) -> Result<()> {
26        // Verify that the provided runtime exists
27        System::find_executable(config.container_runtime())?;
28
29        // Write the policy file
30        let policy_json = Self::policy_json(config);
31        fs::write(&policy_json, include_str!("assets/policy.json"))
32            .context("Unable to write container signature policy")?;
33
34        // Nothing needs to be done on single node or rootless runs
35        if !config.multi_node() || config.is_rootless() {
36            return Ok(());
37        }
38
39        // Skip rebuild if the image already exists
40        if Self::image_exists(config) {
41            info!(
42                "Container image '{}' already exists, skipping build",
43                DEFAULT_IMAGE
44            );
45            return Ok(());
46        }
47
48        // Build the base container image
49        info!("Building base container image '{}'", DEFAULT_IMAGE);
50
51        // Prepare the Dockerfile: use a custom one if provided, otherwise
52        // fall back to the embedded default.
53        let file = config.root().join("Dockerfile");
54        if !file.exists() {
55            if let Some(custom) = config.dockerfile() {
56                debug!("Using custom Dockerfile '{}'", custom.display());
57                fs::copy(custom, &file).context("Unable to copy custom Dockerfile")?;
58            } else {
59                fs::write(
60                    &file,
61                    format!(
62                        include_str!("assets/Dockerfile"),
63                        nix = Nix::DIR,
64                        root = DEFAULT_ROOT
65                    ),
66                )
67                .context("Unable to write Dockerfile")?;
68            }
69        }
70
71        // Exclude .git from the container build context since the
72        // runtime nix directory uses a standalone git repo that the
73        // container image does not need.
74        let dockerignore = config.root().join(".dockerignore");
75        if !dockerignore.exists() {
76            fs::write(&dockerignore, "nix/.git\n").context("Unable to write .dockerignore")?;
77        }
78
79        // Prepare the arguments
80        let mut args = if Podman::is_configured(config) {
81            Podman::build_args(config, &policy_json)?
82        } else {
83            vec!["build".into()]
84        };
85        args.extend([format!("-t={DEFAULT_IMAGE}"), ".".into()]);
86        trace!("Container runtime build args: {:?}", args);
87
88        // Run the build
89        debug!("Running container runtime with args: {}", args.join(" "));
90        let status = Command::new(config.container_runtime())
91            .current_dir(config.root())
92            .args(args)
93            .stderr(Self::stdio(config))
94            .stdout(Self::stdio(config))
95            .status()?;
96        if !status.success() {
97            bail!(
98                "Unable to build container base image '{}' using '{}' (exit: {})",
99                DEFAULT_IMAGE,
100                config.container_runtime(),
101                status,
102            );
103        }
104
105        info!("Container base image built");
106        Ok(())
107    }
108
109    /// Retrieve the default signature policy file location
110    pub fn policy_json(config: &Config) -> PathBuf {
111        config.root().join("policy.json")
112    }
113
114    /// Start a new container based process
115    pub fn start(
116        config: &Config,
117        dir: &Path,
118        identifier: &str,
119        process_name: &str,
120        container_name: &str,
121        args: &[&str],
122    ) -> Result<Process> {
123        // Cleanup possible containers
124        Self::remove(config, container_name);
125
126        // Prepare the arguments
127        let arg_hostname = &format!("--hostname={}", container_name);
128        let arg_name = &format!("--name={}", Self::prefixed_container_name(container_name));
129        let arg_volume_root = &Self::volume_arg(config.root().display());
130        let mut args_vec = vec![
131            "run",
132            "--net=host",
133            "--privileged",
134            "--rm",
135            "--cgroupns=host",
136            arg_hostname,
137            arg_name,
138            arg_volume_root,
139        ];
140
141        // Podman specific arguments
142        let podman_args = Podman::default_args(config)?;
143        if Podman::is_configured(config) {
144            args_vec.extend(podman_args.iter().map(String::as_str));
145        }
146
147        // Mount /dev/mapper if available (requires privileges)
148        let dev_mapper = PathBuf::from("/dev/mapper");
149        let arg_volume_dev_mapper = &Self::volume_arg(dev_mapper.display());
150        if !config.is_rootless() && dev_mapper.exists() {
151            args_vec.push(arg_volume_dev_mapper);
152        }
153
154        // Add the process and the user provided args
155        args_vec.extend(&[DEFAULT_IMAGE, process_name]);
156        args_vec.extend(args);
157
158        // Start the process
159        trace!("Container runtime start args: {:?}", args_vec);
160        Process::start(dir, identifier, config.container_runtime(), &args_vec)
161    }
162
163    fn volume_arg<T: Display>(volume: T) -> String {
164        format!("--volume={v}:{v}", v = volume)
165    }
166
167    /// Exec a command on a container instance
168    pub fn exec(
169        config: &Config,
170        dir: &Path,
171        identifier: &str,
172        process_name: &str,
173        container_name: &str,
174        args: &[&str],
175    ) -> Result<Process> {
176        // Prepare the args
177        let mut args_vec = vec![];
178
179        let podman_args = Podman::default_args(config)?;
180        if Podman::is_configured(config) {
181            args_vec.extend(podman_args.iter().map(String::as_str));
182        }
183
184        let name = Self::prefixed_container_name(container_name);
185        let flake_ref = format!("path:{}", DEFAULT_ROOT);
186        let mut cmd_parts = vec![process_name.to_string()];
187        cmd_parts.extend(args.iter().map(|a| a.to_string()));
188        let run_cmd = cmd_parts.join(" ");
189        args_vec.extend([
190            "exec",
191            &name,
192            "nix",
193            "develop",
194            &flake_ref,
195            "--no-update-lock-file",
196            "--command",
197            "bash",
198            "-c",
199            &run_cmd,
200        ]);
201
202        // Run as usual process
203        trace!("Container runtime exec args: {:?}", args_vec);
204        Process::start(dir, identifier, config.container_runtime(), &args_vec)
205    }

207    /// Remove the provided (maybe running) container.
208    /// Failures are logged but not propagated since the container may not
209    /// exist yet (first run).
210    fn remove(config: &Config, name: &str) {
211        let prefixed = Self::prefixed_container_name(name);
212        match Command::new(config.container_runtime())
213            .arg("rm")
214            .arg("-f")
215            .arg(&prefixed)
216            .stderr(Stdio::null())
217            .stdout(Stdio::null())
218            .status()
219        {
220            Ok(status) if !status.success() => {
221                debug!("Container '{}' removal exited with {}", prefixed, status);
222            }
223            Err(e) => {
224                debug!("Failed to run container rm for '{}': {}", prefixed, e);
225            }
226            _ => {}
227        }
228    }
229
230    /// Retrieve a stdio for the provided config log level
231    fn stdio(config: &Config) -> Stdio {
232        if config.log_level() > LevelFilter::Info {
233            Stdio::inherit()
234        } else {
235            Stdio::null()
236        }
237    }
238
239    /// Check whether the base container image already exists.
240    /// Uses `image inspect` which works for both podman and docker.
241    fn image_exists(config: &Config) -> bool {
242        Command::new(config.container_runtime())
243            .args(["image", "inspect", DEFAULT_IMAGE])
244            .stderr(Stdio::null())
245            .stdout(Stdio::null())
246            .status()
247            .is_ok_and(|s| s.success())
248    }
249
250    /// Retrieve a prefixed container name
251    fn prefixed_container_name(name: &str) -> String {
252        format!("{}-{}", DEFAULT_ROOT, name)
253    }
