2//!
3//! Handles preparing the node image and running cluster components inside
4//! containers for multi-node setups in root mode. Each container shares the
5//! host network namespace, mounts the runtime root directory and the Nix
6//! store of the host read-only, so that the components run from the same
7//! store paths as on the host.

131    /// Prepare the image used for the nodes
132    pub fn build(config: &Config) -> Result<()> {
133        // Verify that the provided runtime exists
134        System::find_executable(config.container_runtime())?;
135
136        // Write the policy file
137        let policy_json = Self::policy_json(config);
138        write_if_changed(&policy_json, include_str!("assets/policy.json"))
139            .context("Unable to write container signature policy")?;
140
141        // Nothing needs to be done on single node or rootless runs
142        if !config.multi_node() || config.is_rootless() {
143            return Ok(());
144        }
145
146        // Older versions wrote a Dockerfile into the run root, which is not
147        // needed any more.
148        let dockerfile = config.root().join("Dockerfile");
149        if let Err(e) = fs::remove_file(&dockerfile)
150            && e.kind() != ErrorKind::NotFound
151        {
152            return Err(e).with_context(|| format!("Unable to remove '{}'", dockerfile.display()));
153        }
154
155        // The stale containers of the run directory are gone already, see
156        // `remove_stale_on_start`.
157        if let Err(e) = Self::remove_legacy(config, var_os("PATH").as_deref()) {
158            warn!(
159                "Unable to remove the node containers of older versions: {:#}",
160                e
161            );
162        }
163
164        // Check the build context before an existing image gets used, so
165        // that a context with the keys of the clusters gets refused even if
166        // its image exists.
167        let exclude_args = match config.dockerfile() {
168            Some(dockerfile) => Self::exclude_root_args(config, &Self::build_context(dockerfile)?)?,
169            None => vec![],
170        };
171
172        // Skip the image creation if it already exists. The tag contains a
173        // hash of the default image content or the custom build inputs, so
174        // changed inputs result in a new image. The nodes use the same tag,
175        // even if the build context changes afterwards.
176        let image = Self::image(config)?;
177        let image = IMAGE.get_or_init(|| image);
178        let runtime_args = Self::runtime_args(config)?;
179        if Self::image_exists(config, &runtime_args, image) {
180            info!("Container image '{}' already exists, skipping build", image);
181            return Ok(());
182        }
183
184        if let Some(dockerfile) = config.dockerfile() {
185            Self::build_image(config, dockerfile, &policy_json, image, exclude_args)?;
186        } else {
187            Self::import_image(config, &policy_json, image)?;
188        }
189        Self::prune_images(config, &runtime_args, image);
190        Ok(())
191    }

193    /// Remove the node containers of a previous run in the run directory
194    /// on startup, which did not get cleaned up, for example because
195    /// kubernix got killed. Their kubelets still listen on their ports, so
196    /// this has to happen before the port check. The lock of the run
197    /// directory ensures that they do not belong to a running instance.
198    pub fn remove_stale_on_start(config: &Config) {
199        if !config.multi_node() || config.is_rootless() {
200            return;
201        }
202        if let Err(e) = Self::remove_stale(config, var_os("PATH").as_deref()) {
203            warn!("Unable to remove stale node containers: {:#}", e);
204        }
205    }

207    /// Remove the node containers of the run root which are left over, for
208    /// example if kubernix got killed. The container runtime and crun get
209    /// looked up in the provided `PATH`, which the runtime uses as well.
210    pub fn remove_stale(config: &Config, path: Option<&OsStr>) -> Result<()> {
211        let command = Self::runtime_command(config, path)?;
212        let filter = format!(
213            "--filter=label={}={}",
214            ROOT_LABEL,
215            System::root_id(config.root())
216        );
217        let output = command(&["ps", "--all", "--quiet", &filter])?;
218        Self::check_output(&output, "Unable to list the node containers")?;
219        let stdout = String::from_utf8_lossy(&output.stdout);
220        let ids: Vec<&str> = stdout.split_whitespace().collect();
221        if ids.is_empty() {
222            return Ok(());
223        }
224
225        info!("Removing {} stale node container(s)", ids.len());
226        let mut args = vec!["rm", "--force"];
227        args.extend(ids);
228        Self::check_output(&command(&args)?, "Unable to remove stale node containers")
229    }

231    /// Remove the node containers of versions before 1.0, which are named
232    /// `kubernix-node-<n>` without the id of the run root and carry no
233    /// labels, so `remove_stale` does not find them. Only call this on
234    /// startup, when the port check ensured that no other cluster runs.
235    fn remove_legacy(config: &Config, path: Option<&OsStr>) -> Result<()> {
236        let names = Self::legacy_containers(config, path, true)?;
237        if names.is_empty() {
238            return Ok(());
239        }
240
241        info!(
242            "Removing {} node container(s) of an older version",
243            names.len()
244        );
245        let mut args = vec!["rm", "--force"];
246        args.extend(names.iter().map(String::as_str));
247        Self::check_output(
248            &Self::runtime_command(config, path)?(&args)?,
249            "Unable to remove the node containers of an older version",
250        )
251    }

253    /// The names of the node containers of versions before 1.0, all of
254    /// them or only the running ones.
255    fn legacy_containers(config: &Config, path: Option<&OsStr>, all: bool) -> Result<Vec<String>> {
256        let command = Self::runtime_command(config, path)?;
257        let format = "--format={{.Names}}";
258        let args: &[&str] = if all {
259            &["ps", "--all", format]
260        } else {
261            &["ps", format]
262        };
263        let output = command(args)?;
264        Self::check_output(&output, "Unable to list the containers")?;
265        Ok(Self::legacy_names(&String::from_utf8_lossy(&output.stdout)))
266    }

268    /// Explain the failed port check of a multi-node cluster in root mode
269    /// if node containers of versions before 1.0 run, whose kubelets may
270    /// hold the ports. They do not get removed automatically, since they
271    /// may belong to a running cluster of an older version.
272    pub fn explain_port_conflict(config: &Config, error: anyhow::Error) -> anyhow::Error {
273        if !config.multi_node() || config.is_rootless() {
274            return error;
275        }
276        match Self::legacy_containers(config, var_os("PATH").as_deref(), false) {
277            Ok(names) if !names.is_empty() => {
278                error.context(Self::legacy_hint(config.container_runtime(), &names))
279            }
280            Ok(_) => error,
281            Err(e) => {
282                debug!(
283                    "Unable to list the node containers of older versions: {:#}",
284                    e
285                );
286                error
287            }
288        }
289    }

303    /// Filter the listed container names for the ones of versions before
304    /// 1.0, which are exactly `kubernix-node-<n>`. Docker separates
305    /// multiple names of a container by commas.
306    fn legacy_names(list: &str) -> Vec<String> {
307        let prefix = format!("{}-node-", NAME_PREFIX);
308        let mut names: Vec<String> = list
309            .split(|c: char| c.is_whitespace() || c == ',')
310            .filter(|name| {
311                name.strip_prefix(&prefix)
312                    .is_some_and(|n| !n.is_empty() && n.bytes().all(|b| b.is_ascii_digit()))
313            })
314            .map(String::from)
315            .collect();
316        names.sort();
317        names.dedup();
318        names
319    }

321    /// A function which runs the container runtime with the provided
322    /// arguments and returns its output. The container runtime and crun
323    /// get looked up in the provided `PATH`, which the runtime uses as
324    /// well.
325    fn runtime_command(
326        config: &Config,
327        path: Option<&OsStr>,
328    ) -> Result<impl Fn(&[&str]) -> Result<Output>> {
329        let runtime = System::find_executable_in(config.container_runtime(), path)?;
330        let runtime_args = Self::runtime_args_in(config, path)?;
331        let path = path.map(OsStr::to_os_string);
332        Ok(move |args: &[&str]| {
333            let mut cmd = process::command(&runtime);
334            if let Some(path) = &path {
335                cmd.env("PATH", path);
336            }
337            cmd.args(&runtime_args)
338                .args(args)
339                .stdin(Stdio::null())
340                .stderr(Stdio::piped())
341                .output()
342                .with_context(|| format!("Unable to run '{}'", runtime.display()))
343        })
344    }

346    /// Fail with the tail of the error output if the command failed.
347    fn check_output(output: &Output, message: &str) -> Result<()> {
348        if !output.status.success() {
349            bail!(
350                "{} (exit: {}){}",
351                message,
352                output.status,
353                Self::stderr_tail(&output.stderr)
354            );
355        }
356        Ok(())
357    }

359    /// The last lines of the error output of a container runtime command,
360    /// prefixed for appending them to an error message.
361    fn stderr_tail(stderr: &[u8]) -> String {
362        let stderr = String::from_utf8_lossy(stderr);
363        let lines: Vec<&str> = stderr
364            .lines()
365            .map(str::trim_end)
366            .filter(|line| !line.is_empty())
367            .collect();
368        let tail = &lines[lines.len().saturating_sub(STDERR_TAIL_LINES)..];
369        if tail.is_empty() {
370            String::new()
371        } else {
372            format!(":\n{}", tail.join("\n"))
373        }
374    }

376    /// Run a container runtime command which writes `stdin` to the
377    /// runtime. Its output gets inherited on debug log levels, otherwise the
378    /// tail of its error output becomes part of the error. It stops on
379    /// termination signals, since a build can take long.
380    fn run_with_output(
381        config: &Config,
382        cmd: &mut Command,
383        stdin: Option<&[u8]>,
384        message: &str,
385    ) -> Result<()> {
386        let verbose = config.log_level() > LevelFilter::Info;
387        cmd.stdout(Self::stdio(config)).stderr(if verbose {
388            Stdio::inherit()
389        } else {
390            Stdio::piped()
391        });
392        let output = process::run_cancelable(cmd, stdin, signals::requested)
393            .with_context(|| format!("Unable to run '{}'", config.container_runtime()))?;
394        Self::check_output(&output, message)
395    }

397    /// Build the node image from the custom Dockerfile, using its directory
398    /// as build context, which `exclude_args` exclude the run roots of.
399    fn build_image(
400        config: &Config,
401        dockerfile: &Path,
402        policy_json: &Path,
403        image: &str,
404        exclude_args: Vec<String>,
405    ) -> Result<()> {
406        info!(
407            "Building container image '{}' from '{}'",
408            image,
409            dockerfile.display()
410        );
411        let mut args = if Podman::is_configured(config) {
412            Podman::signature_args(config, "build", policy_json)?
413        } else {
414            vec!["build".into()]
415        };
416        let context = Self::build_context(dockerfile)?;
417        args.extend(exclude_args);
418        args.extend([
419            format!("--file={}", dockerfile.display()),
420            format!("--tag={image}"),
421            context.display().to_string(),
422        ]);
423
424        debug!("Running container runtime with args: {}", args.join(" "));
425        Self::run_with_output(
426            config,
427            process::command(config.container_runtime()).args(args),
428            None,
429            &format!(
430                "Unable to build container image '{}' using '{}'",
431                image,
432                config.container_runtime()
433            ),
434        )?;
435        info!("Container image built");
436        Ok(())
437    }

464    /// The default node image content as tar archive.
465    fn rootfs() -> Vec<u8> {
466        let mut archive = vec![];
467        for (path, mode) in ROOTFS_DIRS {
468            Self::tar_entry(&mut archive, &format!("{}/", path), mode, b'5', b"");
469        }
470        for (path, content) in ROOTFS_FILES {
471            Self::tar_entry(&mut archive, path, 0o644, b'0', content.as_bytes());
472        }
473        // The end of the archive consists of two zero filled blocks
474        archive.resize(archive.len() + 1024, 0);
475        archive
476    }

478    /// Append a ustar entry owned by root to the archive.
479    fn tar_entry(archive: &mut Vec<u8>, path: &str, mode: u32, kind: u8, content: &[u8]) {
480        let mut header = [0u8; 512];
481        let mut field = |offset: usize, value: &[u8]| {
482            header[offset..offset + value.len()].copy_from_slice(value);
483        };
484        field(0, path.as_bytes());
485        field(100, format!("{:07o}\0", mode).as_bytes());
486        field(108, b"0000000\0");
487        field(116, b"0000000\0");
488        field(124, format!("{:011o}\0", content.len()).as_bytes());
489        field(136, b"00000000000\0");
490        field(148, b"        ");
491        field(156, &[kind]);
492        field(257, b"ustar\0");
493        field(263, b"00");
494        field(265, b"root");
495        field(297, b"root");
496        let checksum: u32 = header.iter().map(|b| u32::from(*b)).sum();
497        header[148..156].copy_from_slice(format!("{:06o}\0 ", checksum).as_bytes());
498        archive.extend_from_slice(&header);
499        archive.extend_from_slice(content);
500        archive.resize(archive.len().next_multiple_of(512), 0);
501    }

508    /// Start a new container based process, with the additional environment
509    /// variables `env`.
510    pub fn start(
511        config: &Config,
512        dir: &Path,
513        identifier: &str,
514        process_name: &str,
515        container_name: &str,
516        args: &[&str],
517        env: &[(&str, &str)],
518    ) -> Result<Process> {
519        let runtime_args = Self::runtime_args(config)?;
520        let image = IMAGE
521            .get()
522            .context("The node container image has not been prepared")?;
523        let name = Self::container_name(config, container_name);
524
525        // Cleanup possible containers
526        Self::remove(config, &runtime_args, &name);
527
528        // Prepare the arguments
529        let arg_hostname = &format!("--hostname={}", container_name);
530        let arg_name = &format!("--name={}", name);
531        let arg_label_root = &format!("--label={}={}", ROOT_LABEL, System::root_id(config.root()));
532        let arg_label_path = &format!("--label={}={}", ROOT_PATH_LABEL, config.root().display());
533        let arg_volume_root = &Self::volume_arg(config.root().display());
534        let arg_volume_store = &format!("{}:ro", Self::volume_arg(NIX_STORE));
535        let arg_entrypoint = &format!(
536            "--entrypoint={}",
537            System::find_executable(process_name)?.display()
538        );
539        let mut arg_env = Self::env_args();
540        arg_env.extend(env.iter().map(|(k, v)| format!("--env={}={}", k, v)));
541        let arg_ca_bundle = Self::ca_bundle_args();
542        let arg_registry = Self::registry_args();
543        let mut args_vec = vec![
544            "run",
545            // Never pull the node image, it only exists locally
546            "--pull=never",
547            "--net=host",
548            "--privileged",
549            "--rm",
550            // An init process reaps the orphans of the node, like the
551            // conmon-rs servers, whose exit CRI-O waits for
552            "--init",
553            "--cgroupns=host",
554            arg_hostname,
555            arg_name,
556            arg_label_root,
557            arg_label_path,
558            arg_volume_root,
559            arg_volume_store,
560            arg_entrypoint,
561        ];
562        args_vec.extend(arg_env.iter().map(String::as_str));
563        args_vec.extend(arg_ca_bundle.iter().map(String::as_str));
564        args_vec.extend(arg_registry.iter().map(String::as_str));
565
566        // The kubelet uses the resolver configuration of systemd-resolved
567        // instead of its stub resolver. The directory gets mounted, since
568        // the file gets replaced on changes.
569        let resolv_conf = Network::resolv_conf();
570        let arg_volume_resolv_conf = resolv_conf
571            .parent()
572            .filter(|_| resolv_conf == Path::new(SYSTEMD_RESOLV_CONF))
573            .map(|dir| format!("{}:ro", Self::volume_arg(dir.display())));
574        args_vec.extend(arg_volume_resolv_conf.as_deref());
575
576        // Podman specific arguments
577        args_vec.extend(runtime_args.iter().map(String::as_str));
578
579        // Mount /dev/mapper if available (requires privileges)
580        let dev_mapper = PathBuf::from("/dev/mapper");
581        let arg_volume_dev_mapper = &Self::volume_arg(dev_mapper.display());
582        if !config.is_rootless() && dev_mapper.exists() {
583            args_vec.push(arg_volume_dev_mapper);
584        }
585
586        // Add the image and the user provided args
587        args_vec.push(image);
588        args_vec.extend(args);
589
590        // Start the process
591        trace!("Container runtime start args: {:?}", args_vec);
592        Process::start(dir, identifier, config.container_runtime(), &args_vec)
593    }

613    /// The environment of the node containers. The processes run from the
614    /// Nix store of the host, so they use the `PATH` of the Nix environment
615    /// as well.

820    /// The build arguments which exclude all run roots from the build
821    /// context, the one of this run and every other directory which
822    /// contains a `kubernix.toml`. The run roots contain the keys of the
823    /// clusters, which must not end up in the image. Podman takes an
824    /// ignore file, which extends the one of the build context. Other
825    /// container runtimes refuse a build context with run roots.
826    fn exclude_root_args(config: &Config, context: &Path) -> Result<Vec<String>> {
827        let root = fs::canonicalize(config.root()).unwrap_or_else(|_| config.root().into());
828        if context
829            .ancestors()
830            .any(|dir| dir == root || dir.join(Config::FILENAME).exists())
831        {
832            bail!(
833                "The build context '{}' of the Dockerfile is part of a run directory, which \
834                 would add the keys of the cluster to the image. Use a Dockerfile outside of it",
835                context.display(),
836            )
837        }
838        let roots = Self::context_roots(context, &root)?;
839        if roots.is_empty() {
840            return Ok(vec![]);
841        }
842
843        if !Podman::is_configured(config) {
844            bail!(
845                "The build context '{}' of the Dockerfile contains the run directories {}, \
846                 which would add the keys of the clusters to the image. Move the Dockerfile \
847                 into its own directory, or use podman as container runtime, which excludes \
848                 them",
849                context.display(),
850                roots
851                    .iter()
852                    .map(|r| format!("'{}'", r.display()))
853                    .collect::<Vec<_>>()
854                    .join(", "),
855            )
856        }
857
858        // The ignore file of the context does not apply with --ignorefile
859        let mut content = [".containerignore", ".dockerignore"]
860            .iter()
861            .find_map(|file| fs::read_to_string(context.join(file)).ok())
862            .unwrap_or_default();
863        if !content.is_empty() && !content.ends_with('\n') {
864            content.push('\n');
865        }
866        // The last matching pattern wins
867        for root in &roots {
868            let Some(pattern) = Self::ignore_pattern(root) else {
869                bail!(
870                    "The build context '{}' of the Dockerfile contains the run directory {:?}, \
871                     which an ignore file cannot exclude, since its path is no valid UTF-8, \
872                     starts or ends with whitespace or contains a line break. Move the \
873                     Dockerfile into its own directory",
874                    context.display(),
875                    root,
876                )
877            };
878            content.push_str(&pattern);
879            content.push('\n');
880        }
881        let file = config.root().join("containerignore");
882        write_file(&file, content.as_bytes())
883            .context("Unable to write the container ignore file")?;
884        Ok(vec![format!("--ignorefile={}", file.display())])
885    }

887    /// The run roots in the build context relative to it: the provided
888    /// one, if it is part of it, and all directories which contain a
889    /// `kubernix.toml`.
890    fn context_roots(context: &Path, root: &Path) -> Result<Vec<PathBuf>> {
891        let mut roots = vec![];
892        Self::find_roots(context, context, root, &mut roots)?;
893        if let Ok(relative) = root.strip_prefix(context)
894            && !roots.iter().any(|r| relative.starts_with(r))
895        {
896            // The run root does not exist yet
897            roots.push(relative.into());
898        }
899        Ok(roots)
900    }

902    /// Collect the run roots below `dir` recursively, see `context_roots`.
903    fn find_roots(context: &Path, dir: &Path, root: &Path, roots: &mut Vec<PathBuf>) -> Result<()> {
904        let mut entries = fs::read_dir(dir)
905            .with_context(|| format!("Unable to read build context '{}'", dir.display()))?
906            .collect::<Result<Vec<_>, _>>()
907            .with_context(|| format!("Unable to read build context '{}'", dir.display()))?;
908        entries.sort_by_key(|entry| entry.file_name());
909        for entry in entries {
910            let path = entry.path();
911            // Symbolic links are not followed, like by the build
912            let is_dir = entry
913                .file_type()
914                .with_context(|| format!("Unable to read '{}'", path.display()))?
915                .is_dir();
916            if path == root || (is_dir && path.join(Config::FILENAME).exists()) {
917                roots.push(path.strip_prefix(context).unwrap_or(&path).into());
918            } else if is_dir {
919                Self::find_roots(context, &path, root, roots)?;
920            }
921        }
922        Ok(())
923    }

925    /// The pattern of an ignore file which matches the relative path
926    /// literally, if there is one. Podman and Docker trim the whitespace
927    /// around patterns and read them line by line, and a leading `#` starts
928    /// a comment unless it is escaped.
929    fn ignore_pattern(path: &Path) -> Option<String> {
930        let path = path.to_str()?;
931        if path.trim() != path || path.contains(['\n', '\r']) {
932            return None;
933        }
934        let mut pattern = String::new();
935        for (i, c) in path.chars().enumerate() {
936            if matches!(c, '*' | '?' | '[' | ']' | '\\' | '!') || (i == 0 && c == '#') {
937                pattern.push('\\');
938            }
939            pattern.push(c);
940        }
941        Some(pattern)
942    }

944    /// Collect the metadata of all entries in the build context
945    /// recursively, except the run root and all other directories which
946    /// contain a `kubernix.toml`, since they are run roots as well. The
947    /// metadata changes together with the content and is cheaper to read.
948    fn context_inputs(
949        context: &Path,
950        dir: &Path,
951        exclude: &Path,
952        inputs: &mut Vec<Vec<u8>>,
953    ) -> Result<()> {
954        let mut entries = fs::read_dir(dir)
955            .with_context(|| format!("Unable to read build context '{}'", dir.display()))?
956            .collect::<Result<Vec<_>, _>>()
957            .with_context(|| format!("Unable to read build context '{}'", dir.display()))?;
958        entries.sort_by_key(|entry| entry.file_name());
959        for entry in entries {
960            let path = entry.path();
961            if path == exclude {
962                continue;
963            }
964            let metadata = entry
965                .metadata()
966                .with_context(|| format!("Unable to read '{}'", path.display()))?;
967            if metadata.is_dir() && path.join(Config::FILENAME).exists() {
968                // Another run root, whose content changes all the time
969                continue;
970            }
971            inputs.push(
972                format!(
973                    "{} {:o} {} {}.{}",
974                    path.strip_prefix(context).unwrap_or(&path).display(),
975                    metadata.mode(),
976                    metadata.len(),
977                    metadata.mtime(),
978                    metadata.mtime_nsec(),
979                )
980                .into_bytes(),
981            );
982            if metadata.is_dir() {
983                Self::context_inputs(context, &path, exclude, inputs)?;
984            }
985        }
986        Ok(())
987    }

1002    /// which are still in use, for example by another run root, fail to be
1003    /// removed.
1004    fn prune_images(config: &Config, runtime_args: &[String], image: &str) {
1005        let output = match process::command(config.container_runtime())
1006            .args(runtime_args)
1007            .args(["images", "--format", "{{.Repository}}:{{.Tag}}"])
1008            .stderr(Stdio::null())
1009            .output()
1010        {
1011            Ok(output) if output.status.success() => output,
1012            Ok(output) => {
1013                debug!("Unable to list container images: {}", output.status);
1014                return;
1015            }
1016            Err(e) => {
1017                debug!("Unable to list container images: {}", e);
1018                return;
1019            }
1020        };
1021        for old in Self::stale_images(&String::from_utf8_lossy(&output.stdout), image) {
1022            debug!("Removing old container image '{}'", old);
1023            Self::run_quietly(config.container_runtime(), runtime_args, &["rmi", &old]);
1024        }

1027    /// Filter the listed `repository:tag` images for `kubernix:base-*`
1028    /// images other than the provided one. Podman prefixes the repository with
1029    /// `localhost/`.
1030    fn stale_images(list: &str, image: &str) -> Vec<String> {
1031        let (repo, prefix) = IMAGE_PREFIX.split_once(':').unwrap_or_default();
1032        let current = image.rsplit_once(':').map(|(_, tag)| tag);
1033        let mut images: Vec<String> = list
1034            .lines()
1035            .map(str::trim)
1036            .filter(|line| {
1037                line.rsplit_once(':').is_some_and(|(r, tag)| {
1038                    (r == repo || r.ends_with(&format!("/{}", repo)))
1039                        && tag.starts_with(prefix)
1040                        && Some(tag) != current
1041                })
1042            })
1043            .map(String::from)
1044            .collect();
1045        images.sort();
1046        images.dedup();
1047        images
1048    }
