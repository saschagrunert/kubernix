1//! CRI-O container runtime component.
2//!
3//! Manages CRI-O instances that provide the container runtime interface
4//! for kubelet. Each node gets its own instance with a separate CNI
5//! network configuration. In multi-node root mode, each instance runs
6//! inside its own node container.

26/// Component wrapper for registry-based startup (per-node).
27pub struct CrioComponent {
28    node: u8,
29    name: String,
30}
31
32impl CrioComponent {
33    /// Create a new CRI-O component for the given node index.
34    pub fn new(node: u8) -> Self {
35        Self {
36            node,
37            name: format!("CRI-O (node {})", node),
38        }
39    }
40}
41
42impl Component for CrioComponent {
43    fn name(&self) -> &str {
44        &self.name
45    }
46
47    fn phase(&self) -> Phase {
48        // CRI-O only needs etcd/apiserver to be up, not the controllers,
49        // so it starts in the Controller phase alongside scheduler/CM.
50        Phase::Controller
51    }
52
53    fn start(&self, ctx: &ClusterContext<'_>) -> ProcessState {
54        Crio::start(ctx.config, self.node, ctx.network)
55    }
56}

58/// Manages a CRI-O process and its associated socket for a single node.
59pub struct Crio {
60    process: Process,
61    socket: CriSocket,
62    container: Option<NodeContainer>,
63    node_name: String,
64    attach_socket_dir: PathBuf,
65}
66
67const CRIO: &str = "crio";
68const CONMONRS: &str = "conmonrs";
69
70/// Registries for short image names, which CRI-O resolves via the
71/// containers-registries.conf(5) file.
72const REGISTRIES_CONF: &str = "unqualified-search-registries = [\"docker.io\", \"quay.io\"]\n";
73
74/// Base directory of the container attach sockets. CRI-O creates them at
75/// `<dir>/<container id>/attach`, which exceeds the unix socket path length
76/// limit for most directories below the run root. In rootless mode, `/run`
77/// belongs to the mount namespace of rootlesskit, in multi-node root mode to
78/// the node container.
79pub const ATTACH_SOCKET_BASE: &str = "/run/kubernix";
80
81impl Crio {
82    /// Start a CRI-O instance for the given node index.
83    pub fn start(config: &Config, node: u8, network: &Network) -> ProcessState {
84        let node_name = Node::name(config, network, node);
85
86        let paths = RuntimePaths::resolve(config)?;
87        let conmonrs = System::find_executable(CONMONRS)?.display().to_string();
88
89        let dir = Self::path(config, network, node);
90        let config_dir = dir.join("crio.conf.d");
91        let config_file = config_dir.join("crio.conf");
92        let registries_file = dir.join("registries.conf");
93        let network_dir = dir.join("cni");
94        let socket = Self::socket(config, network, node)?;
95        let attach_socket_dir = Self::attach_socket_dir(config, node);
96
97        // Always render the configuration, since it references the current
98        // Nix store paths.
99        create_dir_all(&network_dir).context("Unable to create CRI-O CNI directory")?;
100        create_dir_all(&config_dir).context("Unable to create CRI-O config directory")?;
101        create_dir_all(dir.join("ns")).context("Unable to create CRI-O namespace directory")?;
102        if config.is_rootless() {
103            cri::write_rootless_wrappers(&dir, &paths)?;
104        }
105        write_if_changed(
106            &config_file,
107            &Self::render_config(config, &dir, &socket, &attach_socket_dir, &conmonrs, &paths),
108        )
109        .context("Unable to write CRI-O config")?;
110        write_if_changed(&registries_file, REGISTRIES_CONF)
111            .context("Unable to write CRI-O registries config")?;
112        cri::write_cni_config(&network_dir, &node_name, node, network)?;
113
114        if config.is_rootless() {
115            cri::reset_ipam_data(&network_dir)?;
116        }
117        let config_dir_arg = format!("--config-dir={}", config_dir.display());
118        // Short image names get resolved via the registries config, which
119        // does not depend on the one of the host.
120        let registries_arg = format!("--registries-conf={}", registries_file.display());
121        let args: &[&str] = &[&config_dir_arg, &registries_arg];
122
123        let mut process = if config.multi_node() && !config.is_rootless() {
124            // Run inside a container
125            let identifier = format!("CRI-O {}", node_name);
126            Container::start(config, &dir, &identifier, CRIO, &node_name, args)?
127        } else {
128            Process::start(&dir, "CRI-O", CRIO, args)?
129        };
130        let container = Container::node(config, &node_name);
131        process.wait_ready("No systemd watchdog enabled")?;
132
133        Ok(Box::new(Self {
134            process,
135            socket,
136            container,
137            node_name,
138            attach_socket_dir,
139        }))
140    }
141
142    /// Render the CRI-O configuration for the node working directory.
143    ///
144    /// Both OCI runtimes are registered as runtime handlers with their own
145    /// state directory, the configured one is used as default.
146    ///
147    /// The handlers use the pod runtime type, where one conmon-rs instance
148    /// per pod monitors the containers. CRI-O 1.37 creates that instance for
149    /// the infra container of the pod and replaces a dropped infra container
150    /// by a placeholder without runtime handler, which would run all
151    /// containers of the pod with the default handler instead of the one of
152    /// their RuntimeClass, so the infra container is kept. Its OOM score of
153    /// -998 cannot be set in a user namespace, so rootless mode uses the
154    /// runtime wrappers which remove it (see `cri::write_rootless_wrappers`).
155    /// Both workarounds become unnecessary once CRI-O passes the runtime
156    /// handler of the pod to the conmon-rs instance of a dropped infra
157    /// container, and skips or clamps the OOM score of the infra container
158    /// in a user namespace.
159    fn render_config(
160        config: &Config,
161        dir: &Path,
162        socket: &CriSocket,
163        attach_socket_dir: &Path,
164        conmonrs: &str,
165        paths: &RuntimePaths,
166    ) -> String {
167        let runtimes = paths
168            .runtimes
169            .iter()
170            .map(|(runtime, path)| {
171                let path = if config.is_rootless() {
172                    cri::rootless_wrapper(dir, *runtime).display().to_string()
173                } else {
174                    path.clone()
175                };
176                format!(
177                    "[crio.runtime.runtimes.{name}]\n\
178                     runtime_path = \"{path}\"\n\
179                     runtime_root = \"{root}\"\n\
180                     runtime_type = \"pod\"\n\
181                     monitor_path = \"{conmonrs}\"\n\
182                     monitor_cgroup = \"pod\"\n",
183                    name = runtime,
184                    root = dir.join(runtime.name()).display(),
185                )
186            })
187            .collect::<Vec<_>>()
188            .join("\n");
189        let containers_dir = dir.join("containers");
190        format!(
191            include_str!("assets/crio.conf"),
192            attach_socket_dir = attach_socket_dir.display(),
193            containers_root = containers_dir.join("storage").display(),
194            containers_runroot = containers_dir.join("run").display(),
195            listen = socket,
196            log_dir = dir.join("log").display(),
197            namespaces_dir = dir.join("ns").display(),
198            network_dir = dir.join("cni").display(),
199            plugin_dir = paths.plugin_dir,
200            exits_dir = dir.join("exits").display(),
201            default_runtime = config.oci_runtime(),
202            runtimes = runtimes,
203            signature_policy = Container::policy_json(config).display(),
204            storage_driver = "overlay",
205            storage_option = "",
206            version_file = dir.join("version").display(),
207            enable_nri = !config.is_rootless(),
208        )
209    }
210
211    /// Retrieve the CRI socket
212    pub fn socket(config: &Config, network: &Network, node: u8) -> Result<CriSocket> {
213        CriSocket::new(Self::path(config, network, node).join("crio.sock"))
214    }
215
216    /// Retrieve the attach socket directory for the node, which is unique per
217    /// run root and node. CRI-O creates it on startup.
218    fn attach_socket_dir(config: &Config, node: u8) -> PathBuf {
219        PathBuf::from(ATTACH_SOCKET_BASE)
220            .join(System::root_id(config.root()))
221            .join(node.to_string())
222    }
223
224    /// Remove the attach socket directory of the node and its parents, as
225    /// long as they are empty.
226    fn remove_attach_socket_dir(&self) {
227        if let Err(e) = fs::remove_dir_all(&self.attach_socket_dir)
228            && self.attach_socket_dir.exists()
229        {
230            debug!(
231                "Unable to remove '{}': {}",
232                self.attach_socket_dir.display(),
233                e
234            );
235        }
236        for dir in self.attach_socket_dir.ancestors().skip(1).take(2) {
237            if fs::remove_dir(dir).is_err() {
238                break;
239            }
240        }
241    }
242
243    /// Retrieve the working path for the node
244    fn path(config: &Config, network: &Network, node: u8) -> PathBuf {
245        config
246            .root()
247            .join(CRIO)
248            .join(Node::name(config, network, node))
249    }
250}

252impl Stoppable for Crio {
253    fn stop(&mut self) -> Result<()> {
254        // Always stop the process, even if the workloads cannot be removed,
255        // for example because CRI-O crashed.
256        let removed = cri::remove_all_containers("CRI-O", &self.socket, &self.node_name)
257            .with_context(|| format!("Unable to remove CRI-O containers on {}", self.node_name));
258        let stopped = self.process.stop();
259        if let Some(container) = &mut self.container {
260            container.remove();
261        }
262        // A CRI-O which is still running may use the sockets
263        if stopped.is_ok() {
264            self.remove_attach_socket_dir();
265        }
266        removed.and(stopped)
267    }
268
269    fn died(&mut self) -> bool {
270        self.process.died()
271    }
272}
