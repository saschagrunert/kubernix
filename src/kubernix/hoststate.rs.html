1//! Host state of the cluster in root mode.
2//!
3//! The sysctls of the host, the rules of kube-proxy and the bridges are
4//! global. Only one cluster runs in root mode at a time: the components
5//! listen on fixed ports (see `Network::check_ports`), and the state file
6//! stays locked from the setup of the system until the host state got
7//! restored, which also covers the startup of a cluster before its
8//! components listen and its cleanup. The original values get saved before
9//! the setup and restored when the cluster stops.
10
11use crate::files::{PRIVATE, open_lock_file, open_nofollow, try_lock, write_atomic};
12use anyhow::{Context, Result};
13use log::{debug, warn};
14use nix::libc;
15use std::{
16    collections::BTreeMap,
17    fs::{self, File},
18    io::{self, Read},
19    path::{Path, PathBuf},
20};

22/// The state file, locked via `<state file>.lock`. `/run` gets cleared on
23/// reboot, like the sysctls.
24pub const HOST_STATE_FILE: &str = "/run/kubernix-host-state";
25
26/// The location of the sysctls.
27pub const PROC_SYS: &str = "/proc/sys";
28
29/// The path of the sysctl `key` below `proc_sys`, usually `PROC_SYS`.
30pub fn sysctl_path(proc_sys: &Path, key: &str) -> PathBuf {
31    proc_sys.join(key.replace('.', "/"))
32}
33
34/// Read the value of the sysctl `key` below `proc_sys`.
35pub fn read_sysctl(proc_sys: &Path, key: &str) -> io::Result<String> {
36    Ok(fs::read_to_string(sysctl_path(proc_sys, key))?
37        .trim()
38        .to_string())
39}
40
41/// Set the sysctl `key` below `proc_sys` to `value`.
42pub fn write_sysctl(proc_sys: &Path, key: &str, value: &str) -> io::Result<()> {
43    fs::write(sysctl_path(proc_sys, key), value)
44}
45
46/// The sysctls which kubernix, the kubelet and kube-proxy change in root
47/// mode and which get restored. Sysctls which do not exist before the
48/// setup, like the ones of modules which get loaded, are not restored.
49///
50/// `net.ipv4.ip_forward` and the `net.bridge.bridge-nf-call-*` sysctls are
51/// not restored, like kubeadm and kind do not: Docker, libvirt or a VPN
52/// started while the cluster runs may need them, and disabling the
53/// forwarding resets it for every interface. `br_netfilter` stays loaded
54/// as well.
55const SYSCTLS: &[&str] = &[
56    "net.ipv4.conf.all.route_localnet",
57    "net.netfilter.nf_conntrack_max",
58    "net.netfilter.nf_conntrack_tcp_timeout_close_wait",
59    "net.netfilter.nf_conntrack_tcp_timeout_established",
60    "kernel.keys.root_maxbytes",
61    "kernel.keys.root_maxkeys",
62    "kernel.panic",
63    "kernel.panic_on_oops",
64    "vm.overcommit_memory",
65    "vm.panic_on_oom",
66];
67
68/// The host state, locked exclusively while the cluster starts, runs and
69/// stops.
70///
71/// The state file contains the original values of the sysctls as
72/// `key=value` lines, other lines get ignored. It gets replaced atomically,
73/// so a crash while writing it never loses the saved values. The lock is
74/// an open file description lock of a separate lock file next to it
75/// (`<state file>.lock`), which stays the same file while the state file
76/// gets replaced, and gets released when the process exits, also after a
77/// crash or force quit. The original values stay in the state file then,
78/// so the next cluster keeps them and restores them on stop.
79pub struct HostState {
80    path: PathBuf,
81    lock: File,
82    proc_sys: PathBuf,
83    released: bool,
84}
85
86impl HostState {
87    /// Lock the state file `path` exclusively without waiting. Returns
88    /// `None` if another cluster holds the lock.
89    pub fn acquire(path: &Path, proc_sys: &Path) -> Result<Option<Self>> {
90        let lock_path = Self::lock_path(path)?;
91        let lock = open_lock_file(&lock_path)?;
92        if !try_lock(&lock, libc::F_WRLCK)
93            .with_context(|| format!("Unable to lock '{}'", lock_path.display()))?
94        {
95            return Ok(None);
96        }
97        Ok(Some(Self {
98            path: path.into(),
99            lock,
100            proc_sys: proc_sys.into(),
101            released: false,
102        }))
103    }
104
105    /// The lock file of the state file `path`.
106    fn lock_path(path: &Path) -> Result<PathBuf> {
107        let name = path
108            .file_name()
109            .with_context(|| format!("Invalid host state file '{}'", path.display()))?;
110        let mut lock = name.to_os_string();
111        lock.push(".lock");
112        Ok(path.with_file_name(lock))
113    }
114
115    /// Read the saved sysctls. A missing file contains none.
116    fn read(&self) -> Result<BTreeMap<String, String>> {
117        let mut content = String::new();
118        match open_nofollow(&self.path) {
119            Ok(mut file) => {
120                file.read_to_string(&mut content)
121                    .with_context(|| format!("Unable to read '{}'", self.path.display()))?;
122            }
123            Err(e) if e.kind() == io::ErrorKind::NotFound => {}
124            Err(e) => {
125                return Err(e).with_context(|| format!("Unable to open '{}'", self.path.display()));
126            }
127        }
128        Ok(content
129            .lines()
130            .filter_map(|line| line.split_once('='))
131            .map(|(k, v)| (k.into(), v.into()))
132            .collect())
133    }
134
135    /// Save the current values of the sysctls which are not saved yet.
136    /// Saved values are original ones, which a former run saved before
137    /// changing them and did not restore, for example after a crash.
138    pub fn save(&self) -> Result<()> {
139        let mut saved = self.read().context("Unable to read the host state")?;
140        let mut changed = false;
141        for key in SYSCTLS {
142            if saved.contains_key(*key) {
143                continue;
144            }
145            if let Ok(value) = read_sysctl(&self.proc_sys, key) {
146                saved.insert(key.to_string(), value);
147                changed = true;
148            }
149        }
150        if changed {
151            let content = saved
152                .iter()
153                .map(|(k, v)| format!("{}={}\n", k, v))
154                .collect::<String>();
155            self.write(&content)
156                .context("Unable to save the host state")?;
157        }
158        Ok(())
159    }
160
161    /// Replace the state file atomically, which only root may access.
162    fn write(&self, content: &str) -> Result<()> {
163        write_atomic(&self.path, content.as_bytes(), PRIVATE, None)
164    }
165
166    /// Restore the saved sysctls, which did not change on failure. Saved
167    /// values of sysctls which are not restored, like the ones which former
168    /// builds saved before a crash, are skipped. The values which could not
169    /// be restored stay in the state file, so that a later run retries,
170    /// unless the sysctl does not exist anymore, for example since its
171    /// module got unloaded.
172    fn restore(&self) -> Result<()> {
173        let mut failed = String::new();
174        for (key, value) in self.read()? {
175            if !SYSCTLS.contains(&key.as_str()) {
176                debug!("Not restoring sysctl '{}'", key);
177                continue;
178            }
179            if read_sysctl(&self.proc_sys, &key).is_ok_and(|x| x == value) {
180                continue;
181            }
182            if !sysctl_path(&self.proc_sys, &key).exists() {
183                debug!("Not restoring sysctl '{}', it does not exist", key);
184                continue;
185            }
186            debug!("Restoring sysctl '{}' to '{}'", key, value);
187            if let Err(e) = write_sysctl(&self.proc_sys, &key, &value) {
188                warn!(
189                    "Unable to restore sysctl '{}' to '{}', keeping it for the next run: {}",
190                    key, value, e
191                );
192                failed.push_str(&format!("{}={}\n", key, value));
193            }
194        }
195        self.write(&failed)
196    }
197
198    /// Restore the saved sysctls, run `cleanup` to restore the rest of the
199    /// host state and unlock the state file, once.
200    pub fn release(&mut self, cleanup: impl FnOnce()) {
201        if self.released {
202            return;
203        }
204        self.released = true;
205        debug!("Restoring the host state");
206        if let Err(e) = self.restore() {
207            warn!("Unable to restore the host state: {:#}", e);
208        }
209        cleanup();
210        // Closing the file releases the lock as well, but child processes
211        // may share the open file description between fork and exec
212        if let Err(e) = try_lock(&self.lock, libc::F_UNLCK) {
213            debug!("Unable to unlock the host state: {}", e);
214        }
215    }
216}
217
218impl Drop for HostState {
219    /// Restore the state on errors before the cluster started.
220    fn drop(&mut self) {
221        self.release(|| {});
222    }
223}

231    /// A state file and a fake `/proc/sys` with some of the sysctls.
232    fn setup() -> Result<(TempDir, PathBuf, PathBuf)> {
233        let dir = tempdir()?;
234        let file = dir.path().join("state");
235        let proc_sys = dir.path().join("sys");
236        for (key, value) in [
237            ("net.ipv4.ip_forward", "0"),
238            ("net.ipv4.conf.all.route_localnet", "0"),
239            ("kernel.panic", "0"),
240        ] {
241            let path = proc_sys.join(key.replace('.', "/"));
242            fs::create_dir_all(path.parent().context("no parent")?)?;
243            fs::write(path, format!("{}\n", value))?;
244        }
245        Ok((dir, file, proc_sys))
246    }

248    fn get(proc_sys: &Path, key: &str) -> Result<String> {
249        Ok(read_sysctl(proc_sys, key)?)
250    }
251
252    fn set(proc_sys: &Path, key: &str, value: &str) -> Result<()> {
253        let path = sysctl_path(proc_sys, key);
254        fs::create_dir_all(path.parent().context("no parent")?)?;
255        Ok(write_sysctl(proc_sys, key, value)?)
256    }

258    #[test]
259    fn sysctl_read_write() -> Result<()> {
260        let (_dir, _, proc_sys) = setup()?;
261        assert_eq!(
262            sysctl_path(&proc_sys, "net.ipv4.ip_forward"),
263            proc_sys.join("net/ipv4/ip_forward")
264        );
265        assert_eq!(read_sysctl(&proc_sys, "net.ipv4.ip_forward")?, "0");
266        write_sysctl(&proc_sys, "net.ipv4.ip_forward", "1")?;
267        assert_eq!(read_sysctl(&proc_sys, "net.ipv4.ip_forward")?, "1");
268        assert!(read_sysctl(&proc_sys, "net.ipv4.missing").is_err());
269        assert!(write_sysctl(&proc_sys, "net.missing.key", "1").is_err());
270        Ok(())
271    }

273    /// Change the sysctls like a cluster does, loading `br_netfilter`
274    /// adds the bridge sysctls.
275    fn change(proc_sys: &Path) -> Result<()> {
276        set(proc_sys, "net.ipv4.ip_forward", "1")?;
277        set(proc_sys, "net.ipv4.conf.all.route_localnet", "1")?;
278        set(proc_sys, "kernel.panic", "10")?;
279        set(proc_sys, "net.bridge.bridge-nf-call-iptables", "1")
280    }

282    fn start(file: &Path, proc_sys: &Path) -> Result<HostState> {
283        let Some(state) = HostState::acquire(file, proc_sys)? else {
284            bail!("host state locked");
285        };
286        state.save()?;
287        change(proc_sys)?;
288        state.save()?;
289        Ok(state)
290    }

292    #[test]
293    fn restore_on_release() -> Result<()> {
294        let (_dir, file, proc_sys) = setup()?;
295        let mut state = start(&file, &proc_sys)?;
296        let saved = fs::read_to_string(&file)?;
297        assert_eq!(
298            saved,
299            "kernel.panic=0\nnet.ipv4.conf.all.route_localnet=0\n"
300        );
301
302        let mut called = false;
303        state.release(|| called = true);
304        assert!(called);
305        assert_eq!(get(&proc_sys, "net.ipv4.conf.all.route_localnet")?, "0");
306        assert_eq!(get(&proc_sys, "kernel.panic")?, "0");
307        // Other services may need the forwarding and the bridge filtering
308        assert_eq!(get(&proc_sys, "net.ipv4.ip_forward")?, "1");
309        assert_eq!(get(&proc_sys, "net.bridge.bridge-nf-call-iptables")?, "1");
310        assert_eq!(fs::read_to_string(&file)?, "");
311        // Only once
312        called = false;
313        state.release(|| called = true);
314        assert!(!called);
315        Ok(())
316    }

318    #[test]
319    fn exclusive_until_released() -> Result<()> {
320        let (_dir, file, proc_sys) = setup()?;
321        let mut first = start(&file, &proc_sys)?;
322        // Another cluster can neither start nor change the saved values
323        assert!(HostState::acquire(&file, &proc_sys)?.is_none());
324        let mut second = None;
325        first.release(|| {
326            second = HostState::acquire(&file, &proc_sys).ok().flatten();
327        });
328        assert!(second.is_none(), "locked while restoring");
329        // Unlocked before the file gets closed
330        let mut second = start(&file, &proc_sys)?;
331        assert!(HostState::acquire(&file, &proc_sys)?.is_none());
332        drop(first);
333        second.release(|| {});
334        assert_eq!(get(&proc_sys, "net.ipv4.conf.all.route_localnet")?, "0");
335        Ok(())
336    }

338    #[test]
339    fn keep_original_values_after_crash() -> Result<()> {
340        let (_dir, file, proc_sys) = setup()?;
341        // A crash releases the lock without restoring the state
342        let mut crashed = start(&file, &proc_sys)?;
343        crashed.released = true;
344        // Closing the file releases the lock as well, but child processes
345        // which other tests fork concurrently may hold a copy of it
346        try_lock(&crashed.lock, libc::F_UNLCK)?;
347        drop(crashed);
348
349        let mut state = start(&file, &proc_sys)?;
350        state.release(|| {});
351        assert_eq!(get(&proc_sys, "net.ipv4.conf.all.route_localnet")?, "0");
352        assert_eq!(get(&proc_sys, "kernel.panic")?, "0");
353        Ok(())
354    }

356    #[test]
357    fn state_file_replaced_while_locked() -> Result<()> {
358        use std::os::unix::fs::MetadataExt;
359        let (dir, file, proc_sys) = setup()?;
360        assert_eq!(HostState::lock_path(&file)?, dir.path().join("state.lock"));
361        // The format of former runs gets read as well
362        fs::write(&file, "kernel.panic=5\n")?;
363        let inode = fs::metadata(&file)?.ino();
364        let mut state = start(&file, &proc_sys)?;
365        let metadata = fs::metadata(&file)?;
366        assert_ne!(metadata.ino(), inode, "replaced instead of truncated");
367        assert_eq!(metadata.mode() & 0o777, 0o600);
368        assert_eq!(
369            fs::read_to_string(&file)?,
370            "kernel.panic=5\nnet.ipv4.conf.all.route_localnet=0\n"
371        );
372        // The lock stays held although the state file got replaced
373        assert!(HostState::acquire(&file, &proc_sys)?.is_none());
374        state.release(|| {});
375        assert_eq!(get(&proc_sys, "kernel.panic")?, "5");
376
377        // Without a state file, nothing gets restored
378        fs::remove_file(&file)?;
379        let mut state = HostState::acquire(&file, &proc_sys)?.context("locked")?;
380        assert!(state.read()?.is_empty());
381        state.release(|| {});
382        let mut entries = fs::read_dir(dir.path())?
383            .map(|e| Ok(e?.file_name().to_string_lossy().into_owned()))
384            .collect::<Result<Vec<_>>>()?;
385        entries.sort();
386        assert_eq!(entries, ["state", "state.lock", "sys"]);
387        Ok(())
388    }

399    #[test]
400    fn keep_values_which_failed_to_restore() -> Result<()> {
401        let (_dir, file, proc_sys) = setup()?;
402        let mut state = start(&file, &proc_sys)?;
403        // Writing a directory fails
404        let panic = sysctl_path(&proc_sys, "kernel.panic");
405        fs::remove_file(&panic)?;
406        fs::create_dir(&panic)?;
407        state.release(|| {});
408        assert_eq!(get(&proc_sys, "net.ipv4.conf.all.route_localnet")?, "0");
409        assert_eq!(fs::read_to_string(&file)?, "kernel.panic=0\n");
410
411        // The next run keeps the original value and restores it
412        fs::remove_dir(&panic)?;
413        set(&proc_sys, "kernel.panic", "10")?;
414        let mut state = start(&file, &proc_sys)?;
415        state.release(|| {});
416        assert_eq!(get(&proc_sys, "kernel.panic")?, "0");
417        assert_eq!(fs::read_to_string(&file)?, "");
418
419        // Sysctls which do not exist anymore are dropped
420        let mut state = start(&file, &proc_sys)?;
421        fs::remove_file(&panic)?;
422        state.release(|| {});
423        assert_eq!(fs::read_to_string(&file)?, "");
424        Ok(())
425    }
