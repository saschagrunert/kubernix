1//! Nix environment bootstrapping.
2//!
3//! Manages the `nix develop` shell that provides all runtime
4//! dependencies (etcd, kubernetes, CRI runtimes, etc.) at pinned
5//! versions. Re-executes the kubernix binary inside the Nix shell,
6//! forwarding all CLI options. The evaluated environment gets cached in a
7//! Nix profile, which avoids evaluating the flake on every start and keeps
8//! its store paths from being garbage collected.
9
10use crate::{Config, config::Addon, content_hash, system::System};
11use ::nix::fcntl::{Flock, FlockArg};
12use anyhow::{Context, Result, bail};
13use log::{debug, info};
14use std::{
15    env::{current_exe, var},
16    fs::{self, File, OpenOptions, create_dir_all},
17    os::unix::process::CommandExt,
18    path::{Path, PathBuf},
19    process::{Command, Stdio},
20};

22/// Handles Nix shell bootstrapping and command execution.
23pub struct Nix;

25impl Nix {
26    /// Directory of the generated Nix environment inside the run root.
27    pub const DIR: &'static str = "nix";
28
29    /// Files of the generated Nix environment.
30    pub const FILES: [&'static str; 4] = ["flake.nix", "flake.lock", "packages.nix", "overlay.nix"];
31    const NIX_ENV: &'static str = "IN_NIX";
32
33    /// The directory of the cached environment. It cannot be part of the
34    /// Nix directory, since the flake consists of all files in there.
35    const PROFILE_DIR: &'static str = "nix-profile";
36
37    /// The lock file which serializes the creation of the cached
38    /// environment.
39    const PROFILE_LOCK: &'static str = "nix-profile.lock";
40
41    /// The `.dockerignore` content of older versions.
42    const DOCKERIGNORE: &'static str = "nix/.git\n";

44    /// Bootstrap the nix environment
45    pub fn bootstrap(config: Config) -> Result<()> {
46        // Prepare the nix dir
47        debug!("Nix environment not found, bootstrapping one");
48        let dir = config.root().join(Self::DIR);
49
50        // Write the environment, regenerating it if the rendered files
51        // differ, for example after upgrading kubernix or changing packages.
52        // The `path:` flake reference makes Nix use all files of the
53        // directory, even if it is part of a git worktree.
54        if Self::write_files(&dir, &Self::render_files(&config)?)? {
55            debug!("Nix environment written to '{}'", dir.display());
56        }
57        Self::remove_leftovers(config.root())?;
58
59        // Run the shell, forwarding all config options
60        let exe = current_exe()?.display().to_string();
61        let root = config.root().display().to_string();
62        let log_level = config.log_level().to_string();
63        let log_format = config.log_format().to_string();
64        let cidr = config.cidr().to_string();
65        let nodes = config.nodes().to_string();
66        let container_runtime = config.container_runtime();
67        let cri_runtime = config.cri_runtime().to_string();
68        let oci_runtime = config.oci_runtime().to_string();
69
70        let shell_val: String = config.shell().unwrap_or_default().to_owned();
71        let overlay_val = config.overlay().map(|o| o.display().to_string());
72        let mut args = vec![
73            exe.as_str(),
74            "--root",
75            root.as_str(),
76            "--log-level",
77            log_level.as_str(),
78            "--log-format",
79            log_format.as_str(),
80            "--cidr",
81            cidr.as_str(),
82            "--nodes",
83            nodes.as_str(),
84            "--container-runtime",
85            container_runtime,
86            "--cri-runtime",
87            cri_runtime.as_str(),
88            "--oci-runtime",
89            oci_runtime.as_str(),
90        ];
91
92        if let Some(ref overlay) = overlay_val {
93            args.push("--overlay");
94            args.push(overlay.as_str());
95        }
96
97        let dockerfile_val = config.dockerfile().map(|d| d.display().to_string());
98        if let Some(ref dockerfile) = dockerfile_val {
99            args.push("--dockerfile");
100            args.push(dockerfile.as_str());
101        }
102
103        for pkg in config.packages() {
104            args.push("--packages");
105            args.push(pkg.as_str());
106        }
107
108        let addons = Self::addons_arg(config.addons());
109        args.push("--addons");
110        args.push(addons.as_str());
111
112        if config.no_shell() {
113            args.push("--no-shell");
114        } else if !shell_val.is_empty() {
115            args.push("--shell");
116            args.push(&shell_val);
117        }
118
119        Self::exec(&config, &args)
120    }
121
122    /// The `--addons` value for the addons, `none` explicitly disables the
123    /// default addons.
124    fn addons_arg(addons: &[Addon]) -> String {
125        if addons.is_empty() {
126            return Addon::None.to_string();
127        }
128        addons
129            .iter()
130            .map(ToString::to_string)
131            .collect::<Vec<_>>()
132            .join(",")
133    }

135    /// Replace the current process by a command inside the nix develop
136    /// shell, which only returns on failure. Signals reach the command
137    /// directly, since `nix develop` execs it as well.
138    ///
139    /// Each element in `args` is passed as a separate OS argument to
140    /// `nix develop --command`, avoiding shell interpretation.
141    pub fn exec(config: &Config, args: &[&str]) -> Result<()> {
142        let nix = System::find_executable("nix")?;
143        let profile = Self::profile(config, &nix)?;
144
145        let mut cmd = Command::new(&nix);
146        cmd.env(Nix::NIX_ENV, "true")
147            .arg("develop")
148            .arg(&profile)
149            .arg("--log-format")
150            .arg("raw")
151            .arg("--command");
152        for arg in args {
153            cmd.arg(arg);
154        }
155        Err(cmd.exec()).context("Unable to run nix develop")
156    }

158    /// The cached environment of the Nix directory, which gets created if it
159    /// does not exist or is unusable. Its path contains a hash of the
160    /// environment files, so a changed environment results in a new one.
161    /// The profile is a garbage collector root for the store paths of the
162    /// environment.
163    fn profile(config: &Config, nix: &Path) -> Result<PathBuf> {
164        let profile = Self::profile_path(config.root())?;
165        if Self::is_usable(&profile) {
166            debug!("Using cached Nix environment '{}'", profile.display());
167            return Ok(profile);
168        }
169
170        // A new shell may race with the start of the cluster. The lock of
171        // the run directory cannot be used for that, since the running
172        // cluster holds it.
173        let _lock = Self::lock_profile(config.root())?;
174        if Self::is_usable(&profile) {
175            return Ok(profile);
176        }
177        if profile.symlink_metadata().is_ok() {
178            info!("Cached Nix environment is unusable, recreating it");
179        }
180
181        // Remove the profiles of previous environments, which allows the
182        // garbage collector to remove their store paths.
183        let dir = config.root().join(Self::PROFILE_DIR);
184        if dir.exists() {
185            fs::remove_dir_all(&dir)
186                .with_context(|| format!("Unable to remove '{}'", dir.display()))?;
187        }
188        create_dir_all(&dir).with_context(|| format!("Unable to create '{}'", dir.display()))?;
189
190        info!("Evaluating the Nix environment");
191        let nix_dir = config.root().join(Self::DIR);
192        let status = Command::new(nix)
193            .arg("print-dev-env")
194            .arg("--profile")
195            .arg(&profile)
196            .arg(format!("path:{}", nix_dir.display()))
197            .arg("--no-update-lock-file")
198            .arg("--log-format")
199            .arg("raw")
200            .stdout(Stdio::null())
201            .status()
202            .context("Unable to run nix print-dev-env")?;
203        if !status.success() {
204            bail!("Unable to evaluate the Nix environment (exit: {})", status);
205        }
206        if !Self::is_usable(&profile) {
207            bail!(
208                "Created Nix environment '{}' is unusable",
209                profile.display()
210            );
211        }
212        Ok(profile)
213    }

215    /// Check whether `nix develop` is able to enter the provided profile:
216    /// it has to point to the JSON of a development environment, whose
217    /// `PATH` entries in the Nix store exist.
218    fn is_usable(profile: &Path) -> bool {
219        let Ok(content) = fs::read(profile) else {
220            return false;
221        };
222        let Ok(env) = serde_json::from_slice::<serde_json::Value>(&content) else {
223            return false;
224        };
225        env["variables"]["PATH"]["value"]
226            .as_str()
227            .is_some_and(|path| {
228                path.split(':')
229                    .filter(|dir| dir.starts_with("/nix/store/"))
230                    .all(|dir| Path::new(dir).exists())
231            })
232    }

234    /// Take the exclusive lock for creating the cached environment, which
235    /// waits for other instances creating it.
236    fn lock_profile(root: &Path) -> Result<Flock<File>> {
237        let path = root.join(Self::PROFILE_LOCK);
238        let file = OpenOptions::new()
239            .create(true)
240            .truncate(false)
241            .write(true)
242            .open(&path)
243            .with_context(|| format!("Unable to open lock file '{}'", path.display()))?;
244        Flock::lock(file, FlockArg::LockExclusive)
245            .map_err(|(_, e)| e)
246            .with_context(|| format!("Unable to lock '{}'", path.display()))
247    }

249    /// The path of the cached environment for the current files of the Nix
250    /// directory.
251    fn profile_path(root: &Path) -> Result<PathBuf> {
252        let nix_dir = root.join(Self::DIR);
253        let mut inputs = vec![];
254        for file in Self::FILES {
255            let path = nix_dir.join(file);
256            inputs.push(
257                fs::read(&path).with_context(|| format!("Unable to read '{}'", path.display()))?,
258            );
259        }
260        let inputs: Vec<&[u8]> = inputs.iter().map(Vec::as_slice).collect();
261        Ok(root
262            .join(Self::PROFILE_DIR)
263            .join(format!("{:016x}", content_hash(&inputs))))
264    }

266    /// Render all files of the Nix environment as (file name, content).
267    fn render_files(config: &Config) -> Result<Vec<(&'static str, String)>> {
268        for pkg in config.packages() {
269            if !pkg
270                .bytes()
271                .all(|b| b.is_ascii_alphanumeric() || b == b'-' || b == b'_' || b == b'.')
272            {
273                bail!(
274                    "Invalid package name '{}': only alphanumeric characters, dashes, underscores, and dots are allowed",
275                    pkg,
276                );
277            }
278        }
279        debug!("Adding additional packages: {:?}", config.packages());
280
281        let overlay = match config.overlay() {
282            // User defined overlay
283            Some(overlay) => {
284                info!("Using custom overlay '{}'", overlay.display());
285                fs::read_to_string(overlay).context("Unable to read custom overlay")?
286            }
287
288            // The default overlay
289            None => {
290                debug!("Using default overlay");
291                include_str!("../nix/overlay.nix").into()
292            }
293        };
294
295        let [flake, lock, packages, overlay_file] = Self::FILES;
296        Ok(vec![
297            (
298                flake,
299                include_str!("../nix/runtime-flake.nix")
300                    .replace("KUBERNIX_SYSTEM", Self::nix_system()?),
301            ),
302            (lock, include_str!("../flake.lock").into()),
303            (
304                packages,
305                include_str!("../nix/packages.nix")
306                    .replace("/* PACKAGES */", &config.packages().join(" ")),
307            ),
308            (overlay_file, overlay),
309        ])
310    }

312    /// Write the files into the directory if the directory is new or any
313    /// content differs. Returns true if something has been written.
314    fn write_files(dir: &Path, files: &[(&str, String)]) -> Result<bool> {
315        let exists = dir.exists();
316        let changed = files.iter().any(|(name, content)| {
317            fs::read_to_string(dir.join(name)).ok().as_ref() != Some(content)
318        });
319        if exists && !changed {
320            return Ok(false);
321        }
322        if exists {
323            info!("Nix environment is outdated, regenerating it");
324        }
325        create_dir_all(dir).context("Unable to create nix directory")?;
326        for (name, content) in files {
327            fs::write(dir.join(name), content)
328                .with_context(|| format!("Unable to write {}", name))?;
329        }
330        Ok(true)
331    }

333    /// Remove the git repository of the Nix environment and its
334    /// `.dockerignore`, which older versions created.
335    fn remove_leftovers(root: &Path) -> Result<()> {
336        let git = root.join(Self::DIR).join(".git");
337        if git.exists() {
338            debug!("Removing '{}'", git.display());
339            fs::remove_dir_all(&git)
340                .with_context(|| format!("Unable to remove '{}'", git.display()))?;
341        }
342        let dockerignore = root.join(".dockerignore");
343        if fs::read_to_string(&dockerignore).is_ok_and(|c| c == Self::DOCKERIGNORE) {
344            debug!("Removing '{}'", dockerignore.display());
345            fs::remove_file(&dockerignore)
346                .with_context(|| format!("Unable to remove '{}'", dockerignore.display()))?;
347        }
348        Ok(())
349    }

351    /// Returns the Nix system string for the current architecture.
352    fn nix_system() -> Result<&'static str> {
353        match std::env::consts::ARCH {
354            "x86_64" => Ok("x86_64-linux"),
355            "aarch64" => Ok("aarch64-linux"),
356            arch => bail!(
357                "unsupported architecture '{}' (only x86_64 and aarch64 Linux are supported)",
358                arch,
359            ),
360        }
361    }

363    /// Returns true if running in nix environment
364    pub fn is_active() -> bool {
365        var(Nix::NIX_ENV).is_ok()
366    }

382        let dir = tempdir()?.keep().join("nix");
383        let files = vec![("a.nix", "a".to_string()), ("b.nix", "b".to_string())];
384        assert!(Nix::write_files(&dir, &files)?);
385        assert!(!Nix::write_files(&dir, &files)?);
386
387        let changed = vec![("a.nix", "a".to_string()), ("b.nix", "c".to_string())];
388        assert!(Nix::write_files(&dir, &changed)?);
389        assert_eq!(fs::read_to_string(dir.join("b.nix"))?, "c");
390
391        fs::remove_file(dir.join("a.nix"))?;
392        assert!(Nix::write_files(&dir, &changed)?);
393        assert!(dir.join("a.nix").exists());
394        Ok(())
395    }

398    fn profile_path_depends_on_files() -> Result<()> {
399        let root = tempdir()?;
400        let dir = root.path().join(Nix::DIR);
401        let files: Vec<(&str, String)> = Nix::FILES
402            .iter()
403            .map(|name| (*name, format!("{} content", name)))
404            .collect();
405        Nix::write_files(&dir, &files)?;
406        let profile = Nix::profile_path(root.path())?;
407        assert!(profile.starts_with(root.path().join(Nix::PROFILE_DIR)));
408        assert_eq!(profile, Nix::profile_path(root.path())?);
409
410        fs::write(dir.join("overlay.nix"), "changed")?;
411        assert_ne!(profile, Nix::profile_path(root.path())?);
412
413        fs::remove_file(dir.join("flake.lock"))?;
414        assert!(Nix::profile_path(root.path()).is_err());
415        Ok(())
416    }
