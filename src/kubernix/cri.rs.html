22/// Environment variable name for the CRI socket endpoint.
23pub const RUNTIME_ENV: &str = "CONTAINER_RUNTIME_ENDPOINT";

25/// Resolved paths for the OCI runtime and CNI plugins, shared between
26/// CRI-O and containerd startup.
27pub struct RuntimePaths {
28    pub crun: String,
29    pub plugin_dir: String,
30}
31
32impl RuntimePaths {
33    /// Resolve crun and CNI plugin paths. In multi-node non-rootless mode,
34    /// binaries are resolved from $PATH inside the container at runtime.
35    pub fn resolve(config: &Config) -> Result<Self> {
36        use crate::system::System;
37        if config.multi_node() && !config.is_rootless() {
38            Ok(Self {
39                crun: "crun".to_string(),
40                plugin_dir: "/tmp/cni-plugins".to_string(),
41            })
42        } else {
43            let crun = System::find_executable("crun")?.display().to_string();
44            let loopback = System::find_executable("loopback")?;
45            let plugin_dir = loopback
46                .parent()
47                .context("Unable to find CNI plugin dir")?
48                .display()
49                .to_string();
50            Ok(Self { crun, plugin_dir })
51        }
52    }
53}

55/// Maximum usable bytes in a Unix socket path (108 - 1 for null terminator).
56pub const MAX_SOCKET_PATH_LEN: usize = 107;
57
58/// Simple CRI socket abstraction.
59pub struct CriSocket(PathBuf);
60
61impl Display for CriSocket {
62    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
63        write!(f, "{}", self.0.display())
64    }
65}
66
67impl CriSocket {
68    /// Create a new CRI socket, validating the path length against Unix limits.
69    pub fn new(path: PathBuf) -> Result<CriSocket> {
70        if path.as_os_str().len() > MAX_SOCKET_PATH_LEN {
71            bail!("Socket path '{}' is too long", path.display())
72        }
73        Ok(CriSocket(path))
74    }
75
76    /// Format the socket path as a `unix://` URI for CRI clients.
77    pub fn to_socket_string(&self) -> String {
78        format!("unix://{}", self.0.display())
79    }
80}

82/// Return the CRI socket for the configured runtime and node.
83pub fn cri_socket(config: &Config, network: &Network, node: u8) -> Result<CriSocket> {
84    match config.cri_runtime() {
85        CriRuntime::Crio => Crio::socket(config, network, node),
86        CriRuntime::Containerd => Containerd::socket(config, network, node),
87    }
88}

90/// Write CNI network configuration for a node, selecting the appropriate
91/// plugin based on whether rootless mode is active.
92pub fn write_pod_network_config(
93    config: &Config,
94    cni_conf_dir: &Path,
95    node_name: &str,
96    node: u8,
97    network: &Network,
98) -> Result<()> {
99    if config.is_rootless() {
100        write_rootless_cni_config(cni_conf_dir, node_name, node, network)
101    } else {
102        write_cni_config(cni_conf_dir, node_name, node, network)
103    }
104}

106/// Write the CNI bridge network configuration for a node (root mode).
107fn write_cni_config(
108    cni_conf_dir: &Path,
109    node_name: &str,
110    node: u8,
111    network: &Network,
112) -> Result<()> {
113    let cidr = network
114        .pod_cidrs()
115        .get(node as usize)
116        .with_context(|| format!("Unable to find CIDR for {}", node_name))?;
117    fs::write(
118        cni_conf_dir.join("10-bridge.json"),
119        to_string_pretty(&json!({
120            "cniVersion": "0.3.1",
121            "name": format!("kubernix-{}", node_name),
122            "type": "bridge",
123            "bridge": format!("{}.{}", Network::INTERFACE_PREFIX, node),
124            "isGateway": true,
125            "ipMasq": true,
126            "hairpinMode": true,
127            "ipam": {
128                "type": "host-local",
129                "routes": [{ "dst": "0.0.0.0/0" }],
130                "ranges": [[{ "subnet": cidr }]]
131            }
132        }))?,
133    )?;
134    Ok(())
135}

164/// Remove all containers and pods via crictl.
165pub fn remove_all_containers(label: &str, socket: &CriSocket, node_name: &str) -> Result<()> {
166    debug!("Removing all {} workloads on {}", label, node_name);
167
168    let output = Command::new("crictl")
169        .env(RUNTIME_ENV, socket.to_socket_string())
170        .arg("pods")
171        .arg("-q")
172        .output()?;
173    let stdout = String::from_utf8(output.stdout)?;
174    if !output.status.success() {
175        debug!("crictl pods stdout ({}): {}", node_name, stdout);
176        debug!(
177            "crictl pods stderr ({}): {}",
178            node_name,
179            String::from_utf8(output.stderr)?
180        );
181        bail!("crictl pods command failed ({})", node_name);
182    }
183
184    for x in stdout.lines() {
185        debug!("Removing pod {} on {}", x, node_name);
186        let output = Command::new("crictl")
187            .env(RUNTIME_ENV, socket.to_socket_string())
188            .arg("rmp")
189            .arg("-f")
190            .arg(x)
191            .output()?;
192        if !output.status.success() {
193            debug!("crictl rmp ({}): {:?}", node_name, output);
194            bail!("crictl rmp command failed ({})", node_name);
195        }
196    }
197
198    debug!("All workloads removed on {}", node_name);
199    Ok(())
200}
