31/// Resolved paths for the OCI runtimes and CNI plugins, shared between
32/// CRI-O and containerd startup.
33pub struct RuntimePaths {
34    /// Available OCI runtimes and their executable paths, registered as
35    /// runtime handlers.
36    pub runtimes: Vec<(OciRuntime, String)>,
37    pub plugin_dir: String,
38}
39
40impl RuntimePaths {
41    /// Resolve the OCI runtime and CNI plugin paths. Node containers mount
42    /// the Nix store of the host, so the paths are valid in all modes.
43    pub fn resolve(config: &Config) -> Result<Self> {
44        let runtimes = Self::available_runtimes(config.oci_runtime(), |r| {
45            Ok(System::find_executable(r.name())?.display().to_string())
46        })?;
47        let loopback = System::find_executable("loopback")?;
48        let plugin_dir = loopback
49            .parent()
50            .context("Unable to find CNI plugin dir")?
51            .display()
52            .to_string();
53        Ok(Self {
54            runtimes,
55            plugin_dir,
56        })
57    }
58
59    /// The OCI runtimes with the executable which the CRI runtime runs: the
60    /// rootless wrapper in the node directory `dir` in rootless mode (see
61    /// [`write_rootless_wrappers`]), otherwise the runtime itself.
62    pub fn handlers(&self, config: &Config, dir: &Path) -> Vec<(OciRuntime, String)> {
63        self.runtimes
64            .iter()
65            .map(|(runtime, path)| {
66                let path = if config.is_rootless() {
67                    rootless_wrapper(dir, *runtime).display().to_string()
68                } else {
69                    path.clone()
70                };
71                (*runtime, path)
72            })
73            .collect()
74    }
75
76    /// Look up all OCI runtimes. A missing default runtime is an error,
77    /// while other missing runtimes are skipped with a warning, for
78    /// example in an environment created by an older kubernix version.
79    fn available_runtimes<F>(default: OciRuntime, find: F) -> Result<Vec<(OciRuntime, String)>>
80    where
81        F: Fn(OciRuntime) -> Result<String>,
82    {
83        let mut runtimes = vec![];
84        for runtime in OciRuntime::ALL {
85            match find(runtime) {
86                Ok(path) => runtimes.push((runtime, path)),
87                Err(e) if runtime == default => {
88                    return Err(e.context(format!(
89                        "Unable to find the default OCI runtime '{}'",
90                        runtime
91                    )));
92                }
93                Err(e) => warn!(
94                    "Not registering the OCI runtime handler '{}': {:#}",
95                    runtime, e
96                ),
97            }
98        }
99        Ok(runtimes)
100    }
101}

103/// Path of the rootless wrapper script for the given OCI runtime.
104pub fn rootless_wrapper(dir: &Path, runtime: OciRuntime) -> PathBuf {
105    dir.join(format!("{}-rootless", runtime))
106}
107
108/// Write a wrapper script for every OCI runtime, used as runtime path in
109/// rootless mode. The wrapper patches the `config.json` of the bundle
110/// before running the runtime:
111///
112/// - Lowering the OOM score is not permitted in a user namespace, so both
113///   crun and runc fail writing `/proc/self/oom_score_adj` for the values
114///   which the kubelet or the CRI runtime (like for the infra container)
115///   set. The wrapper raises lower values to its own OOM score, which keeps
116///   the higher scores of BestEffort and Burstable pods.
117/// - It drops namespace paths which point to the namespaces of the runtime
118///   itself, which it cannot join.
119/// - It adds a private UTS namespace, which setting the hostname requires,
120///   unless one is left.
121///
122/// The wrapper fails if it cannot patch the `config.json`. It runs jq and
123/// the runtime by their absolute paths, so it does not depend on the `PATH`
124/// of the CRI runtime.
125pub fn write_rootless_wrappers(dir: &Path, paths: &RuntimePaths) -> Result<()> {
126    let jq = System::find_executable("jq")?;
127    write_rootless_wrappers_with(dir, paths, &jq.display().to_string())
128}
129
130fn write_rootless_wrappers_with(dir: &Path, paths: &RuntimePaths, jq: &str) -> Result<()> {
131    use std::os::unix::fs::PermissionsExt;
132    for (runtime, path) in &paths.runtimes {
133        let wrapper = rootless_wrapper(dir, *runtime);
134        let script = include_str!("assets/oci-rootless.sh")
135            .replace("__JQ_PATH__", &shell::quote(jq))
136            .replace("__RUNTIME_PATH__", &shell::quote(path));
137        write_if_changed(&wrapper, &script)
138            .with_context(|| format!("Unable to write {} rootless wrapper", runtime))?;
139        fs::set_permissions(&wrapper, fs::Permissions::from_mode(0o755))
140            .with_context(|| format!("Unable to set {} rootless wrapper permissions", runtime))?;
141    }
142    Ok(())
143}

148/// Simple CRI socket abstraction.
149pub struct CriSocket(PathBuf);
150
151impl Display for CriSocket {
152    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
153        write!(f, "{}", self.0.display())
154    }
155}
156
157impl CriSocket {
158    /// Create a new CRI socket, validating the path length against Unix limits.
159    pub fn new(path: PathBuf) -> Result<CriSocket> {
160        if path.as_os_str().len() > MAX_SOCKET_PATH_LEN {
161            bail!("Socket path '{}' is too long", path.display())
162        }
163        Ok(CriSocket(path))
164    }
165
166    /// Format the socket path as a `unix://` URI for CRI clients.
167    pub fn to_socket_string(&self) -> String {
168        format!("unix://{}", self.0.display())
169    }
170}

180/// Directory used by the host-local IPAM plugin to store the allocated IPs,
181/// instead of the default `/var/lib/cni/networks`, which is not writable in
182/// rootless mode. CNI skips directories when loading the network configs.
183fn ipam_data_dir(cni_conf_dir: &Path) -> PathBuf {
184    cni_conf_dir.join("networks")
185}

187/// Remove the IP allocations of the host-local IPAM plugin before the CRI
188/// runtime starts. A clean stop releases them when removing the pods, but
189/// after a force quit nothing else would. The allocations of a former run
190/// are stale in any case: no pod of the node can run while the run root is
191/// locked and the CRI runtime is not started, and the bridges of a former
192/// run get removed on startup (see `Network::setup_bridges`), which
193/// detaches the network of a container that may have survived a force
194/// quit. Every rootless run gets a new network namespace anyway.
195pub fn reset_ipam_data(cni_conf_dir: &Path) -> Result<()> {
196    let dir = ipam_data_dir(cni_conf_dir);
197    if dir.exists() {
198        fs::remove_dir_all(&dir)
199            .with_context(|| format!("Unable to remove IPAM data '{}'", dir.display()))?;
200    }
201    Ok(())
202}

204/// Write the CNI bridge network configuration for a node.
205///
206/// In root mode, kubernix creates the bridge with the gateway address
207/// before (see `Network::setup_bridges`), which the plugin keeps. The
208/// plugin checks the IPv4 addresses of the bridge in the order the kernel
209/// lists them until it finds the gateway, and fails on any other address
210/// before it. Addresses added later with the same scope come after the
211/// gateway, but the kernel lists addresses with a narrower scope first,
212/// like link-local ones (`scope link`) of an IPv4LL client, and all others
213/// come first once the gateway got removed. With `forceAddress`, the plugin
214/// removes these addresses instead of failing to set up the pods. With
215/// `portmap`, the portmap plugin maps the host ports of the pods, which
216/// containerd passes as capability. CRI-O maps them itself, so the plugin
217/// would map them twice.
218pub fn write_cni_config(
219    cni_conf_dir: &Path,
220    node_name: &str,
221    node: u8,
222    network: &Network,
223    portmap: bool,
224) -> Result<()> {
225    let cidr = network
226        .pod_cidrs()
227        .get(node as usize)
228        .with_context(|| format!("Unable to find CIDR for {}", node_name))?;
229    let mut plugins = vec![json!({
230        "type": "bridge",
231        "bridge": network.bridge(node),
232        "isGateway": true,
233        "forceAddress": true,
234        "ipMasq": true,
235        "hairpinMode": true,
236        "ipam": {
237            "type": "host-local",
238            "dataDir": ipam_data_dir(cni_conf_dir),
239            "routes": [{ "dst": "0.0.0.0/0" }],
240            "ranges": [[{ "subnet": cidr }]]
241        }
242    })];
243    if portmap {
244        plugins.push(json!({
245            "type": "portmap",
246            "capabilities": { "portMappings": true }
247        }));
248    }
249    // The firewall plugin accepts the pod traffic in the FORWARD chain,
250    // which may have a DROP policy, for example when Docker is running.
251    plugins.push(json!({ "type": "firewall" }));
252    write_file(
253        &cni_conf_dir.join("10-bridge.conflist"),
254        to_string_pretty(&json!({
255            "cniVersion": "1.0.0",
256            "name": format!("kubernix-{}", node_name),
257            "plugins": plugins
258        }))?
259        .as_bytes(),
260    )
261}

263/// Timeout of the crictl requests. The CRI runtime stops the containers of
264/// a removed pod within the deadline of the request, which defaults to two
265/// seconds in crictl, while the kubelet grants 30 seconds by default.
266const CRICTL_TIMEOUT: &str = "--timeout=30s";
267
268/// How often removing a pod gets tried.
269const RMP_ATTEMPTS: usize = 3;
270
271/// Delay between the attempts to remove a pod.
272const RMP_RETRY_DELAY: Duration = if cfg!(test) {
273    Duration::from_millis(10)
274} else {
275    Duration::from_secs(1)
276};
277
278/// Executables which monitor the containers of a pod. They run in their own
279/// process groups and outlive the CRI runtime if removing a pod failed.
280const MONITORS: &[&str] = &["conmonrs", "conmon", "containerd-shim-runc-v2"];
281
282/// Kill the container monitors of the node which are left after the CRI
283/// runtime stopped, found by the node directory in their command line.
284pub fn kill_leftover_monitors(socket: &CriSocket, node_name: &str) {
285    let Some(dir) = socket.0.parent().and_then(Path::to_str) else {
286        return;
287    };
288    let leftovers = process::find_processes_containing(MONITORS, dir);
289    if !leftovers.is_empty() {
290        warn!(
291            "Killing {} leftover container monitor processes on {}",
292            leftovers.len(),
293            node_name
294        );
295        process::signal_processes(&leftovers, Signal::SIGKILL);
296    }
297}

299/// Remove all containers and pods via crictl.
300pub fn remove_all_containers(label: &str, socket: &CriSocket, node_name: &str) -> Result<()> {
301    debug!("Removing all {} workloads on {}", label, node_name);
302    remove_all_pods(OsStr::new("crictl"), socket, node_name)?;
303    debug!("All workloads removed on {}", node_name);
304    Ok(())
305}

307/// Remove all pods via the crictl executable `crictl`. Every pod gets
308/// removed, even if removing another one failed, since the monitors of
309/// their containers run in their own process groups and would be leaked.
310fn remove_all_pods(crictl: &OsStr, socket: &CriSocket, node_name: &str) -> Result<()> {
311    let command = |args: &[&str]| {
312        // An interrupt in the terminal must not abort the removal
313        let mut command = process::command(crictl);
314        command
315            .env(RUNTIME_ENV, socket.to_socket_string())
316            .arg(CRICTL_TIMEOUT)
317            .args(args);
318        command
319    };
320
321    let output = command(&["pods", "-q"])
322        .output()
323        .context("Unable to run crictl")?;
324    let stdout = String::from_utf8_lossy(&output.stdout);
325    if !output.status.success() {
326        debug!("crictl pods stdout ({}): {}", node_name, stdout);
327        debug!(
328            "crictl pods stderr ({}): {}",
329            node_name,
330            String::from_utf8_lossy(&output.stderr)
331        );
332        bail!("crictl pods command failed ({})", node_name);
333    }
334
335    let pods: Vec<&str> = stdout.lines().filter(|x| !x.is_empty()).collect();
336    let mut failed = pods.clone();
337    for attempt in 1..=RMP_ATTEMPTS {
338        if attempt > 1 {
339            // The runtime may still process a request of the stopped
340            // kubelet for the same pod, which makes the removal fail once
341            thread::sleep(RMP_RETRY_DELAY);
342        }
343        let pending = std::mem::take(&mut failed);
344        for pod in pending {
345            debug!(
346                "Removing pod {} on {} (attempt {})",
347                pod, node_name, attempt
348            );
349            match command(&["rmp", "-f", pod]).output() {
350                Ok(output) if output.status.success() => {}
351                Ok(output) => {
352                    debug!("crictl rmp {} ({}): {:?}", pod, node_name, output);
353                    failed.push(pod);
354                }
355                Err(e) => {
356                    debug!("crictl rmp {} ({}): {}", pod, node_name, e);
357                    failed.push(pod);
358                }
359            }
360        }
361        if failed.is_empty() {
362            break;
363        }
364    }
365    if !failed.is_empty() {
366        bail!(
367            "Unable to remove {} of {} pods ({}): {}",
368            failed.len(),
369            pods.len(),
370            node_name,
371            failed.join(", ")
372        );
373    }
374    Ok(())
375}

501    /// Run a command, retrying while another test thread holds the freshly
502    /// written executable open for writing during a fork.
503    fn output(command: &mut Command) -> Result<std::process::Output> {
504        for _ in 0..50 {
505            match command.output() {
506                Err(e) if e.raw_os_error() == Some(nix::libc::ETXTBSY) => {
507                    std::thread::sleep(std::time::Duration::from_millis(20))
508                }
509                result => return Ok(result?),

515    /// A rootless wrapper around a fake runtime, which records its arguments
516    /// in `<dir>/runtime.args`.
517    struct Wrapper {
518        dir: tempfile::TempDir,
519    }
520
521    impl Wrapper {
522        /// Create the wrapper, `None` if jq is not available.
523        fn new() -> Result<Option<Self>> {
524            let Ok(jq) = System::find_executable("jq") else {
525                eprintln!("Skipping the rootless wrapper test, jq not found");
526                return Ok(None);
527            };
528            let dir = tempfile::tempdir()?;
529            let runtime = dir.path().join("runtime");
530            write_script(&runtime, "#!/bin/sh\nprintf '%s\\n' \"$@\" > \"$0.args\"\n")?;
531            let paths = RuntimePaths {
532                runtimes: vec![(OciRuntime::Crun, runtime.display().to_string())],
533                plugin_dir: "/nix/cni".into(),
534            };
535            write_rootless_wrappers_with(dir.path(), &paths, &jq.display().to_string())?;
536            fs::create_dir(dir.path().join("bundle"))?;
537            Ok(Some(Self { dir }))
538        }
539
540        fn bundle(&self) -> PathBuf {
541            self.dir.path().join("bundle")
542        }
543
544        /// Run the wrapper with the arguments.
545        fn run(&self, args: &[&str]) -> Result<std::process::Output> {
546            output(Command::new(rootless_wrapper(self.dir.path(), OciRuntime::Crun)).args(args))
547        }
548
549        /// Run `create` on a bundle with the config, and return the patched
550        /// config.
551        fn create(&self, config: &serde_json::Value) -> Result<serde_json::Value> {
552            let bundle = self.bundle();
553            fs::write(bundle.join("config.json"), config.to_string())?;
554            let bundle = bundle.display().to_string();
555            let output = self.run(&["create", "--bundle", &bundle, "id"])?;
556            assert!(output.status.success(), "{:?}", output);
557            assert_eq!(
558                self.runtime_args()?,
559                Some(format!("create\n--bundle\n{}\nid\n", bundle))
560            );
561            Ok(serde_json::from_str(&fs::read_to_string(
562                self.bundle().join("config.json"),
563            )?)?)
564        }
565
566        fn runtime_args(&self) -> Result<Option<String>> {
567            let file = self.dir.path().join("runtime.args");
568            Ok(file
569                .exists()
570                .then(|| fs::read_to_string(file))
571                .transpose()?)
572        }
573    }

665    fn remove_all_pods_tries_every_pod() -> Result<()> {
666        let dir = tempfile::tempdir()?;
667        let crictl = dir.path().join("crictl");
668        write_script(
669            &crictl,
670            "#!/bin/sh\n\
671             echo \"$CONTAINER_RUNTIME_ENDPOINT $*\" >> \"$0.calls\"\n\
672             case \"$*\" in\n\
673             *' pods -q') printf 'a\\nb\\nc\\n' ;;\n\
674             *' rmp -f b') exit 1 ;;\n\
675             esac\n",
676        )?;
677        // Not bound, a fixed path fits into a unix socket path for any TMPDIR
678        let socket = CriSocket::new("/kx/crio.sock".into())?;
679        let err = loop {
680            match remove_all_pods(crictl.as_os_str(), &socket, "node") {
681                Err(e) if format!("{:#}", e).contains("busy") => {
682                    std::thread::sleep(std::time::Duration::from_millis(20))
683                }
684                result => break result.unwrap_err(),
685            }
686        };
687        assert_eq!(
688            format!("{:#}", err),
689            "Unable to remove 1 of 3 pods (node): b"
690        );
691        let endpoint = socket.to_socket_string();
692        let calls: Vec<String> = [
693            "pods -q", "rmp -f a", "rmp -f b", "rmp -f c", "rmp -f b", "rmp -f b",
694        ]
695        .iter()
696        .map(|args| format!("{} --timeout=30s {}\n", endpoint, args))
697        .collect();
698        assert_eq!(
699            fs::read_to_string(dir.path().join("crictl.calls"))?,
700            calls.concat()
701        );
702        Ok(())
703    }
