26/// Resolved paths for the OCI runtimes and CNI plugins, shared between
27/// CRI-O and containerd startup.
28pub struct RuntimePaths {
29    /// Available OCI runtimes and their executable paths, registered as
30    /// runtime handlers.
31    pub runtimes: Vec<(OciRuntime, String)>,
32    pub plugin_dir: String,
33}
34
35impl RuntimePaths {
36    /// Resolve the OCI runtime and CNI plugin paths. Node containers mount
37    /// the Nix store of the host, so the paths are valid in all modes.
38    pub fn resolve(config: &Config) -> Result<Self> {
39        use crate::system::System;
40        let runtimes = Self::available_runtimes(config.oci_runtime(), |r| {
41            Ok(System::find_executable(r.name())?.display().to_string())
42        })?;
43        let loopback = System::find_executable("loopback")?;
44        let plugin_dir = loopback
45            .parent()
46            .context("Unable to find CNI plugin dir")?
47            .display()
48            .to_string();
49        Ok(Self {
50            runtimes,
51            plugin_dir,
52        })
53    }
54
55    /// Look up all OCI runtimes. A missing default runtime is an error,
56    /// while other missing runtimes are skipped with a warning, for
57    /// example in an environment created by an older kubernix version.
58    fn available_runtimes<F>(default: OciRuntime, find: F) -> Result<Vec<(OciRuntime, String)>>
59    where
60        F: Fn(OciRuntime) -> Result<String>,
61    {
62        let mut runtimes = vec![];
63        for runtime in OciRuntime::ALL {
64            match find(runtime) {
65                Ok(path) => runtimes.push((runtime, path)),
66                Err(e) if runtime == default => {
67                    return Err(e.context(format!(
68                        "Unable to find the default OCI runtime '{}'",
69                        runtime
70                    )));
71                }
72                Err(e) => warn!(
73                    "Not registering the OCI runtime handler '{}': {:#}",
74                    runtime, e
75                ),
76            }
77        }
78        Ok(runtimes)
79    }
80}

82/// Path of the rootless wrapper script for the given OCI runtime.
83pub fn rootless_wrapper(dir: &Path, runtime: OciRuntime) -> PathBuf {
84    dir.join(format!("{}-rootless", runtime))
85}
86
87/// Write a wrapper script for every OCI runtime, used as runtime path in
88/// rootless mode. Lowering the OOM score is not permitted in a user
89/// namespace, so both crun and runc fail writing `/proc/self/oom_score_adj`
90/// for the values the kubelet or the CRI runtime (like for the infra
91/// container) set. The wrapper patches the `config.json` of the bundle
92/// before running the runtime: it removes `oomScoreAdj`, adds a private UTS
93/// namespace, which setting the hostname requires, and drops namespace
94/// paths which point to the namespaces of the runtime itself.
95pub fn write_rootless_wrappers(dir: &Path, paths: &RuntimePaths) -> Result<()> {
96    use std::os::unix::fs::PermissionsExt;
97    for (runtime, path) in &paths.runtimes {
98        let wrapper = rootless_wrapper(dir, *runtime);
99        let script = include_str!("assets/oci-rootless.sh").replace("__RUNTIME_PATH__", path);
100        write_if_changed(&wrapper, &script)
101            .with_context(|| format!("Unable to write {} rootless wrapper", runtime))?;
102        fs::set_permissions(&wrapper, fs::Permissions::from_mode(0o755))
103            .with_context(|| format!("Unable to set {} rootless wrapper permissions", runtime))?;
104    }
105    Ok(())
106}

108/// Maximum usable bytes in a Unix socket path (108 - 1 for null terminator).
109pub const MAX_SOCKET_PATH_LEN: usize = 107;
110
111/// Simple CRI socket abstraction.
112pub struct CriSocket(PathBuf);
113
114impl Display for CriSocket {
115    fn fmt(&self, f: &mut Formatter<'_>) -> fmt::Result {
116        write!(f, "{}", self.0.display())
117    }
118}
119
120impl CriSocket {
121    /// Create a new CRI socket, validating the path length against Unix limits.
122    pub fn new(path: PathBuf) -> Result<CriSocket> {
123        if path.as_os_str().len() > MAX_SOCKET_PATH_LEN {
124            bail!("Socket path '{}' is too long", path.display())
125        }
126        Ok(CriSocket(path))
127    }
128
129    /// Format the socket path as a `unix://` URI for CRI clients.
130    pub fn to_socket_string(&self) -> String {
131        format!("unix://{}", self.0.display())
132    }
133}

135/// Return the CRI socket for the configured runtime and node.
136pub fn cri_socket(config: &Config, network: &Network, node: u8) -> Result<CriSocket> {
137    match config.cri_runtime() {
138        CriRuntime::Crio => Crio::socket(config, network, node),
139        CriRuntime::Containerd => Containerd::socket(config, network, node),
140    }
141}

143/// Directory used by the host-local IPAM plugin to store the allocated IPs,
144/// instead of the default `/var/lib/cni/networks`, which is not writable in
145/// rootless mode. CNI skips directories when loading the network configs.
146fn ipam_data_dir(cni_conf_dir: &Path) -> PathBuf {
147    cni_conf_dir.join("networks")
148}
149
150/// Remove the IP allocations of the host-local IPAM plugin. Every rootless
151/// run gets a new network namespace, so the allocations of a former run are
152/// stale, and after an unclean stop nothing else would release them.
153pub fn reset_ipam_data(cni_conf_dir: &Path) -> Result<()> {
154    let dir = ipam_data_dir(cni_conf_dir);
155    if dir.exists() {
156        fs::remove_dir_all(&dir)
157            .with_context(|| format!("Unable to remove IPAM data '{}'", dir.display()))?;
158    }
159    Ok(())
160}

162/// Write the CNI bridge network configuration for a node.
163pub fn write_cni_config(
164    cni_conf_dir: &Path,
165    node_name: &str,
166    node: u8,
167    network: &Network,
168) -> Result<()> {
169    let cidr = network
170        .pod_cidrs()
171        .get(node as usize)
172        .with_context(|| format!("Unable to find CIDR for {}", node_name))?;
173    // The firewall plugin accepts the pod traffic in the FORWARD chain,
174    // which may have a DROP policy, for example when Docker is running.
175    fs::write(
176        cni_conf_dir.join("10-bridge.conflist"),
177        to_string_pretty(&json!({
178            "cniVersion": "1.0.0",
179            "name": format!("kubernix-{}", node_name),
180            "plugins": [
181                {
182                    "type": "bridge",
183                    "bridge": format!("{}.{}", Network::INTERFACE_PREFIX, node),
184                    "isGateway": true,
185                    "ipMasq": true,
186                    "hairpinMode": true,
187                    "ipam": {
188                        "type": "host-local",
189                        "dataDir": ipam_data_dir(cni_conf_dir),
190                        "routes": [{ "dst": "0.0.0.0/0" }],
191                        "ranges": [[{ "subnet": cidr }]]
192                    }
193                },
194                { "type": "firewall" }
195            ]
196        }))?,
197    )?;
198    Ok(())
199}

201/// Remove all containers and pods via crictl.
202pub fn remove_all_containers(label: &str, socket: &CriSocket, node_name: &str) -> Result<()> {
203    debug!("Removing all {} workloads on {}", label, node_name);
204
205    let output = Command::new("crictl")
206        .env(RUNTIME_ENV, socket.to_socket_string())
207        .arg("pods")
208        .arg("-q")
209        .output()?;
210    let stdout = String::from_utf8(output.stdout)?;
211    if !output.status.success() {
212        debug!("crictl pods stdout ({}): {}", node_name, stdout);
213        debug!(
214            "crictl pods stderr ({}): {}",
215            node_name,
216            String::from_utf8(output.stderr)?
217        );
218        bail!("crictl pods command failed ({})", node_name);
219    }
220
221    for x in stdout.lines() {
222        debug!("Removing pod {} on {}", x, node_name);
223        let output = Command::new("crictl")
224            .env(RUNTIME_ENV, socket.to_socket_string())
225            .arg("rmp")
226            .arg("-f")
227            .arg(x)
228            .output()?;
229        if !output.status.success() {
230            debug!("crictl rmp ({}): {:?}", node_name, output);
231            bail!("crictl rmp command failed ({})", node_name);
232        }
233    }
234
235    debug!("All workloads removed on {}", node_name);
236    Ok(())
237}

301    #[test]
302    fn write_rootless_wrappers_success() -> Result<()> {
303        use std::os::unix::fs::PermissionsExt;
304        let dir = tempfile::tempdir()?;
305        let paths = RuntimePaths {
306            runtimes: vec![
307                (OciRuntime::Crun, "/nix/bin/crun".into()),
308                (OciRuntime::Runc, "/nix/bin/runc".into()),
309            ],
310            plugin_dir: "/nix/cni".into(),
311        };
312        // Writing them twice keeps them unchanged
313        for _ in 0..2 {
314            write_rootless_wrappers(dir.path(), &paths)?;
315        }
316        for (runtime, path) in &paths.runtimes {
317            let wrapper = rootless_wrapper(dir.path(), *runtime);
318            assert_eq!(wrapper, dir.path().join(format!("{}-rootless", runtime)));
319            let script = std::fs::read_to_string(&wrapper)?;
320            assert!(script.starts_with("#!/bin/sh\n"));
321            assert!(script.contains("del(.process.oomScoreAdj)"));
322            assert!(script.ends_with(&format!("exec {} \"$@\"\n", path)));
323            assert!(!script.contains("__RUNTIME_PATH__"));
324            let mode = std::fs::metadata(&wrapper)?.permissions().mode();
325            assert_eq!(mode & 0o777, 0o755);
326        }
327        Ok(())
328    }
