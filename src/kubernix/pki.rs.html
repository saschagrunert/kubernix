1//! Public key infrastructure (PKI) for cluster TLS.
2//!
3//! Generates a self-signed CA and per-component certificates using
4//! `cfssl`/`cfssljson`. Certificates cover the admin user, API server,
5//! etcd, kubelet nodes, controller-manager, scheduler, proxy, and service
6//! accounts.
7

37impl Pki {
38    /// Cluster administrator identity (system:masters group).
39    pub fn admin(&self) -> &Identity {
40        &self.admin
41    }
42
43    /// API server TLS and etcd client identity.
44    pub fn apiserver(&self) -> &Identity {

46    }
47
48    /// Certificate authority used to sign all other certificates.
49    pub fn ca(&self) -> &Identity {
50        &self.ca
51    }
52
53    /// Controller manager identity (system:kube-controller-manager).
54    pub fn controller_manager(&self) -> &Identity {
55        &self.controller_manager
56    }
57
58    /// etcd server and peer TLS identity.
59    pub fn etcd(&self) -> &Identity {
60        &self.etcd
61    }
62
63    /// Per-node kubelet identities (`system:node:<name>`).
64    pub fn kubelets(&self) -> &[Identity] {
65        &self.kubelets
66    }
67
68    /// Kube-proxy identity (system:kube-proxy).
69    pub fn proxy(&self) -> &Identity {
70        &self.proxy
71    }
72
73    /// Scheduler identity (system:kube-scheduler).
74    pub fn scheduler(&self) -> &Identity {
75        &self.scheduler
76    }
77
78    /// Key pair used to sign and verify ServiceAccount tokens.
79    pub fn service_account(&self) -> &Identity {
80        &self.service_account
81    }
82
83    /// All identities, which have to be extended for new ones.
84    fn identities(&self) -> Vec<&Identity> {
85        let mut identities = vec![
86            &self.admin,
87            &self.apiserver,
88            &self.ca,
89            &self.controller_manager,
90            &self.etcd,
91            &self.proxy,
92            &self.scheduler,
93            &self.service_account,
94        ];
95        identities.extend(&self.kubelets);
96        identities
97    }
98
99    /// Returns true if the files of all identities exist.
100    fn complete(&self) -> bool {
101        self.identities()
102            .iter()
103            .all(|i| i.cert().exists() && i.key().exists())
104    }
105}

107#[must_use]
108pub struct Identity {
109    name: String,
110    user: String,
111    cert: PathBuf,
112    key: PathBuf,
113}
114
115impl Identity {
116    /// Short identifier used for file naming (e.g. `admin`, `kube-proxy`).
117    pub fn name(&self) -> &str {
118        &self.name
119    }
120
121    /// Kubernetes user or CN embedded in the certificate.
122    pub fn user(&self) -> &str {
123        &self.user
124    }
125
126    /// Path to the PEM-encoded certificate file.
127    pub fn cert(&self) -> &Path {
128        &self.cert
129    }
130
131    /// Path to the PEM-encoded private key file.
132    pub fn key(&self) -> &Path {
133        &self.key
134    }
135
136    /// Create a new identity with derived cert/key paths under `dir`.
137    pub fn new(dir: &Path, name: &str, user: &str) -> Identity {
138        Identity {
139            cert: dir.join(format!("{}.pem", name)),
140            key: dir.join(format!("{}-key.pem", name)),
141            name: name.into(),
142            user: user.into(),
143        }
144    }
145}

172const ADMIN_NAME: &str = "admin";
173const APISERVER_NAME: &str = "kubernetes";
174const CA_NAME: &str = "ca";
175const CONTROLLER_MANAGER_NAME: &str = "kube-controller-manager";
176const CONTROLLER_MANAGER_USER: &str = "system:kube-controller-manager";
177const ETCD_NAME: &str = "etcd";
178const PROXY_NAME: &str = "kube-proxy";
179const PROXY_USER: &str = "system:kube-proxy";
180const SCHEDULER_NAME: &str = "kube-scheduler";
181const SCHEDULER_USER: &str = "system:kube-scheduler";
182const SERVICE_ACCOUNT_NAME: &str = "service-account";

184impl Pki {
185    /// Generate or load all cluster certificates.
186    ///
187    /// If the PKI directory already exists, identities are loaded from
188    /// the existing files. Otherwise, a new CA and all component
189    /// certificates are generated via cfssl.
190    pub fn new(config: &Config, network: &Network) -> Result<Pki> {
191        let dir = &config.root().join("pki");
192        let nodes = (0..config.nodes())
193            .map(|n| Node::name(config, network, n))
194            .collect::<Vec<String>>();
195
196        // Create the CA only if necessary
197        if dir.exists() {
198            let kubelets = if config.multi_node() {
199                // Multiple nodes get identified via their node name
200                nodes
201                    .iter()
202                    .map(|n| Identity::new(dir, n, &Self::node_user(n)))
203                    .collect()
204            } else {
205                // Single node gets identified via its hostname
206                vec![Identity::new(
207                    dir,
208                    network.hostname(),
209                    &Self::node_user(network.hostname()),
210                )]
211            };
212
213            let pki = Pki {
214                admin: Identity::new(dir, ADMIN_NAME, ADMIN_NAME),
215                apiserver: Identity::new(dir, APISERVER_NAME, APISERVER_NAME),
216                ca: Identity::new(dir, CA_NAME, CA_NAME),
217                controller_manager: Identity::new(
218                    dir,
219                    CONTROLLER_MANAGER_NAME,
220                    CONTROLLER_MANAGER_USER,
221                ),
222                etcd: Identity::new(dir, ETCD_NAME, ETCD_NAME),
223                kubelets,
224                proxy: Identity::new(dir, PROXY_NAME, PROXY_USER),
225                scheduler: Identity::new(dir, SCHEDULER_NAME, SCHEDULER_USER),
226                service_account: Identity::new(dir, SERVICE_ACCOUNT_NAME, SERVICE_ACCOUNT_NAME),
227            };
228
229            // A failed generation leaves an incomplete directory behind
230            if !pki.complete() {
231                warn!("PKI directory is incomplete, regenerating it");
232                fs::remove_dir_all(dir).context("Unable to remove incomplete PKI directory")?;
233
234                // The kubeconfigs embed the certificates
235                let kubeconfig = config.root().join(KubeConfig::DIR);
236                if kubeconfig.exists() {
237                    fs::remove_dir_all(&kubeconfig)
238                        .context("Unable to remove outdated kubeconfig directory")?;
239                }
240                return Self::new(config, network);
241            }
242            info!("PKI directory already exists, skipping generation");
243            Ok(pki)
244        } else {
245            info!("Generating certificates");
246            create_dir_all(dir).context("Unable to create PKI directory")?;
247            let ca_config = Self::write_ca_config(dir)?;
248            let ca = Self::setup_ca(dir)?;
249
250            let mut hostnames = vec![
251                network.api()?.to_string(),
252                Ipv4Addr::LOCALHOST.to_string(),
253                network.hostname().into(),
254                "kubernetes".into(),
255                "kubernetes.default".into(),
256                "kubernetes.default.svc".into(),
257                "kubernetes.default.svc.cluster".into(),
258                "kubernetes.svc.cluster.local".into(),
259            ];
260            hostnames.extend(nodes.clone());
261
262            let pki_config = &PkiConfig {
263                dir,
264                ca: &ca,
265                ca_config,
266                hostnames: &hostnames.join(","),
267            };
268
269            // Generate all leaf certificates in parallel since they
270            // only depend on the CA, not on each other.
271            let kubelet_nodes: Vec<&str> = if config.multi_node() {
272                nodes.iter().map(|n| n.as_str()).collect()
273            } else {
274                vec![network.hostname()]
275            };
276
277            type CertFn = fn(&PkiConfig) -> Result<Identity>;
278            let cert_fns: [CertFn; 7] = [
279                Self::setup_admin,
280                Self::setup_apiserver,
281                Self::setup_controller_manager,
282                Self::setup_etcd,
283                Self::setup_proxy,
284                Self::setup_scheduler,
285                Self::setup_service_account,
286            ];
287
288            let (certs, kubelets) = rayon::join(
289                || {
290                    cert_fns
291                        .into_par_iter()
292                        .map(|f| f(pki_config))
293                        .collect::<Vec<_>>()
294                },
295                || {
296                    kubelet_nodes
297                        .par_iter()
298                        .map(|n| Self::setup_kubelet(pki_config, n))
299                        .collect::<Result<Vec<_>>>()
300                },
301            );
302
303            let mut it = certs.into_iter();
304            let admin = it.next().context("missing admin cert result")??;
305            let apiserver = it.next().context("missing apiserver cert result")??;
306            let controller_manager = it
307                .next()
308                .context("missing controller-manager cert result")??;
309            let etcd = it.next().context("missing etcd cert result")??;
310            let proxy = it.next().context("missing proxy cert result")??;
311            let scheduler = it.next().context("missing scheduler cert result")??;
312            let service_account = it.next().context("missing service-account cert result")??;
313            debug_assert!(it.next().is_none());
314
315            Ok(Pki {
316                admin,
317                apiserver,
318                controller_manager,
319                etcd,
320                proxy,
321                scheduler,
322                service_account,
323                kubelets: kubelets?,
324                ca,
325            })
326        }
327    }

329    fn setup_ca(dir: &Path) -> Result<Identity> {
330        debug!("Creating CA certificates");
331        const CN: &str = "kubernetes";
332        let csr = dir.join("ca-csr.json");
333        Self::write_csr(CN, CN, &csr)?;
334
335        let cfssl = Command::new("cfssl")
336            .arg("gencert")
337            .arg("-initca")
338            .arg(csr)
339            .stdout(Stdio::piped())
340            .stderr(Stdio::piped())
341            .spawn()
342            .context("Unable to spawn cfssl for CA generation")?;
343
344        Self::pipe_cfssl_to_cfssljson(cfssl, &dir.join(CA_NAME), CA_NAME)?;
345        debug!("CA certificates created");
346        Ok(Identity::new(dir, CA_NAME, CA_NAME))
347    }

349    fn setup_kubelet(pki_config: &PkiConfig, node: &str) -> Result<Identity> {
350        let user = Self::node_user(node);
351        let csr_file = pki_config.dir().join(format!("{}-csr.json", node));
352        Self::write_csr(&user, "system:nodes", &csr_file)?;
353        Self::generate(pki_config, node, &csr_file, &user)
354    }

408    fn generate(pki_config: &PkiConfig, name: &str, csr: &Path, user: &str) -> Result<Identity> {
409        debug!("Creating certificate for {}", name);
410
411        let cfssl = Command::new("cfssl")
412            .arg("gencert")
413            .arg(format!("-ca={}", pki_config.ca().cert().display()))
414            .arg(format!("-ca-key={}", pki_config.ca().key().display()))
415            .arg(format!("-config={}", pki_config.ca_config().display()))
416            .arg("-profile=kubernetes")
417            .arg(format!("-hostname={}", pki_config.hostnames()))
418            .arg(csr)
419            .stdout(Stdio::piped())
420            .stderr(Stdio::piped())
421            .spawn()
422            .with_context(|| format!("Unable to spawn cfssl for '{}'", name))?;
423
424        Self::pipe_cfssl_to_cfssljson(cfssl, &pki_config.dir().join(name), name)?;
425        debug!("Certificate created for {}", name);
426
427        Ok(Identity::new(pki_config.dir(), name, user))
428    }

430    /// Pipe cfssl output to cfssljson for certificate generation.
431    fn pipe_cfssl_to_cfssljson(
432        mut cfssl: std::process::Child,
433        output_prefix: &Path,
434        name: &str,
435    ) -> Result<()> {
436        let pipe = cfssl.stdout.take().context("Unable to get cfssl stdout")?;
437        let output = Command::new("cfssljson")
438            .arg("-bare")
439            .arg(output_prefix)
440            .stdin(pipe)
441            .output()
442            .with_context(|| format!("Unable to run cfssljson for '{}'", name))?;
443
444        let cfssl_output = cfssl
445            .wait_with_output()
446            .with_context(|| format!("Unable to wait for cfssl output for '{}'", name))?;
447        if !output.status.success() {
448            debug!(
449                "cfssl stderr: {}",
450                String::from_utf8_lossy(&cfssl_output.stderr)
451            );
452            debug!("cfssljson output: {:?}", output);
453            bail!("Unable to generate certificate for {}", name);
454        }
455
456        let key_file = output_prefix.with_file_name(format!("{}-key.pem", name));
457        if key_file.exists() {
458            fs::set_permissions(&key_file, fs::Permissions::from_mode(0o600)).with_context(
459                || format!("Unable to restrict permissions on '{}'", key_file.display()),
460            )?;
461        }
462        Ok(())
463    }

465    fn write_csr(cn: &str, o: &str, dest: &Path) -> Result<()> {
466        let csr = json!({
467            "CN": cn,
468            "key": {
469                "algo": "ecdsa",
470                "size": ECDSA_KEY_SIZE
471            },
472            "names": [{
473                "O": o,
474                "OU": "kubernetes",
475            }]
476        });
477        fs::write(dest, to_string_pretty(&csr)?)
478            .with_context(|| format!("Unable to write CSR to '{}'", dest.display()))?;
479        Ok(())
480    }
481
482    fn write_ca_config(dir: &Path) -> Result<PathBuf> {
483        let cfg = json!({
484            "signing": {
485                "default": {
486                    "expiry": CERT_EXPIRY
487                },
488                "profiles": {
489                    "kubernetes": {
490                        "usages": [
491                            "signing",
492                            "server auth",
493                            "client auth"
494                        ],
495                        "expiry": CERT_EXPIRY
496                    }
497                }
498            }
499        });
500        let dest = dir.join("ca-config.json");
501        fs::write(&dest, to_string_pretty(&cfg)?).context("Unable to write CA config")?;
502        Ok(dest)
503    }
