1//! System preparation and host utilities.
2//!
3//! Handles kernel module loading, sysctl configuration, hosts file
4//! management, executable lookup in `$PATH`, and container detection.

6use crate::{Config, node::Node};

9use std::{

18/// Manages host-level system preparation and cleanup.
19pub struct System {
20    hosts: Option<String>,
21}
22
23impl System {
24    /// Create a new system
25    pub fn setup(config: &Config) -> Result<Self> {
26        if config.is_rootless() {
27            let in_userns = Self::in_user_namespace();
28            if !in_userns {
29                warn!(
30                    "Rootless mode but not inside a user namespace, \
31                     skipping directory setup"
32                );
33            } else {
34                // Inside rootlesskit, pre-existing host directories under the
35                // copy-up overlay retain host root ownership (mapped to nobody
36                // in the user namespace) and are not writable. Recreate paths
37                // that kubernix components write to under rootlesskit copy-ups.
38                // When adding a new component that writes to a host path under
39                // /var or /run, add the path here too.
40                for dir in &[
41                    "/var/lib/kubelet",
42                    "/var/lib/crio",
43                    "/var/lib/containers",
44                    "/var/cache/containers",
45                    "/var/log/pods",
46                    "/var/log/containers",
47                    "/var/log/crio",
48                    "/run/lock",
49                    "/run/containers",
50                ] {
51                    let path = PathBuf::from(dir);
52                    if path.exists()
53                        && let Err(e) = fs::remove_dir_all(&path)
54                    {
55                        warn!("Unable to remove '{}': {}", dir, e);
56                    }
57                    fs::create_dir_all(&path).with_context(|| {
58                        format!("Unable to create rootless directory '{}'", dir)
59                    })?;
60                }
61                fs::create_dir_all("/var/lib/kubelet/device-plugins")
62                    .context("Unable to create /var/lib/kubelet/device-plugins")?;
63
64                // Override containers-storage.conf to prevent
65                // overlay-specific options from leaking into vfs mode.
66                // Written to the run directory (not /etc/containers/) to
67                // avoid permission issues inside rootlesskit copy-ups.
68                let storage_conf = config.root().join("storage.conf");
69                fs::write(&storage_conf, "[storage]\ndriver = \"vfs\"\n")
70                    .context("Unable to write rootless storage.conf")?;
71                // SAFETY: called before any threads are spawned.
72                unsafe { std::env::set_var("CONTAINERS_STORAGE_CONF", &storage_conf) };
73
74                // On NixOS, /etc/hosts is a symlink into the read-only nix
75                // store. Replace it with a regular file so multi-node can
76                // write host entries.
77                let hosts_path = Self::hosts();
78                if hosts_path.is_symlink() {
79                    let content = read_to_string(&hosts_path).unwrap_or_default();
80                    fs::remove_file(&hosts_path).context("Unable to remove hosts symlink")?;
81                    fs::write(&hosts_path, content).context("Unable to write hosts file")?;
82                }
83            }
84        }
85
86        if !config.is_rootless() && !Self::in_container()? {
87            for module in &["overlay", "br_netfilter", "ip_conntrack"] {
88                Self::modprobe(module)?;
89            }
90            for sysctl in &[
91                "net.bridge.bridge-nf-call-ip6tables",
92                "net.bridge.bridge-nf-call-iptables",
93                "net.ipv4.conf.all.route_localnet",
94                "net.ipv4.ip_forward",
95            ] {
96                Self::sysctl_enable(sysctl)?;
97            }
98        } else {
99            info!("Skipping modprobe and sysctl (containerized or rootless)");
100        }
101
102        let hosts = if config.multi_node() {
103            // Try to write the hostnames, which does not work on every system
104            let hosts_file = Self::hosts();
105            let hosts = read_to_string(&hosts_file)?;
106            let local_hosts = (0..config.nodes())
107                .map(|x| format!("{} {}", Ipv4Addr::LOCALHOST, Node::raw(x)))
108                .collect::<Vec<_>>();
109
110            let mut new_hosts = hosts
111                .lines()
112                .filter(|x| !local_hosts.iter().any(|y| x == y))
113                .map(|x| x.into())
114                .collect::<Vec<_>>();
115            new_hosts.extend(local_hosts);
116
117            match fs::write(&hosts_file, new_hosts.join("\n")) {
118                Err(e) => {
119                    debug!(
120                        "Unable to write hosts file '{}'. The nodes may be not reachable: {}",
121                        hosts_file.display(),
122                        e
123                    );
124                    None
125                }
126                _ => Some(hosts),
127            }
128        } else {
129            None
130        };
131
132        Ok(Self { hosts })
133    }

135    /// Returns true if running inside a user namespace (uid 0 maps to
136    /// a non-zero host UID). Used to guard destructive host path cleanup
137    /// and to validate the KUBERNIX_ROOTLESS env var at startup.
138    ///
139    /// This function is intentionally side-effect-free (no logging)
140    /// because it may be called before the logger is initialized.
141    pub(crate) fn in_user_namespace() -> bool {
142        read_to_string("/proc/self/uid_map")
143            .map(|s| {
144                s.lines().any(|line| {
145                    let mut fields = line.split_whitespace();
146                    fields.next() == Some("0")
147                        && fields.next().is_some_and(|host_uid| host_uid != "0")
148                })
149            })
150            .unwrap_or(false)
151    }
152
153    /// Returns true if the process is running inside a container
154    pub fn in_container() -> Result<bool> {
155        Ok(
156            read_to_string(PathBuf::from("/").join("proc").join("1").join("cgroup"))
157                .context("Unable to retrieve systems container status")?
158                .lines()
159                .any(|x| x.contains("libpod") || x.contains("podman") || x.contains("docker")),
160        )
161    }
162
163    /// Restore the initial system state
164    pub fn cleanup(&self) {
165        if let Some(hosts) = &self.hosts
166            && let Err(e) = fs::write(Self::hosts(), hosts)
167        {
168            warn!(
169                "Unable to restore hosts file, may need manual cleanup: {}",
170                e
171            );
172        }
173    }
174
175    /// Find an executable inside the current $PATH environment
176    pub fn find_executable<P>(name: P) -> Result<PathBuf>
177    where
178        P: AsRef<Path> + Display,
179    {
180        var_os("PATH")
181            .and_then(|paths| {
182                split_paths(&paths)
183                    .filter_map(|dir| {
184                        let full_path = dir.join(&name);
185                        if full_path.is_file() {
186                            Some(full_path)
187                        } else {
188                            None
189                        }
190                    })
191                    .next()
192            })
193            .with_context(|| format!("Unable to find executable '{}' in $PATH", name))
194    }
195
196    /// Return the full path to the default system shell
197    pub fn shell() -> Result<String> {
198        let shell = var("SHELL").unwrap_or_else(|_| "sh".into());
199        Ok(format!(
200            "{}",
201            Self::find_executable(&shell)
202                .with_context(|| format!("Unable to find system shell '{}'", shell))?
203                .display()
204        ))
205    }

207    /// Check if a kernel module is already loaded
208    fn module_loaded(module: &str) -> bool {
209        read_to_string(PathBuf::from("/proc/modules"))
210            .map(|content| {
211                content
212                    .lines()
213                    .any(|l| l.split_whitespace().next() == Some(module))
214            })
215            .unwrap_or(false)
216    }
217
218    /// Load a single kernel module via 'modprobe', skipping if already loaded
219    fn modprobe(module: &str) -> Result<()> {
220        if Self::module_loaded(module) {
221            debug!("Kernel module '{}' already loaded, skipping", module);
222            return Ok(());
223        }
224        debug!("Loading kernel module '{}'", module);
225        let output = Command::new("modprobe").arg(module).output()?;
226        if !output.status.success() {
227            bail!(
228                "Unable to load '{}' kernel module: {}",
229                module,
230                String::from_utf8(output.stderr)?,
231            );
232        }
233        Ok(())
234    }
235
236    /// Check if a sysctl is already enabled (value is "1")
237    fn sysctl_enabled(key: &str) -> bool {
238        let path = format!("/proc/sys/{}", key.replace('.', "/"));
239        read_to_string(&path)
240            .map(|v| v.trim() == "1")
241            .unwrap_or(false)
242    }
243
244    /// Enable a single sysctl by setting it to '1', skipping if already set
245    fn sysctl_enable(key: &str) -> Result<()> {
246        if Self::sysctl_enabled(key) {
247            debug!("Sysctl '{}' already enabled, skipping", key);
248            return Ok(());
249        }
250        debug!("Enabling sysctl '{}'", key);
251        let enable_arg = format!("{}=1", key);
252        let output = Command::new("sysctl").arg("-w").arg(&enable_arg).output()?;
253        let stderr = String::from_utf8(output.stderr)?;
254        if !stderr.is_empty() {
255            bail!("Unable to set sysctl '{}': {}", enable_arg, stderr);
256        }
257        Ok(())
258    }
259
260    fn hosts() -> PathBuf {
261        PathBuf::from("/").join("etc").join("hosts")
262    }
263}
