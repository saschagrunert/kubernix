3//! Each cluster component (etcd, API server, etc.) implements the [`Component`]

27/// Boot phases control the startup order of cluster components.
28/// Components in the same phase start concurrently; phases run sequentially.
29///
30/// Phase ordering is optimized so that independent components overlap:
31/// the CRI runtime (CRI-O or containerd) and kube-proxy start alongside the
32/// controllers, since they only need the API server, not the controllers.
33#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
34pub enum Phase {
35    /// Infrastructure services (etcd)
36    Infrastructure,
37    /// Control plane services (API server)
38    ControlPlane,
39    /// Controllers, node runtimes, and proxy all start together since they
40    /// only depend on the API server, not on each other.
41    /// Contains: scheduler, controller-manager, CRI runtime (per node), proxy.
42    Controller,
43    /// Node agents that require a running CRI runtime (kubelet).
44    NodeAgent,
45}

47/// Shared cluster state passed to components during startup.
48pub struct ClusterContext<'a> {
49    /// Cluster configuration
50    pub config: &'a Config,
51    /// Network topology
52    pub network: &'a Network,
53    /// Public key infrastructure (certificates)
54    pub pki: &'a Pki,
55    /// Kubeconfig files for each component
56    pub kubeconfig: &'a KubeConfig,
57    /// Encryption configuration for the API server
58    pub encryptionconfig: &'a EncryptionConfig,
59    /// Kubectl handle for RBAC and workload operations
60    pub kubectl: &'a Kubectl,
61}

63/// Trait implemented by each cluster component.
64///
65/// This provides a uniform interface for the orchestrator to start the
66/// registered components and to order them by their boot phase.
67pub trait Component: Send + Sync {
68    /// Human-readable name for logging
69    fn name(&self) -> &str;
70
71    /// The boot phase this component belongs to
72    fn phase(&self) -> Phase;
73
74    /// Start the component. Returns a handle that can stop it.
75    fn start(&self, ctx: &ClusterContext<'_>) -> ProcessState;
76}

78/// A registry that collects components and groups them by phase.
79#[must_use]
80pub struct ComponentRegistry {
81    components: Vec<Box<dyn Component>>,
82}
83
84impl ComponentRegistry {
85    /// Create an empty registry
86    pub fn new() -> Self {
87        Self {
88            components: Vec::new(),
89        }
90    }
91
92    /// Register a component
93    pub fn register(&mut self, component: Box<dyn Component>) {
94        self.components.push(component);
95    }
96
97    /// Return components grouped and sorted by phase
98    pub fn by_phase(&self) -> Vec<(Phase, Vec<&dyn Component>)> {
99        let mut phases: Vec<Phase> = self.components.iter().map(|c| c.phase()).collect();
100        phases.sort();
101        phases.dedup();
102
103        phases
104            .into_iter()
105            .map(|phase| {
106                let group: Vec<&dyn Component> = self
107                    .components
108                    .iter()
109                    .filter(|c| c.phase() == phase)
110                    .map(|c| c.as_ref())
111                    .collect();
112                (phase, group)
113            })
114            .collect()
115    }
116
117    /// Execute all registered components phase by phase.
118    ///
119    /// Within each phase, components start in parallel, each on its own
120    /// thread since a start blocks until the component is ready. Once a
121    /// component fails, the others of the phase stop waiting for their
122    /// readiness.
123    /// If any component in a phase fails or a termination signal has been
124    /// received, subsequent phases are skipped (later phases depend on
125    /// earlier ones).
126    ///
127    /// The Controller phase runs the controllers, the CRI runtimes and the
128    /// proxy in parallel, since they all only depend on the API server.
129    ///
130    /// Returns stoppable handles grouped by phase in reverse phase order
131    /// (for proper shutdown: node agents before runtimes, runtimes before
132    /// control plane, etc.) and the startup failures.
133    pub fn run(&self, ctx: &ClusterContext<'_>) -> (Vec<Stoppables>, Vec<String>) {
134        info!("Starting processes");
135        let mut groups: Vec<Stoppables> = Vec::new();
136        let mut failures: Vec<String> = Vec::new();
137
138        for (phase, components) in &self.by_phase() {
139            if let Err(e) = signals::check() {
140                failures.push(e.to_string());
141                break;
142            }
143            debug!(
144                "Starting {:?} phase ({} components)",
145                phase,
146                components.len()
147            );
148            let (group, errors) = Self::start_phase(components, |c| c.start(ctx));
149            groups.push(group);
150            failures.extend(errors);
151
152            if !failures.is_empty() {
153                error!(
154                    "Phase {:?} failed ({} errors: {}), skipping remaining phases",
155                    phase,
156                    failures.len(),
157                    failures.join(", ")
158                );
159                break;
160            }
161        }
162
163        groups.reverse();
164        (groups, failures)
165    }
166
167    /// Start the components of a phase in parallel via `start`. Returns the
168    /// started ones and the failures, without the cancellations which a
169    /// failure caused.
170    fn start_phase<F>(components: &[&dyn Component], start: F) -> (Stoppables, Vec<String>)
171    where
172        F: Fn(&dyn Component) -> ProcessState + Sync,
173    {
174        let cancel = Arc::new(AtomicBool::new(false));
175        let results: Vec<ProcessState> = thread::scope(|s| {
176            let handles: Vec<_> = components
177                .iter()
178                .map(|c| {
179                    let cancel = Arc::clone(&cancel);
180                    let start = &start;
181                    s.spawn(move || {
182                        debug!("Starting {}", c.name());
183                        process::set_cancel(Arc::clone(&cancel));
184                        let result = start(*c);
185                        if result.is_err() {
186                            cancel.store(true, Ordering::SeqCst);
187                        }
188                        result
189                    })
190                })
191                .collect();
192            handles
193                .into_iter()
194                .zip(components)
195                .map(|(h, c)| {
196                    h.join()
197                        .unwrap_or_else(|_| Err(anyhow!("Starting {} panicked", c.name())))
198                })
199                .collect()
200        });
201
202        let mut group = Stoppables::new();
203        let mut failures = Vec::new();
204        for result in results {
205            match result {
206                Ok(p) => group.push(p),
207                Err(e) if e.downcast_ref::<Canceled>().is_some() => debug!("{:#}", e),
208                Err(e) => failures.push(format!("{:#}", e)),
209            }
210        }
211        (group, failures)
212    }
213}

241    #[test]
242    fn registry_groups_by_phase() {
243        let mut reg = ComponentRegistry::new();
244        reg.register(Box::new(FakeComponent {
245            name: "etcd".into(),
246            phase: Phase::Infrastructure,
247        }));
248        reg.register(Box::new(FakeComponent {
249            name: "apiserver".into(),
250            phase: Phase::ControlPlane,
251        }));
252        reg.register(Box::new(FakeComponent {
253            name: "scheduler".into(),
254            phase: Phase::Controller,
255        }));
256        reg.register(Box::new(FakeComponent {
257            name: "controller-manager".into(),
258            phase: Phase::Controller,
259        }));
260
261        let phases = reg.by_phase();
262        assert_eq!(phases.len(), 3);
263        assert_eq!(phases[0].0, Phase::Infrastructure);
264        assert_eq!(phases[0].1.len(), 1);
265        assert_eq!(phases[1].0, Phase::ControlPlane);
266        assert_eq!(phases[1].1.len(), 1);
267        assert_eq!(phases[2].0, Phase::Controller);
268        assert_eq!(phases[2].1.len(), 2);
269    }

283    /// Start the fake components: "fail" fails, "wait" waits until it gets
284    /// canceled and the others succeed.
285    fn start_fake(c: &dyn Component) -> ProcessState {
286        match c.name() {
287            "fail" => bail!("failed"),
288            "wait" => {
289                let deadline = Instant::now() + Duration::from_secs(10);
290                while !canceled() {
291                    if Instant::now() >= deadline {
292                        bail!("not canceled")
293                    }
294                    std::thread::sleep(Duration::from_millis(10));
295                }
296                Err(Canceled).context("stopped waiting")
297            }
298            _ => Ok(Box::new(FakeStarted)),
299        }
300    }

309    #[test]
310    fn start_phase_cancels_siblings() {
311        let (a, b, c) = (fake("ok"), fake("wait"), fake("fail"));
312        let (group, failures) = ComponentRegistry::start_phase(&[&a, &b, &c], start_fake);
313        assert_eq!(group.len(), 1);
314        assert_eq!(failures, ["failed"]);
315
316        let (group, failures) = ComponentRegistry::start_phase(&[&a, &a], start_fake);
