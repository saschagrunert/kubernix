1//! containerd container runtime component.
2//!
3//! Manages containerd instances that provide the container runtime
4//! interface for kubelet. Each node gets its own instance with a separate
5//! CNI network configuration. In multi-node root mode, each instance runs
6//! inside its own node container.

24/// Component wrapper for registry-based startup (per-node).
25pub struct ContainerdComponent {
26    node: u8,
27    name: String,
28}

30impl ContainerdComponent {
31    /// Create a new containerd component for the given node index.
32    pub fn new(node: u8) -> Self {
33        Self {
34            node,
35            name: format!("Containerd (node {})", node),
36        }
37    }
38}

54/// Manages a containerd process and its associated socket for a single node.
55pub struct Containerd {
56    process: Process,
57    socket: CriSocket,
58    container: Option<NodeContainer>,
59    node_name: String,
60}
61
62const CONTAINERD: &str = "containerd";
63
64/// The registry host configuration directory (`hosts.toml` and
65/// certificates) of the host. Without `config_path`, containerd 2.4 uses
66/// `/etc/containerd/certs.d:/etc/docker/certs.d`, which its transfer
67/// service treats as a single directory, so no registry configuration of
68/// the host would apply. A single directory works with both the transfer
69/// service and the local image pull.
70const CERTS_DIR: &str = "/etc/containerd/certs.d";
71
72impl Containerd {
73    /// Start a containerd instance for the given node index.
74    pub fn start(config: &Config, node: u8, network: &Network) -> ProcessState {
75        let node_name = Node::name(config, network, node);
76
77        let paths = RuntimePaths::resolve(config)?;
78
79        let dir = Self::path(config, network, node);
80        let config_file = dir.join("config.toml");
81        let cni_conf_dir = dir.join("cni");
82        let socket = Self::socket(config, network, node)?;
83        let nri = NriPaths::new(&dir)?;
84
85        // Always render the configuration, since it references the current
86        // Nix store paths.
87        for d in [cni_conf_dir.as_path(), &dir.join("cdi")]
88            .into_iter()
89            .chain(nri.dirs().into_iter().filter(|_| config.nri()))
90        {
91            create_dir_all(d).with_context(|| {
92                format!("Unable to create containerd directory {}", d.display())
93            })?;
94        }
95
96        if config.is_rootless() {
97            cri::write_rootless_wrappers(&dir, &paths)?;
98        }
99
100        write_if_changed(
101            &config_file,
102            &Self::render_config(config, &dir, &socket, &nri, &cni_conf_dir, &paths),
103        )
104        .context("Unable to write containerd config")?;
105
106        // containerd leaves the host ports of the pods to the portmap plugin
107        cri::write_cni_config(&cni_conf_dir, &node_name, node, network, true)?;
108        cri::reset_ipam_data(&cni_conf_dir)?;
109
110        // containerd-shim-runc-v2 hardcodes its socket path to /run/containerd/s/.
111        // System::setup recreates /run/containerd with the correct ownership
112        // in rootless mode, before the nodes start in parallel.
113        if config.is_rootless() {
114            create_dir_all("/run/containerd/s").context("Unable to create /run/containerd/s")?;
115        }
116
117        let config_arg = format!("--config={}", config_file.display());
118        let args: &[&str] = &[&config_arg];
119
120        let mut process = if config.multi_node() && !config.is_rootless() {
121            // Run inside a container
122            let identifier = format!("Containerd {}", node_name);
123            Container::start(config, &dir, &identifier, CONTAINERD, &node_name, args, &[])?
124        } else {
125            Process::start(&dir, "Containerd", CONTAINERD, args)?
126        };
127        let container = Container::node(config, &node_name);
128        process.wait_ready("containerd successfully booted")?;
129
130        Ok(Box::new(Self {
131            process,
132            socket,
133            container,
134            node_name,
135        }))
136    }
137
138    /// Render the containerd configuration for the node working directory.
139    ///
140    /// Both OCI runtimes are registered as runtime handlers with their own
141    /// state directory, the configured one is used as default.
142    ///
143    /// NRI is disabled unless `--nri` is set, and its socket and plugin
144    /// directories belong to the node instead of the host, see `NriPaths`. The
145    /// CDI spec directory of the node is empty, so that the CDI specs of the
146    /// host do not apply. The registry host configuration of the host
147    /// (`/etc/containerd/certs.d`) still applies on purpose, so that private
148    /// registries work like on the host. In rootless mode, AppArmor is disabled
149    /// and the OOM scores are restricted to the one of containerd, which cannot
150    /// be lowered in a user namespace.
151    fn render_config(
152        config: &Config,
153        dir: &Path,
154        socket: &CriSocket,
155        nri: &NriPaths,
156        cni_conf_dir: &Path,
157        paths: &RuntimePaths,
158    ) -> String {
159        const RUNTIMES: &str = r#"plugins."io.containerd.cri.v1.runtime".containerd.runtimes"#;
160        let runtimes = paths
161            .handlers(config, dir)
162            .iter()
163            .map(|(runtime, path)| {
164                format!(
165                    "[{RUNTIMES}.{name}]\n\
166                     runtime_type = \"io.containerd.runc.v2\"\n\
167                     \n\
168                     [{RUNTIMES}.{name}.options]\n\
169                     BinaryName = \"{path}\"\n\
170                     Root = \"{root}\"\n\
171                     SystemdCgroup = false\n",
172                    name = runtime,
173                    root = dir.join(runtime.name()).display(),
174                )
175            })
176            .collect::<Vec<_>>()
177            .join("\n");
178        // The native snapshotter of containerd 2.3.3 failed with "no unpack
179        // platforms defined", overlayfs works in all modes.
180        let snapshotter = "overlayfs";
181        format!(
182            include_str!("assets/containerd.toml"),
183            root = dir.join("root").display(),
184            state = dir.join("state").display(),
185            socket = socket,
186            plugin_dir = paths.plugin_dir,
187            cni_conf_dir = cni_conf_dir.display(),
188            cdi_spec_dir = dir.join("cdi").display(),
189            certs_dir = CERTS_DIR,
190            default_runtime = config.oci_runtime(),
191            runtimes = runtimes,
192            snapshotter = snapshotter,
193            rootless = config.is_rootless(),
194            disable_nri = !config.nri(),
195            nri_socket = nri.socket.display(),
196            nri_plugin_dir = nri.plugin_dir.display(),
197            nri_plugin_config_dir = nri.plugin_config_dir.display(),
198        )
199    }
200
201    /// Retrieve the CRI socket for the given node.
202    pub fn socket(config: &Config, network: &Network, node: u8) -> Result<CriSocket> {
203        CriSocket::new(Self::path(config, network, node).join("containerd.sock"))
204    }
205
206    /// Retrieve the working path for the node.
207    fn path(config: &Config, network: &Network, node: u8) -> PathBuf {
208        config
209            .root()
210            .join(CONTAINERD)
211            .join(Node::name(config, network, node))
212    }
213}

215impl Stoppable for Containerd {
216    fn stop(&mut self) -> Result<()> {
217        // Always stop the process, even if the workloads cannot be removed,
218        // for example because containerd crashed.
219        let removed = cri::remove_all_containers("containerd", &self.socket, &self.node_name)
220            .with_context(|| {
221                format!(
222                    "Unable to remove containerd containers on {}",
223                    self.node_name,
224                )
225            });
226        let stopped = self.process.stop();
227        cri::kill_leftover_monitors(&self.socket, &self.node_name);
228        if let Some(container) = &mut self.container {
229            container.remove();
230        }
231        removed.and(stopped)
232    }
233
234    fn died(&mut self) -> bool {
235        self.process.died()
236    }
237}
