1//! containerd container runtime component.
2//!
3//! Manages containerd instances that provide the container runtime
4//! interface for kubelet. In multi-node mode, each containerd instance
5//! runs inside its own container with an isolated CNI network
6//! configuration.

24/// Component wrapper for registry-based startup (per-node).
25pub struct ContainerdComponent {
26    node: u8,
27    name: String,
28}
29
30impl ContainerdComponent {
31    /// Create a new containerd component for the given node index.
32    pub fn new(node: u8) -> Self {
33        Self {
34            node,
35            name: format!("Containerd (node {})", node),
36        }
37    }
38}

40impl Component for ContainerdComponent {
41    fn name(&self) -> &str {
42        &self.name
43    }
44
45    fn phase(&self) -> Phase {
46        Phase::Controller
47    }
48
49    fn start(&self, ctx: &ClusterContext<'_>) -> ProcessState {
50        Containerd::start(ctx.config, self.node, ctx.network)
51    }
52}

54/// Manages a containerd process and its associated socket for a single node.
55pub struct Containerd {
56    process: Process,
57    socket: CriSocket,
58    node_name: String,
59}
60
61const CONTAINERD: &str = "containerd";
62
63impl Containerd {
64    /// Start a containerd instance for the given node index.
65    pub fn start(config: &Config, node: u8, network: &Network) -> ProcessState {
66        let node_name = Node::name(config, network, node);
67
68        let paths = RuntimePaths::resolve(config)?;
69
70        let dir = Self::path(config, network, node);
71        let config_file = dir.join("config.toml");
72        let cni_conf_dir = dir.join("cni");
73        let socket = Self::socket(config, network, node)?;
74
75        if !dir.exists() {
76            create_dir_all(&dir).context("Unable to create containerd directory")?;
77            create_dir_all(&cni_conf_dir).context("Unable to create containerd CNI directory")?;
78
79            // In rootless mode, kubelet sends oomScoreAdj via CRI, containerd
80            // passes it to the OCI spec, and both crun and runc fail writing
81            // /proc/self/oom_score_adj in a user namespace. Wrap each runtime
82            // to patch config.json before exec (strip oomScoreAdj, fix
83            // namespace config).
84            if config.is_rootless() {
85                for (runtime, path) in &paths.runtimes {
86                    let wrapper = Self::rootless_wrapper(&dir, *runtime);
87                    let script =
88                        include_str!("assets/oci-rootless.sh").replace("__RUNTIME_PATH__", path);
89                    fs::write(&wrapper, script)
90                        .with_context(|| format!("Unable to write {} rootless wrapper", runtime))?;
91                    #[cfg(unix)]
92                    {
93                        use std::os::unix::fs::PermissionsExt;
94                        fs::set_permissions(&wrapper, fs::Permissions::from_mode(0o755))
95                            .with_context(|| {
96                                format!("Unable to set {} rootless wrapper permissions", runtime)
97                            })?;
98                    }
99                }
100            }
101
102            fs::write(
103                &config_file,
104                Self::render_config(config, &dir, &socket, &cni_conf_dir, &paths),
105            )
106            .context("Unable to write containerd config")?;
107
108            cri::write_pod_network_config(config, &cni_conf_dir, &node_name, node, network)?;
109        }
110
111        // containerd-shim-runc-v2 hardcodes its socket path to /run/containerd/s/.
112        // rootlesskit's --copy-up=/run creates an overlay that preserves
113        // host-owned subdirectories with wrong ownership (host root maps to
114        // UID 65534). Recreate with correct ownership. Safe: runs inside
115        // rootlesskit's private mount namespace before containerd starts.
116        if config.is_rootless() && create_dir_all("/run/containerd/s").is_err() {
117            fs::remove_dir_all("/run/containerd")
118                .context("Failed to remove host-owned /run/containerd")?;
119            create_dir_all("/run/containerd/s").context("Unable to create /run/containerd/s")?;
120        }
121
122        let config_arg = format!("--config={}", config_file.display());
123        let args: &[&str] = &[&config_arg];
124
125        let mut process = if config.multi_node() && !config.is_rootless() {
126            // containerd has no --cni-plugin-dir CLI flag (unlike CRI-O), so
127            // patch bin_dirs in the config before starting the daemon. The
128            // container entrypoint runs all args through `bash -c "$*"`, so
129            // shell constructs (&&, $(...)) are interpreted.
130            let identifier = format!("Containerd {}", node_name);
131            let patch_bin_dirs = format!(
132                r#"sed -i "s|bin_dirs = .*|bin_dirs = [\"$(dirname $(which loopback))\"]|" {} &&"#,
133                config_file.display(),
134            );
135            let container_args: &[&str] = &[CONTAINERD, &config_arg];
136            Container::start(
137                config,
138                &dir,
139                &identifier,
140                &patch_bin_dirs,
141                &node_name,
142                container_args,
143            )?
144        } else {
145            Process::start(&dir, "Containerd", CONTAINERD, args)?
146        };
147        process.wait_ready("containerd successfully booted")?;
148
149        Ok(Box::new(Self {
150            process,
151            socket,
152            node_name,
153        }))
154    }

156    /// Path of the rootless wrapper script for the given OCI runtime.
157    fn rootless_wrapper(dir: &Path, runtime: OciRuntime) -> PathBuf {
158        dir.join(format!("{}-rootless", runtime))
159    }
160
161    /// Render the containerd configuration for the node working directory.
162    ///
163    /// Both OCI runtimes are registered as runtime handlers with their own
164    /// state directory, the configured one is used as default.
165    fn render_config(
166        config: &Config,
167        dir: &Path,
168        socket: &CriSocket,
169        cni_conf_dir: &Path,
170        paths: &RuntimePaths,
171    ) -> String {
172        const RUNTIMES: &str = r#"plugins."io.containerd.cri.v1.runtime".containerd.runtimes"#;
173        let runtimes = paths
174            .runtimes
175            .iter()
176            .map(|(runtime, path)| {
177                let path = if config.is_rootless() {
178                    Self::rootless_wrapper(dir, *runtime).display().to_string()
179                } else {
180                    path.clone()
181                };
182                format!(
183                    "[{RUNTIMES}.{name}]\n\
184                     runtime_type = \"io.containerd.runc.v2\"\n\
185                     \n\
186                     [{RUNTIMES}.{name}.options]\n\
187                     BinaryName = \"{path}\"\n\
188                     Root = \"{root}\"\n\
189                     SystemdCgroup = false\n",
190                    name = runtime,
191                    root = dir.join(runtime.name()).display(),
192                )
193            })
194            .collect::<Vec<_>>()
195            .join("\n");
196        // containerd 2.3.3 native snapshotter fails with "no unpack
197        // platforms defined"; overlayfs works in all modes.
198        let snapshotter = "overlayfs";
199        format!(
200            include_str!("assets/containerd.toml"),
201            root = dir.join("root").display(),
202            state = dir.join("state").display(),
203            socket = socket,
204            plugin_dir = paths.plugin_dir,
205            cni_conf_dir = cni_conf_dir.display(),
206            default_runtime = config.oci_runtime(),
207            runtimes = runtimes,
208            snapshotter = snapshotter,
209            disable_apparmor = config.is_rootless(),
210            disable_nri = config.is_rootless(),
211        )
212    }
213
214    /// Retrieve the CRI socket for the given node.
215    pub fn socket(config: &Config, network: &Network, node: u8) -> Result<CriSocket> {
216        CriSocket::new(Self::path(config, network, node).join("containerd.sock"))
217    }
218
219    /// Retrieve the working path for the node.
220    fn path(config: &Config, network: &Network, node: u8) -> PathBuf {
221        config
222            .root()
223            .join(CONTAINERD)
224            .join(Node::name(config, network, node))
225    }
226}

228impl Stoppable for Containerd {
229    fn stop(&mut self) -> Result<()> {
230        cri::remove_all_containers("containerd", &self.socket, &self.node_name).with_context(
231            || {
232                format!(
233                    "Unable to remove containerd containers on {}",
234                    self.node_name,
235                )
236            },
237        )?;
238
239        self.process.stop()
240    }
241}

261    #[test]
262    fn render_config_skips_missing_runtime() -> Result<()> {
263        let dir = PathBuf::from("/kubernix/containerd/node");
264        let socket = CriSocket::new(dir.join("containerd.sock"))?;
265        let paths = RuntimePaths {
266            runtimes: vec![(OciRuntime::Runc, "/nix/bin/runc".into())],
267            plugin_dir: "/nix/cni".into(),
268        };
269        let config = Config::parse_from(["kubernix", "--oci-runtime=runc"]);
270        let content = Containerd::render_config(&config, &dir, &socket, &dir.join("cni"), &paths);
271        let v: toml::Value = toml::from_str(&content)?;
272        let containerd = cri_runtime(&v);
273        assert_eq!(containerd["default_runtime_name"].as_str(), Some("runc"));
274        assert!(containerd["runtimes"].get("runc").is_some());
275        assert!(containerd["runtimes"].get("crun").is_none());

309            assert_eq!(
310                runtime["options"]["Root"].as_str(),
311                Some(format!("/kubernix/containerd/node/{}", name).as_str())
