1//! containerd container runtime component.
2//!
3//! Manages containerd instances that provide the container runtime
4//! interface for kubelet. In multi-node mode, each containerd instance
5//! runs inside its own container with an isolated CNI network
6//! configuration.
7
8use crate::{
9    Config,
10    component::{ClusterContext, Component, Phase},
11    container::Container,
12    cri::{self, CriSocket, RuntimePaths},
13    network::Network,
14    node::Node,
15    process::{Process, ProcessState, Stoppable},
16};
17use anyhow::{Context, Result};
18use std::{
19    fs::{self, create_dir_all},
20    path::PathBuf,
21};

23/// Component wrapper for registry-based startup (per-node).
24pub struct ContainerdComponent {
25    node: u8,
26    name: String,
27}
28
29impl ContainerdComponent {
30    /// Create a new containerd component for the given node index.
31    pub fn new(node: u8) -> Self {
32        Self {
33            node,
34            name: format!("Containerd (node {})", node),
35        }
36    }
37}
38
39impl Component for ContainerdComponent {
40    fn name(&self) -> &str {
41        &self.name
42    }
43
44    fn phase(&self) -> Phase {
45        Phase::Controller
46    }
47
48    fn start(&self, ctx: &ClusterContext<'_>) -> ProcessState {
49        Containerd::start(ctx.config, self.node, ctx.network)
50    }
51}

53/// Manages a containerd process and its associated socket for a single node.
54pub struct Containerd {
55    process: Process,
56    socket: CriSocket,
57    node_name: String,
58}
59
60const CONTAINERD: &str = "containerd";
61
62impl Containerd {
63    /// Start a containerd instance for the given node index.
64    pub fn start(config: &Config, node: u8, network: &Network) -> ProcessState {
65        let node_name = Node::name(config, network, node);
66
67        let paths = RuntimePaths::resolve(config)?;
68        let crun_path = paths.crun;
69        let plugin_dir = paths.plugin_dir;
70
71        let dir = Self::path(config, network, node);
72        let config_file = dir.join("config.toml");
73        let cni_conf_dir = dir.join("cni");
74        let socket = Self::socket(config, network, node)?;
75
76        // In rootless mode, kubelet sends oomScoreAdj via CRI, containerd
77        // passes it to the OCI spec, and crun fails writing
78        // /proc/self/oom_score_adj in a user namespace. Wrap crun to patch
79        // config.json before exec (strip oomScoreAdj, fix namespace config).
80        let runtime_path = if config.is_rootless() {
81            dir.join("crun-rootless").display().to_string()
82        } else {
83            crun_path.clone()
84        };
85
86        if !dir.exists() {
87            create_dir_all(&dir)?;
88            create_dir_all(&cni_conf_dir)?;
89
90            if config.is_rootless() {
91                let wrapper = dir.join("crun-rootless");
92                let script =
93                    include_str!("assets/crun-rootless.sh").replace("__CRUN_PATH__", &crun_path);
94                fs::write(&wrapper, script)?;
95                #[cfg(unix)]
96                {
97                    use std::os::unix::fs::PermissionsExt;
98                    fs::set_permissions(&wrapper, fs::Permissions::from_mode(0o755))?;
99                }
100            }
101
102            // containerd 2.3.3 native snapshotter fails with "no unpack
103            // platforms defined"; overlayfs works in all modes.
104            let snapshotter = "overlayfs";
105            fs::write(
106                &config_file,
107                format!(
108                    include_str!("assets/containerd.toml"),
109                    root = dir.join("root").display(),
110                    state = dir.join("state").display(),
111                    socket = socket,
112                    plugin_dir = plugin_dir,
113                    cni_conf_dir = cni_conf_dir.display(),
114                    runtime_path = runtime_path,
115                    snapshotter = snapshotter,
116                    disable_apparmor = config.is_rootless(),
117                    disable_nri = config.is_rootless(),
118                ),
119            )?;
120
121            cri::write_pod_network_config(config, &cni_conf_dir, &node_name, node, network)?;
122        }
123
124        // containerd-shim-runc-v2 hardcodes its socket path to /run/containerd/s/.
125        // rootlesskit's --copy-up=/run creates an overlay that preserves
126        // host-owned subdirectories with wrong ownership (host root maps to
127        // UID 65534). Recreate with correct ownership. Safe: runs inside
128        // rootlesskit's private mount namespace before containerd starts.
129        if config.is_rootless() && create_dir_all("/run/containerd/s").is_err() {
130            fs::remove_dir_all("/run/containerd")
131                .context("Failed to remove host-owned /run/containerd")?;
132            create_dir_all("/run/containerd/s")?;
133        }
134
135        let config_arg = format!("--config={}", config_file.display());
136        let args: &[&str] = &[&config_arg];
137
138        let mut process = if config.multi_node() && !config.is_rootless() {
139            // containerd has no --cni-plugin-dir CLI flag (unlike CRI-O), so
140            // patch bin_dirs in the config before starting the daemon. The
141            // container entrypoint runs all args through `bash -c "$*"`, so
142            // shell constructs (&&, $(...)) are interpreted.
143            let identifier = format!("Containerd {}", node_name);
144            let patch_bin_dirs = format!(
145                r#"sed -i "s|bin_dirs = .*|bin_dirs = [\"$(dirname $(which loopback))\"]|" {} &&"#,
146                config_file.display(),
147            );
148            let container_args: &[&str] = &[CONTAINERD, &config_arg];
149            Container::start(
150                config,
151                &dir,
152                &identifier,
153                &patch_bin_dirs,
154                &node_name,
155                container_args,
156            )?
157        } else {
158            Process::start(&dir, "Containerd", CONTAINERD, args)?
159        };
160        process.wait_ready("containerd successfully booted")?;
161
162        Ok(Box::new(Self {
163            process,
164            socket,
165            node_name,
166        }))
167    }
168
169    /// Retrieve the CRI socket for the given node.
170    pub fn socket(config: &Config, network: &Network, node: u8) -> Result<CriSocket> {
171        CriSocket::new(Self::path(config, network, node).join("containerd.sock"))
172    }
173
174    /// Retrieve the working path for the node.
175    fn path(config: &Config, network: &Network, node: u8) -> PathBuf {
176        config
177            .root()
178            .join(CONTAINERD)
179            .join(Node::name(config, network, node))
180    }
181}
182
183impl Stoppable for Containerd {
184    fn stop(&mut self) -> Result<()> {
185        cri::remove_all_containers("containerd", &self.socket, &self.node_name).with_context(
186            || {
187                format!(
188                    "Unable to remove containerd containers on {}",
189                    self.node_name,
190                )
191            },
192        )?;
193
194        self.process.stop()
195    }
196}
