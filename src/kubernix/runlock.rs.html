1//! The lock of a run directory.
2//!
3//! The lock is an open file description lock on `kubernix.lock`, which
4//! contains the PID of the process which took it. The lock stays held
5//! while kubernix execs into its next stage (`nix develop` and the
6//! rootless session), since the next stage inherits the file descriptor
7//! and adopts the lock instead of taking a new one. Other processes can
8//! check the lock without taking it.
9//!
10//! Only the owner can open the lock file, since a process which is able to
11//! open it could take a read lock and block kubernix from starting. Other
12//! users check the lock via `/proc/locks` instead.
13

35/// The environment variable which passes the file descriptor of the lock
36/// to the next stage, which is internal and not part of the interface.
37const FD_ENV: &str = "_KUBERNIX_LOCK_FD";
38
39/// The locks of all processes.
40const PROC_LOCKS: &str = "/proc/locks";
41
42/// An exclusive lock of the run directory, released once the file
43/// descriptor gets closed in all processes which inherited it.
44pub(crate) struct RunLock {
45    /// The lock file, unless an ancestor holds the lock
46    file: Option<File>,
47}
48
49impl RunLock {
50    /// Take the lock of `path`, or adopt the lock inherited from the
51    /// previous stage. Returns `None` if another process holds the lock.
52    pub fn acquire(path: &Path) -> Result<Option<Self>> {
53        let inherited = std::env::var(FD_ENV).ok().and_then(|fd| fd.parse().ok());
54        Self::acquire_with(path, inherited)
55    }
56
57    fn acquire_with(path: &Path, inherited: Option<RawFd>) -> Result<Option<Self>> {
58        if let Some(lock) = inherited.and_then(|fd| Self::adopt(path, fd)) {
59            return Ok(Some(lock));
60        }
61
62        // rootlesskit does not pass the descriptor to its child, but keeps
63        // the lock held until the child exits.
64        if inherited.is_some()
65            && let Some(Some(holder)) = Self::holder(path)?
66            && ProcStat::is_ancestor(holder)
67        {
68            return Ok(Some(Self { file: None }));
69        }
70
71        let mut file = open_lock_file(path)?;
72        if !try_lock(&file, libc::F_WRLCK)
73            .with_context(|| format!("Unable to lock '{}'", path.display()))?
74        {
75            return Ok(None);
76        }
77        file.set_len(0)
78            .and_then(|()| write!(file, "{}", id()))
79            .with_context(|| format!("Unable to write lock file '{}'", path.display()))?;
80        Ok(Some(Self { file: Some(file) }))
81    }
82
83    /// Adopt the inherited file descriptor if it refers to the lock file.
84    fn adopt(path: &Path, fd: RawFd) -> Option<Self> {
85        if fd < 0 {
86            return None;
87        }
88        let expected = fs::metadata(path).ok()?;
89        // SAFETY: F_GETFD only checks whether the descriptor is open.
90        if unsafe { libc::fcntl(fd, libc::F_GETFD) } < 0 {
91            return None;
92        }
93        // SAFETY: the descriptor is open, and the file is not dropped, so
94        // that a foreign descriptor stays open.
95        let actual = ManuallyDrop::new(unsafe { File::from_raw_fd(fd) })
96            .metadata()
97            .ok()?;
98        if actual.ino() != expected.ino() || actual.dev() != expected.dev() {
99            return None;
100        }
101        // SAFETY: the descriptor is open, refers to the lock file and is
102        // owned by nobody else in this process, since it got inherited.
103        let file = unsafe { File::from_raw_fd(fd) };
104        // Do not pass it to the components, and take the lock again,
105        // which succeeds if the open file description already holds it.
106        fcntl(&file, FcntlArg::F_SETFD(FdFlag::FD_CLOEXEC)).ok()?;
107        try_lock(&file, libc::F_WRLCK)
108            .ok()?
109            .then_some(Self { file: Some(file) })
110    }
111
112    /// Pass the lock to the command, which replaces the current process.
113    pub fn inherit(&self, cmd: &mut Command) -> Result<()> {
114        let fd = match &self.file {
115            Some(file) => {
116                fcntl(file, FcntlArg::F_SETFD(FdFlag::empty()))
117                    .context("Unable to pass the lock file to the next stage")?;
118                file.as_raw_fd()
119            }
120            // An ancestor holds the lock
121            None => -1,
122        };
123        cmd.env(FD_ENV, fd.to_string());
124        Ok(())
125    }
126
127    /// Remove the variable which passes the lock from the command
128    /// environment, for commands which are not a stage of kubernix.
129    pub fn forget(cmd: &mut Command) {
130        cmd.env_remove(FD_ENV);
131    }
132
133    /// Check if a process holds the lock of `path` without taking it.
134    /// Returns `None` if the lock is free, otherwise the PID of the process
135    /// which took it, if known.
136    pub fn holder(path: &Path) -> Result<Option<Option<i32>>> {
137        let file = match File::open(path) {
138            Ok(file) => file,
139            Err(e) if e.kind() == ErrorKind::NotFound => return Ok(None),
140            // The lock file of another user, which is only readable by
141            // its owner
142            Err(e) if e.kind() == ErrorKind::PermissionDenied => {
143                return Ok(Self::locked(path)?.then_some(None));
144            }
145            Err(e) => {
146                return Err(e)
147                    .with_context(|| format!("Unable to open lock file '{}'", path.display()));
148            }
149        };
150        let mut flock = whole_file_lock(libc::F_WRLCK);
151        fcntl(&file, FcntlArg::F_OFD_GETLK(&mut flock))
152            .with_context(|| format!("Unable to check the lock '{}'", path.display()))?;
153        if i32::from(flock.l_type) == libc::F_UNLCK {
154            return Ok(None);
155        }
156        Ok(Some(
157            fs::read_to_string(path)
158                .ok()
159                .and_then(|pid| pid.trim().parse().ok()),
160        ))
161    }
162
163    /// Check via `/proc/locks` whether a process holds a lock of `path`,
164    /// which works without opening it.
165    fn locked(path: &Path) -> Result<bool> {
166        let metadata = fs::metadata(path)
167            .with_context(|| format!("Unable to access lock file '{}'", path.display()))?;
168        let locks = fs::read_to_string(PROC_LOCKS)
169            .with_context(|| format!("Unable to read '{}'", PROC_LOCKS))?;
170        Ok(Self::locks_contain(&locks, metadata.dev(), metadata.ino()))
171    }
172
173    /// Returns true if the content of `/proc/locks` contains a granted lock
174    /// of the file with the device and inode number. The lines look like
175    /// `1: OFDLCK ADVISORY WRITE -1 00:1d:1234 0 EOF`, where blocked
176    /// requests have a `->` after the number.
177    fn locks_contain(locks: &str, dev: u64, ino: u64) -> bool {
178        let id = format!("{:02x}:{:02x}:{}", libc::major(dev), libc::minor(dev), ino);
179        locks.lines().any(|line| {
180            let fields: Vec<&str> = line.split_whitespace().collect();
181            fields.get(1) != Some(&"->") && fields.get(5) == Some(&id.as_str())
182        })
183    }
184}

197    /// Wait until the lock is free. Child processes which other tests fork
198    /// concurrently hold a copy of the descriptor until they exec.
199    pub(crate) fn released(path: &Path) -> Result<bool> {
200        let deadline = Instant::now() + Duration::from_secs(5);
201        while RunLock::holder(path)?.is_some() {
202            if Instant::now() >= deadline {
203                return Ok(false);
204            }
205            sleep(Duration::from_millis(10));
206        }
207        Ok(true)
208    }

210    fn file(lock: &RunLock) -> Result<&File> {
211        lock.file.as_ref().context("no lock file")
212    }

214    /// A duplicate of the descriptor, which shares the open file
215    /// description like an inherited descriptor after exec.
216    fn dup(file: &File) -> Result<RawFd> {
217        Ok(fcntl(file, FcntlArg::F_DUPFD_CLOEXEC(0))?)
218    }

220    #[test]
221    fn acquire_is_exclusive() -> Result<()> {
222        let dir = tempdir()?;
223        let path = dir.path().join("lock");
224        assert_eq!(RunLock::holder(&path)?, None);
225
226        let lock = RunLock::acquire_with(&path, None)?.context("not acquired")?;
227        assert!(RunLock::acquire_with(&path, None)?.is_none());
228        assert_eq!(RunLock::holder(&path)?, Some(Some(i32::try_from(id())?)));
229
230        drop(lock);
231        assert!(released(&path)?);
232        assert!(RunLock::acquire_with(&path, None)?.is_some());
233        Ok(())
234    }

236    #[test]
237    fn acquire_restricts_lock_file() -> Result<()> {
238        let dir = tempdir()?;
239        let path = dir.path().join("lock");
240        fs::write(&path, "")?;
241        fs::set_permissions(&path, Permissions::from_mode(0o644))?;
242        let lock = RunLock::acquire_with(&path, None)?.context("not acquired")?;
243        assert_eq!(fs::metadata(&path)?.mode() & 0o777, 0o600);
244        drop(lock);
245
246        // Symbolic links are not followed
247        let link = dir.path().join("link");
248        std::os::unix::fs::symlink(&path, &link)?;
249        assert!(RunLock::acquire_with(&link, None).is_err());
250        Ok(())
251    }

265    #[test]
266    fn locks_contain_granted_locks() {
267        let locks = "1: OFDLCK ADVISORY  WRITE -1 00:1d:1234 0 EOF\n\
268                     1: -> OFDLCK ADVISORY  WRITE -1 fd:01:99 0 EOF\n\
269                     2: POSIX  ADVISORY  READ 42 fd:01:5 0 EOF\n";
270        let dev = |major, minor| libc::makedev(major, minor);
271        assert!(RunLock::locks_contain(locks, dev(0, 0x1d), 1234));
272        assert!(RunLock::locks_contain(locks, dev(0xfd, 1), 5));
273        // Blocked requests do not hold the lock
274        assert!(!RunLock::locks_contain(locks, dev(0xfd, 1), 99));
275        assert!(!RunLock::locks_contain(locks, dev(0, 0x1d), 123));
276        assert!(!RunLock::locks_contain("", dev(0, 0x1d), 1234));
277    }

294    #[test]
295    fn acquire_ignores_foreign_fd() -> Result<()> {
296        let dir = tempdir()?;
297        let path = dir.path().join("lock");
298        let other = File::create(dir.path().join("other"))?;
299        let lock = RunLock::acquire_with(&path, Some(other.as_raw_fd()))?;
300        assert!(lock.is_some());
301        // The foreign descriptor is still open
302        assert!(other.metadata().is_ok());

311    #[test]
312    fn inherit_passes_descriptor() -> Result<()> {
313        let dir = tempdir()?;
314        let lock =
315            RunLock::acquire_with(&dir.path().join("lock"), None)?.context("not acquired")?;
316        let flags = || -> Result<FdFlag> {
317            Ok(FdFlag::from_bits_truncate(fcntl(
318                file(&lock)?,
319                FcntlArg::F_GETFD,
320            )?))
321        };
322        assert!(flags()?.contains(FdFlag::FD_CLOEXEC));
323
324        let mut cmd = Command::new("true");
325        lock.inherit(&mut cmd)?;
326        assert!(!flags()?.contains(FdFlag::FD_CLOEXEC));
327        let fd = file(&lock)?.as_raw_fd().to_string();
328        assert!(
329            cmd.get_envs()
330                .any(|(k, v)| k == FD_ENV && v == Some(fd.as_ref()))
331        );
332
333        RunLock::forget(&mut cmd);
334        assert!(cmd.get_envs().any(|(k, v)| k == FD_ENV && v.is_none()));
335        Ok(())
336    }
