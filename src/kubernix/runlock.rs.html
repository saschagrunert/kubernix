1//! The lock of a run directory.
2//!
3//! The lock is an open file description lock on `kubernix.lock`, which
4//! contains the PID of the process which took it. The lock stays held
5//! while kubernix execs into its next stage (`nix develop` and the
6//! rootless session), since the next stage inherits the file descriptor
7//! and adopts the lock instead of taking a new one. Other processes can
8//! check the lock without taking it.
9
10use crate::process::ProcStat;
11use anyhow::{Context, Result};
12use nix::{
13    errno::Errno,
14    fcntl::{FcntlArg, FdFlag, fcntl},
15    libc,
16};
17use std::{
18    fs::{self, File, OpenOptions},

22        fd::{AsRawFd, FromRawFd, RawFd},
23        unix::fs::MetadataExt,
24    },
25    path::Path,
26    process::{Command, id},
27};

29/// The environment variable which passes the file descriptor of the lock
30/// to the next stage.
31const FD_ENV: &str = "KUBERNIX_LOCK_FD";
32
33/// An exclusive lock of the run directory, released once the file
34/// descriptor gets closed in all processes which inherited it.
35pub(crate) struct RunLock {
36    /// The lock file, unless an ancestor holds the lock
37    file: Option<File>,
38}
39
40impl RunLock {
41    /// Take the lock of `path`, or adopt the lock inherited from the
42    /// previous stage. Returns `None` if another process holds the lock.
43    pub fn acquire(path: &Path) -> Result<Option<Self>> {
44        let inherited = std::env::var(FD_ENV).ok().and_then(|fd| fd.parse().ok());
45        Self::acquire_with(path, inherited)
46    }
47
48    fn acquire_with(path: &Path, inherited: Option<RawFd>) -> Result<Option<Self>> {
49        if let Some(lock) = inherited.and_then(|fd| Self::adopt(path, fd)) {
50            return Ok(Some(lock));
51        }
52
53        // rootlesskit does not pass the descriptor to its child, but keeps
54        // the lock held until the child exits.
55        if inherited.is_some()
56            && let Some(Some(holder)) = Self::holder(path)?
57            && ProcStat::is_ancestor(holder)
58        {
59            return Ok(Some(Self { file: None }));
60        }
61
62        let mut file = OpenOptions::new()
63            .create(true)
64            .truncate(false)
65            .read(true)
66            .write(true)
67            .open(path)
68            .with_context(|| format!("Unable to open lock file '{}'", path.display()))?;
69        match fcntl(&file, FcntlArg::F_OFD_SETLK(&Self::flock(libc::F_WRLCK))) {
70            Ok(_) => {}
71            Err(Errno::EAGAIN | Errno::EACCES) => return Ok(None),
72            Err(e) => {
73                return Err(e).with_context(|| format!("Unable to lock '{}'", path.display()));
74            }
75        }
76        file.set_len(0)
77            .and_then(|()| write!(file, "{}", id()))
78            .with_context(|| format!("Unable to write lock file '{}'", path.display()))?;
79        Ok(Some(Self { file: Some(file) }))
80    }
81
82    /// Adopt the inherited file descriptor if it refers to the lock file.
83    fn adopt(path: &Path, fd: RawFd) -> Option<Self> {
84        if fd < 0 {
85            return None;
86        }
87        let expected = fs::metadata(path).ok()?;
88        // SAFETY: F_GETFD only checks whether the descriptor is open.
89        if unsafe { libc::fcntl(fd, libc::F_GETFD) } < 0 {
90            return None;
91        }
92        // SAFETY: the descriptor is open, and the file is not dropped, so
93        // that a foreign descriptor stays open.
94        let actual = ManuallyDrop::new(unsafe { File::from_raw_fd(fd) })
95            .metadata()
96            .ok()?;
97        if actual.ino() != expected.ino() || actual.dev() != expected.dev() {
98            return None;
99        }
100        // SAFETY: the descriptor is open, refers to the lock file and is
101        // owned by nobody else in this process, since it got inherited.
102        let file = unsafe { File::from_raw_fd(fd) };
103        // Do not pass it to the components, and take the lock again,
104        // which succeeds if the open file description already holds it.
105        fcntl(&file, FcntlArg::F_SETFD(FdFlag::FD_CLOEXEC)).ok()?;
106        fcntl(&file, FcntlArg::F_OFD_SETLK(&Self::flock(libc::F_WRLCK))).ok()?;
107        Some(Self { file: Some(file) })
108    }
109
110    /// Pass the lock to the command, which replaces the current process.
111    pub fn inherit(&self, cmd: &mut Command) -> Result<()> {
112        let fd = match &self.file {
113            Some(file) => {
114                fcntl(file, FcntlArg::F_SETFD(FdFlag::empty()))
115                    .context("Unable to pass the lock file to the next stage")?;
116                file.as_raw_fd()
117            }
118            // An ancestor holds the lock
119            None => -1,
120        };
121        cmd.env(FD_ENV, fd.to_string());
122        Ok(())
123    }
124
125    /// Remove the variable which passes the lock from the command
126    /// environment, for commands which are not a stage of kubernix.
127    pub fn forget(cmd: &mut Command) {
128        cmd.env_remove(FD_ENV);
129    }
130
131    /// Check if a process holds the lock of `path` without taking it.
132    /// Returns `None` if the lock is free, otherwise the PID of the process
133    /// which took it, if known.
134    pub fn holder(path: &Path) -> Result<Option<Option<i32>>> {
135        let file = match File::open(path) {
136            Ok(file) => file,
137            Err(e) if e.kind() == ErrorKind::NotFound => return Ok(None),
138            Err(e) => {
139                return Err(e)
140                    .with_context(|| format!("Unable to open lock file '{}'", path.display()));
141            }
142        };
143        let mut flock = Self::flock(libc::F_WRLCK);
144        fcntl(&file, FcntlArg::F_OFD_GETLK(&mut flock))
145            .with_context(|| format!("Unable to check the lock '{}'", path.display()))?;
146        if i32::from(flock.l_type) == libc::F_UNLCK {
147            return Ok(None);
148        }
149        Ok(Some(
150            fs::read_to_string(path)
151                .ok()
152                .and_then(|pid| pid.trim().parse().ok()),
153        ))
154    }
155
156    /// A lock description for the whole file.
157    fn flock(lock_type: libc::c_int) -> libc::flock {
158        // SAFETY: flock is a plain C struct, for which all zero is valid.
159        let mut flock: libc::flock = unsafe { std::mem::zeroed() };
160        flock.l_type = lock_type as libc::c_short;
161        flock.l_whence = libc::SEEK_SET as libc::c_short;
162        flock
163    }
164}

166#[cfg(test)]
167pub(crate) mod tests {
168    use super::*;
169    use std::{
170        thread::sleep,
171        time::{Duration, Instant},
172    };
173    use tempfile::tempdir;
174
175    /// Wait until the lock is free. Child processes which other tests fork
176    /// concurrently hold a copy of the descriptor until they exec.
177    pub(crate) fn released(path: &Path) -> Result<bool> {
178        let deadline = Instant::now() + Duration::from_secs(5);
179        while RunLock::holder(path)?.is_some() {
180            if Instant::now() >= deadline {
181                return Ok(false);
182            }
183            sleep(Duration::from_millis(10));
184        }
185        Ok(true)
186    }
187
188    fn file(lock: &RunLock) -> Result<&File> {
189        lock.file.as_ref().context("no lock file")
190    }
191
192    /// A duplicate of the descriptor, which shares the open file
193    /// description like an inherited descriptor after exec.
194    fn dup(file: &File) -> Result<RawFd> {
195        Ok(fcntl(file, FcntlArg::F_DUPFD_CLOEXEC(0))?)
196    }
197
198    #[test]
199    fn acquire_is_exclusive() -> Result<()> {
200        let dir = tempdir()?;
201        let path = dir.path().join("lock");
202        assert_eq!(RunLock::holder(&path)?, None);
203
204        let lock = RunLock::acquire_with(&path, None)?.context("not acquired")?;
205        assert!(RunLock::acquire_with(&path, None)?.is_none());
206        assert_eq!(RunLock::holder(&path)?, Some(Some(i32::try_from(id())?)));
207
208        drop(lock);
209        assert!(released(&path)?);
210        assert!(RunLock::acquire_with(&path, None)?.is_some());
211        Ok(())
212    }
213
214    #[test]
215    fn acquire_adopts_inherited_lock() -> Result<()> {
216        let dir = tempdir()?;
217        let path = dir.path().join("lock");
218        let lock = RunLock::acquire_with(&path, None)?.context("not acquired")?;
219        let adopted =
220            RunLock::acquire_with(&path, Some(dup(file(&lock)?)?))?.context("not adopted")?;
221
222        drop(lock);
223        assert!(RunLock::holder(&path)?.is_some());
224        drop(adopted);
225        assert!(released(&path)?);
226        Ok(())
227    }
228
229    #[test]
230    fn acquire_ignores_foreign_fd() -> Result<()> {
231        let dir = tempdir()?;
232        let path = dir.path().join("lock");
233        let other = File::create(dir.path().join("other"))?;
234        let lock = RunLock::acquire_with(&path, Some(other.as_raw_fd()))?;
235        assert!(lock.is_some());
236        // The foreign descriptor is still open
237        assert!(other.metadata().is_ok());
238
239        // Closed and invalid descriptors are ignored as well
240        drop(lock);
241        assert!(released(&path)?);
242        assert!(RunLock::acquire_with(&path, Some(-1))?.is_some());
243        Ok(())
244    }
245
246    #[test]
247    fn inherit_passes_descriptor() -> Result<()> {
248        let dir = tempdir()?;
249        let lock =
250            RunLock::acquire_with(&dir.path().join("lock"), None)?.context("not acquired")?;
251        let flags = || -> Result<FdFlag> {
252            Ok(FdFlag::from_bits_truncate(fcntl(
253                file(&lock)?,
254                FcntlArg::F_GETFD,
255            )?))
256        };
257        assert!(flags()?.contains(FdFlag::FD_CLOEXEC));
258
259        let mut cmd = Command::new("true");
260        lock.inherit(&mut cmd)?;
261        assert!(!flags()?.contains(FdFlag::FD_CLOEXEC));
262        let fd = file(&lock)?.as_raw_fd().to_string();
263        assert!(
264            cmd.get_envs()
265                .any(|(k, v)| k == FD_ENV && v == Some(fd.as_ref()))
266        );
267
268        RunLock::forget(&mut cmd);
269        assert!(cmd.get_envs().any(|(k, v)| k == FD_ENV && v.is_none()));
270        Ok(())
271    }
