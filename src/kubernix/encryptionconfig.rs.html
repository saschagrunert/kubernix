1//! Kubernetes encryption-at-rest configuration.
2//!
3//! Generates a random secretbox key and writes the EncryptionConfiguration
4//! manifest consumed by the API server's `--encryption-provider-config` flag.
5
6use crate::{Config, etcd::Etcd, files::open_nofollow, write_private};
7use anyhow::{Context, Result, bail};
8use base64::{Engine, engine::general_purpose::STANDARD};
9use log::{debug, info};
10use std::{
11    fs::create_dir_all,
12    io::Read,
13    path::{Path, PathBuf},
14};
15
16/// Encryption-at-rest configuration file for the API server.
17#[must_use]
18pub struct EncryptionConfig {
19    path: PathBuf,
20}
21
22impl EncryptionConfig {
23    /// Path to the generated EncryptionConfiguration YAML file.
24    pub fn path(&self) -> &Path {
25        &self.path
26    }
27
28    /// Generate the encryption config with a random secretbox key, or
29    /// reuse an existing one to support cluster restarts. A new key is
30    /// never generated if etcd contains data, since the secrets stored in
31    /// it would become unreadable.
32    pub fn new(config: &Config) -> Result<EncryptionConfig> {
33        let dir = &config.root().join("encryptionconfig");
34        create_dir_all(dir).context("Unable to create encryption config directory")?;
35        let path = dir.join("config.yml");
36
37        if let Err(e) = Self::check(&path) {
38            if Etcd::has_data(config) {
39                bail!(
40                    "The encryption config '{}' is missing or invalid ({:#}), but the secrets \
41                     in the etcd data '{}' are encrypted with its key. Restore it or remove the \
42                     run directory",
43                    path.display(),
44                    e,
45                    Etcd::data_dir(config).display(),
46                )
47            }
48            debug!("Encryption config is unusable: {:#}", e);
49            info!("Creating encryption config");
50            let mut rnd = [0u8; 32];
51            getrandom::fill(&mut rnd).context("Unable to generate encryption key")?;
52            let b64 = STANDARD.encode(rnd);
53            let yml = format!(include_str!("assets/encryptionconfig.yml"), secret = b64);
54            // Only the owner can read the key
55            write_private(&path, yml.as_bytes()).context("Unable to write encryption config")?;
56        }
57
58        Ok(EncryptionConfig { path })
59    }
60
61    /// Check that the config at `path` contains a valid key.
62    fn check(path: &Path) -> Result<()> {
63        let mut content = String::new();
64        open_nofollow(path)
65            .and_then(|mut f| f.read_to_string(&mut content))
66            .with_context(|| format!("Unable to read '{}'", path.display()))?;
67        let secret = content
68            .lines()
69            .find_map(|l| l.trim().strip_prefix("secret: "))
70            .context("no secret")?;
71        if STANDARD.decode(secret).context("invalid secret")?.len() != 32 {
72            bail!("invalid secret length")
73        }
74        Ok(())
75    }
76}

78#[cfg(test)]
79mod tests {
80    use super::*;
81    use crate::config::tests::{test_config, test_config_wrong_root};
82    use std::os::unix::fs::PermissionsExt;
83
84    #[test]
85    fn encryptionconfig_success() -> Result<()> {
86        let c = test_config()?;
87        let e = EncryptionConfig::new(&c)?;
88        let mode = std::fs::metadata(e.path())?.permissions().mode();
89        assert_eq!(mode & 0o777, 0o600);
90        let content = std::fs::read_to_string(e.path())?;
91        let secret = content
92            .lines()
93            .find_map(|l| l.trim().strip_prefix("secret: "))
94            .context("missing secret")?;
95        assert_eq!(STANDARD.decode(secret)?.len(), 32);
96        Ok(())
97    }
98
99    #[test]
100    fn encryptionconfig_reused() -> Result<()> {
101        let c = test_config()?;
102        let first = std::fs::read(EncryptionConfig::new(&c)?.path())?;
103        let second = std::fs::read(EncryptionConfig::new(&c)?.path())?;
104        assert_eq!(first, second);
105        Ok(())
106    }
107
108    #[test]
109    fn encryptionconfig_replaces_invalid() -> Result<()> {
110        let c = test_config()?;
111        let path = EncryptionConfig::new(&c)?.path().to_path_buf();
112        // Left behind by a crash of an older version
113        std::fs::write(&path, "")?;
114        let _ = EncryptionConfig::new(&c)?;
115        EncryptionConfig::check(&path)
116    }
117
118    #[test]
119    fn encryptionconfig_refuses_new_key_with_etcd_data() -> Result<()> {
120        let c = test_config()?;
121        let path = EncryptionConfig::new(&c)?.path().to_path_buf();
122        std::fs::create_dir_all(Etcd::data_dir(&c).join("member"))?;
123        let content = std::fs::read(&path)?;
124        let _ = EncryptionConfig::new(&c)?;
125        assert_eq!(std::fs::read(&path)?, content);
126
127        std::fs::remove_file(&path)?;
128        let err = EncryptionConfig::new(&c).err().context("no error")?;
129        assert!(err.to_string().contains("etcd data"), "{err}");
130        assert!(!path.exists());
131        Ok(())
132    }
133
134    #[test]
135    fn encryptionconfig_failure() -> Result<()> {
136        let c = test_config_wrong_root()?;
137        assert!(EncryptionConfig::new(&c).is_err());
138        Ok(())
139    }
