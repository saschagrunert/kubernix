1//! Kubernetes encryption-at-rest configuration.
2//!
3//! Generates a random secretbox key and writes the EncryptionConfiguration
4//! manifest consumed by the API server's `--encryption-provider-config` flag.
5
6use crate::Config;
7use anyhow::{Context, Result};
8use base64::{Engine, engine::general_purpose::STANDARD};
9use log::info;
10use rand::random;
11use std::{
12    fs::{self, create_dir_all},
13    os::unix::fs::PermissionsExt,
14    path::{Path, PathBuf},
15};
16
17#[must_use]
18pub struct EncryptionConfig {
19    path: PathBuf,
20}
21
22impl EncryptionConfig {
23    /// Path to the generated EncryptionConfiguration YAML file.
24    pub fn path(&self) -> &Path {
25        &self.path
26    }
27
28    /// Generate the encryption config with a random secretbox key, or
29    /// reuse an existing one to support cluster restarts.
30    pub fn new(config: &Config) -> Result<EncryptionConfig> {
31        let dir = &config.root().join("encryptionconfig");
32        create_dir_all(dir).context("Unable to create encryption config directory")?;
33        let path = dir.join("config.yml");
34
35        // Create only if not already existing to make cluster reuse work
36        if !path.exists() {
37            info!("Creating encryption config");
38            let rnd: [u8; 32] = random();
39            let b64 = STANDARD.encode(rnd);
40            let yml = format!(include_str!("assets/encryptionconfig.yml"), secret = b64);
41            fs::write(&path, yml).context("Unable to write encryption config")?;
42            fs::set_permissions(&path, fs::Permissions::from_mode(0o600))
43                .context("Unable to restrict encryption config permissions")?;
44        }
45
46        Ok(EncryptionConfig { path })
47    }
48}
